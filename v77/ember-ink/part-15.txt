function updateArtists(sc,time){

 for(const a of sc.npcs){
  const id=a.def?.id;if(!ARTISTS[id])continue;
  const sprite=a.sprite,off=a.inside||!sprite?.visible||!cam||sc.chat?.isOpen;
  let el=objects.activity.get(id);
  if(!el){el=sc.add.text(0,0,'',{fontFamily:'monospace',fontStyle:'bold',fontSize:'8px',color:'#ffe2b6',backgroundColor:'#243349d9',padding:{x:3,y:2}}).setOrigin(.5,1).setDepth(19);objects.activity.set(id,el);}

function connectedRoutes(a){if(!a?.isTileBlocked)return null;
