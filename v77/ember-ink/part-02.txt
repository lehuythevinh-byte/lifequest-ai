/*
 * Shrine cutscene: plays when the 20th finger is picked up (see fingers.js).
 *
 * 1. Four huge red parallelogram eyes fill a white, speckled screen.
 * 2. The camera pulls back: the eyes shrink up into a blood-red cloudy sky while the white clears, revealing the
 *    domain - a shrine rising out of cyan, rippling water with its reflection.
 * 3. Sukuna is a tiny figure on the shrine roof, dead centre between the eyes, so the domain feels vast around him.
 * 4. The picture holds for a moment, then fades out.
 *
 * It is drawn at 480 x 270 "pixels" and scaled up without smoothing for the pixel-art look.
 * window.LQShrine.play({ freeze }) returns a promise that resolves when the cutscene is over. `freeze` (ms) pins the
 * picture at that moment and never fades; it is only used to take screenshots while tuning the artwork.
 */
(function () {
  'use strict';

  const VW = 480;
  const VH = 270;
  const T_EYES = 600; // eyes fill the screen
  const T_PULL = 900; // camera pulls back, white clears
  const T_HOLD = 1100; // the domain, held
  const T_FADE = 1500; // fade out
  const RED = '#ed1c24';
  // Lead-in before the flash: the screen goes dark and a line is typed out letter by letter.
  const LEAD_TEXT = 'King of Curses has arrived';
  const LEAD_FONT = 'Cinzel, "Trajan Pro", "Times New Roman", Georgia, serif';
  const LEAD_FADE = 400; // fade to black
  const LEAD_CHAR = 55; // ms per letter
  const LEAD_HOLD = 800; // pause on the finished line before the cut
  const T_LEAD = LEAD_FADE + LEAD_TEXT.length * LEAD_CHAR + LEAD_HOLD;
  const HORIZON = 150;
  const CX = VW / 2;

  // The four eyes at full-screen size (upper pair large, lower pair smaller); right side is the mirror image.
  const EYES_BIG = [
    [[34, 52], [212, 96], [212, 150], [34, 106]],
    [[62, 176], [200, 176], [186, 236], [48, 236]],
  ];
  const EYES = [...EYES_BIG, ...EYES_BIG.map((e) => e.map(([x, y]) => [VW - x, y]))];
  const BIG_CENTER = [CX, 135];
  const SKY_CENTER = [CX, 50];
  const SKY_SCALE = 0.38;

  const ease = (x) => 1 - Math.pow(1 - Math.min(1, Math.max(0, x)), 3);
  const gauss = () => (Math.random() + Math.random() + Math.random() - 1.5) / 0.75;

  function seeded(seed) {
    let s = seed | 0;
    return () => {
      s = (s + 0x6d2b79f5) | 0;
      let t = Math.imul(s ^ (s >>> 15), 1 | s);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  function canvas(w, h) {
    const c = document.createElement('canvas');
    c.width = w;
    c.height = h;
    return c;
  }

  // ---- static backdrops, painted once ---------------------------------------------------------------------------
  function paintSky() {
    const c = canvas(VW, HORIZON);
    const g = c.getContext('2d');
    const rnd = seeded(7);
    const bands = ['#1a0610', '#220813', '#2b0a16', '#340c1a', '#3d0f1e', '#481222', '#531526'];
    const bh = HORIZON / bands.length;
    bands.forEach((col, i) => {
      g.fillStyle = col;
      g.fillRect(0, Math.floor(i * bh), VW, Math.ceil(bh) + 1);
    });
    const shades = ['#2a0b15', '#3a0f1c', '#56142a', '#6c1a2e', '#7d2234'];
    for (let i = 0; i < 130; i++) {
      const x = rnd() * VW;
      const y = rnd() * HORIZON;
      const rx = 18 + rnd() * 52;

        g.fill();

    return c;
  }

  function paintWater() {

    const rnd = seeded(21);

  function ripples(g, t) {
    const cy = BASE + 4;
    for (let i = 0; i < 7; i++) {
      const r = ((t * 0.028 + i * 17) % 119) + 4;

      for (let s = 0; s < steps; s++) {
        const ang = (s / steps) * Math.PI * 2;
        g.fillRect(Math.round(CX + Math.cos(ang) * r * 1.6), Math.round(cy + Math.sin(ang) * r * 0.34), 1, 1);
      }

      g.fillRect(Math.floor(rnd() * VW), HORIZON + 4 + Math.floor(rnd() * (VH - HORIZON - 4)), 1, 1);

  function ash(g, t) {

    for (let i = 0; i < 70; i++) {

      const y0 = rnd() * HORIZON;

      const y = (((y0 - t * sp) % HORIZON) + HORIZON) % HORIZON;

      g.fillRect(Math.floor(x0 + Math.sin(t / 700 + i) * 3), Math.floor(y), 1, 1);
    }
  }

  function scene(g, t, sky, water) {

  // ---- marks in the world -----------------------------------------------------------------------------------------
  // World position (pixels) and scale of a mark in what the player currently sees, or null when it is not here.
  function spotFor(sc, s) {
    if (s.area !== sc.area.id) return null;
    const ins = sc.inside;
    if (!s.in) return ins ? null : { x: s.x * 32 + 16, y: s.y * 32 + 16, k: 1 };
    if (!ins || ins.floor !== 0 || ins.b.name !== s.in || !ins.objs || !ins.objs[0]) return null;
    const unit = 32 * ins.k;
    const t = ins.spec;
    const at = (pt) => ({ x: ins.objs[0].x + (pt[0] + 0.5) * unit, y: ins.objs[0].y + (pt[1] + 0.5) * unit, k: ins.k });
    const margin = 12 * ins.k;
    const covered = (p) => (ins.rects || []).some((r) => p.x > r.x - margin && p.x < r.x + r.w + margin && p.y > r.y - margin && p.y < r.y + r.h + margin);
    // Try the guest tiles from the end so a mark does not share a tile with a finger (those use the first free one).
    const cands = [...(t.guests || []), [Math.floor(t.w / 2) + 1, Math.max(3, t.h - 4)]].map(at).reverse();
    return cands.find((p) => !covered(p)) || cands[0];
  }

  function drawMark(g, cx, cy, s, time, seed) {
    const pulse = 0.5 + 0.5 * Math.sin(time * 0.003 + seed);
    // Blue-violet glow that builds up towards the centre.
    for (let i = 0; i < 6; i++) {
      g.fillStyle(i % 2 ? 0x5a6cff : 0x8a4bff, 0.03 + pulse * 0.014);
      g.fillCircle(cx, cy, (22 - i * 3) * s);
    }
    // A dark pool on the ground with a slowly spreading ring.
    const ring = (time * 0.0008 + seed * 0.37) % 1;
    g.fillStyle(0x6a5cff, 0.2 * (1 - ring));
    g.fillEllipse(cx, cy + 5 * s, (14 + 16 * ring) * s, (5 + 6 * ring) * s);
    g.fillStyle(0x07070f, 0.85);
    g.fillEllipse(cx, cy + 5 * s, 18 * s, 7 * s);
    g.fillStyle(0x15152a, 0.9);
    g.fillEllipse(cx - 2 * s, cy + 4 * s, 9 * s, 3 * s);
  }

  function draw(sc, time) {
    const st = sc.__lqTenShadows;
    if (!st) return;
    st.g.clear();
    st.texts.forEach((t) => t.setVisible(false));
    if (!sc.area || sc.switching || !sc.player || !sc.store || !sc.store.game) return;
    const d = data(false);
    if (!d || !d.started) return;
    const px = sc.player.x;
    const py = sc.player.y;
    SHADOWS.forEach((s, i) => {
      // A mark exists in the world only after its villager told the clue, and until it is collected.
      if (!d.told[s.id] || d.found[s.id]) return;
      const p = spotFor(sc, s);
      if (!p) return;
      if (Math.hypot(px - p.x, py - p.y) < PICK_RADIUS * p.k) {
        collect(sc, s);
        return;
      }
      drawMark(st.g, p.x, p.y, p.k, time, i);
      const t = st.texts[i];
      t.setPosition(p.x, p.y - 4 * p.k + Math.sin(time * 0.002 + i) * 1.5 * p.k).setScale(p.k).setVisible(true);
    });
  }

  function attach(sc) {
    if (sc.__lqTenShadows) return;
    sc.__lqTenShadows = {
      g: sc.add.graphics().setDepth(8.8),
      texts: SHADOWS.map((s) => sc.add.text(0, 0, s.icon, { fontSize: '16px' }).setOrigin(0.5, 1).setDepth(8.9).setVisible(false)),
    };
    sc.events.on('update', (time) => draw(sc, time));
    sc.events.once('shutdown', () => {
      sc.__lqTenShadows = null;
    });
  }

  setInterval(() => {
    const sc = getScene();
    if (sc && sc.add && sc.events && sc.player && !sc.__lqTenShadows) {
      try {
        attach(sc);
      } catch (e) {
        console.warn('ten shadows attach failed', e);
      }
    }
  }, 1000);

  // All ten found but the achievement is not marked done (for example an older save): finish it quietly.
  setInterval(() => {
    const a = A();
    if (a && getScene() && count() >= SHADOWS.length && !a.isUnlocked(ACH_ID)) finish();
  }, 2000);

  window.LQTenShadows = {
    list: SHADOWS,
    total: SHADOWS.length,
    count,
    isFound,
    isTold,
    isStarted,
    status,
    onNpcMessage,
    drawMark,
    // Where a mark is in what the player sees right now ({x, y, k}) or null; for testing.
    spot: (id) => {
      const sc = getScene();
      const s = SHADOWS.find((x) => x.id === id);
      return sc && s ? spotFor(sc, s) : null;
    },
    // Handy while testing: LQTenShadows.reset() starts over (does not lock Megumi again).
    reset() {
      const d = data(false);
      if (d) {
        d.started = 0;
        d.told = {};
        d.found = {};
        save();
        refreshQuest();
      }
    },
  };
})();

  function rnd(seed) {
    let s = seed | 0;
    return () => {
      s = (s + 0x6d2b79f5) | 0;
      let t = Math.imul(s ^ (s >>> 15), 1 | s);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  // Stone wall: irregular blocks, a lit top, crenels on the outer rows and a shadow on the ground below.
  function wallTile(g, o) {
    const x = o.x * T;
    const y = o.y * T;
    const r = rnd(o.x * 131 + o.y * 977);
    const STONES = [0x8e897f, 0x98938a, 0x86817a, 0xa19c92];
    g.fillStyle(0x1a160f, 0.18).fillRect(x, y + T, T, 6);
    g.fillStyle(0x5d5850, 1).fillRect(x, y, T, T);
    const rows = [[1, 9], [11, 9], [21, 10]];
    rows.forEach(([ry, rh], i) => {
      let cx = x + (i % 2 ? -6 : 0);
      while (cx < x + T) {
        const sw = 9 + Math.floor(r() * 9);
        const sx = Math.max(x, cx + 1);
        const ex = Math.min(x + T, cx + sw - 1);
        if (ex - sx > 2) {
          const c = STONES[Math.floor(r() * STONES.length)];
          g.fillStyle(c, 1).fillRoundedRect(sx, y + ry, ex - sx, rh - 1, 2);
          g.fillStyle(shade(c, 1.12), 1).fillRect(sx + 1, y + ry, ex - sx - 2, 2);
          g.fillStyle(shade(c, 0.82), 1).fillRect(sx + 1, y + ry + rh - 3, ex - sx - 2, 2);
        }
        cx += sw;
      }
    });
    // walkway cap and crenels on top
    g.fillStyle(0xb2ada2, 1).fillRect(x, y - 3, T, 4);
    if ((o.x + o.y) % 2 === 0) {
      g.fillStyle(0x9a958b, 1).fillRect(x + 4, y - 10, T - 8, 8);
      g.fillStyle(0xb8b3a8, 1).fillRect(x + 4, y - 10, T - 8, 2);
    }
    // a little moss
    if (r() < 0.35) g.fillStyle(0x6f7d4c, 0.8).fillEllipse(x + r() * T, y + T - 3, 8, 4);
  }

    const GOLD = [0xc9a84d, 0xd8bb62, 0xb89640];
    for (let yy = y + 20; yy < y + h - 8; yy += 15)
      for (let xx = x + 14 + (Math.round((yy - y) / 15) % 2) * 6; xx < x + w - 12; xx += 13) {

  function building(g, b) {
    const x = b.x * T;
    const y = b.y * T;
    const w = b.w * T;
    const h = b.h * T;
    shadow(g, x + w / 2 + 8, y + h + 2, w + 20, 18, 0.26);
    if (b.kind === 'hall') return hall(g, b, x, y, w, h);
    if (b.kind === 'barn') return barn(g, b, x, y, w, h);
    if (b.kind === 'wood') return woodHouse(g, b, x, y, w, h);

  // ---- tall sprites: depth-sorted against the player ----------------------------------------------------------------
  // Each tall thing is baked once into its own texture (translateCanvas moves it to 0,0), then shown as an image.
  // Every frame: if the player's feet are above (north of) the thing's base, the thing is drawn over the player.
  const PLAYER_DEPTH = 10;
  function tallList(a) {
    const out = [];
    a.obstacles.forEach((o, i) => {
      if (!TALL.has(o.kind)) return;
      const x = o.x * T;
      const y = o.y * T;
      const w = o.w * T;
      const h = o.h * T;
      const pad = {
        tower: [16, 96, 16, 16],
        tree: [12, 30, 12, 14],
        well: [14, 40, 14, 22],
        banner: [4, 32, 14, 8],
        torch: [10, 24, 10, 8],
        stall: [10, 24, 10, 10],
        coop: [8, 4, 30, 18],
        rack: [6, 4, 6, 12],
      }[o.kind];
      out.push({ key: `past_o${i}_${o.kind}`, x0: x - pad[0], y0: y - pad[1], x1: x + w + pad[2], y1: y + h + pad[3], base: (o.y + o.h) * T, draw: (g) => drawThing(g, o, a) });
    });
    a.buildings.forEach((b, i) => {
      const x = b.x * T;
      const y = b.y * T;
      out.push({ key: `past_b${i}`, x0: x - 36, y0: y - 104, x1: x + b.w * T + 36, y1: y + b.h * T + 24, base: (b.y + b.h) * T, draw: (g) => building(g, b) });
    });
    return out;
  }
  function buildTall(sc) {
    const items = tallList(sc.area).map((t) => {
      const w = Math.ceil(t.x1 - t.x0);
      const h = Math.ceil(t.y1 - t.y0);
      if (!sc.textures.exists(t.key)) {
        const g = sc.add.graphics();
        g.translateCanvas(-t.x0, -t.y0);
        t.draw(g);
        g.generateTexture(t.key, w, h);
        g.destroy();
      }
      return { img: sc.add.image(t.x0, t.y0, t.key).setOrigin(0).setDepth(PLAYER_DEPTH - 0.6), base: t.base };
    });
    const st = { area: sc.area, items };
    st.update = () => {
      if (sc.area !== st.area) return clearTall(sc);
      if (!sc.player) return;
      const hide = !!sc.inside;
      const py = sc.player.y;
      for (const it of items) {
        it.img.setVisible(!hide);
        it.img.setDepth(py < it.base ? PLAYER_DEPTH + 0.6 : PLAYER_DEPTH - 0.6);
      }
    };
    sc.events.on('update', st.update);
    sc.__lqPastTall = st;
    st.update();
  }
  function clearTall(sc) {
    const st = sc && sc.__lqPastTall;
    if (!st) return;
    sc.events.off('update', st.update);
    st.items.forEach((it) => it.img.destroy());
    sc.__lqPastTall = null;
  }
  function syncTall(sc) {
    const st = sc.__lqPastTall;
    if (here(sc) && !sc.switching) {
      if (st && st.area !== sc.area) clearTall(sc);
      if (!sc.__lqPastTall) {
        try {
          buildTall(sc);
        } catch (e) {
          console.warn('past tall sprites failed', e);
          sc.__lqPastTall = { area: sc.area, items: [], update() {} };
        }
      }
    } else if (st) clearTall(sc);
  }

  // ---- the flock: hens, a rooster and chicks that wander, peck, and run away from the player ------------------------
  // Each group stays near its home (tile cx, cy, radius r). Drawn every frame into one Graphics just below the player.
  const FLOCKS = [
    { cx: 7, cy: 23.5, r: 2.8, hens: 5, rooster: true, chicks: 3 },
    { cx: 25, cy: 19.5, r: 2.6, hens: 3, chicks: 2 },
    { cx: 11, cy: 9.5, r: 2.2, hens: 2 },
    { cx: 21, cy: 29.5, r: 2, hens: 2, chicks: 2 },
  ];
  const HEN_SPEED = 26;
  const RUN_SPEED = 95;
  const SCARE = 46;
  function makeFlock(a) {
    const r = rnd(2024);
    const birds = [];
    const free = (x, y) => !a.isTileBlocked(Math.floor(x / T), Math.floor(y / T));
    const spot = (f) => {
      for (let i = 0; i < 40; i++) {
        const ang = r() * Math.PI * 2;
        const d = Math.sqrt(r()) * f.r * T;
        const x = (f.cx + 0.5) * T + Math.cos(ang) * d;
        const y = (f.cy + 0.5) * T + Math.sin(ang) * d;
        if (free(x, y)) return { x, y };
      }
      return { x: (f.cx + 0.5) * T, y: (f.cy + 0.5) * T };
    };
    FLOCKS.forEach((f) => {
      const add = (kind) => {
        const p = spot(f);
        birds.push({ kind, f, x: p.x, y: p.y, tx: p.x, ty: p.y, state: 'idle', t: r() * 2, face: r() < 0.5 ? -1 : 1, phase: r() * 10, flap: 0, mom: null });
      };
      for (let i = 0; i < f.hens; i++) add('hen');
      if (f.rooster) add('rooster');
      const moms = birds.filter((b) => b.f === f && b.kind === 'hen');
      for (let i = 0; i < (f.chicks || 0); i++) {
        add('chick');
        birds[birds.length - 1].mom = moms[i % moms.length] || null;
      }
    });
    return { birds, free, spot, r };
  }
  function henTick(fl, sc, dt) {
    const who = scarers(sc);
    for (const b of fl.birds) {
      b.t -= dt;
      b.phase += dt * (b.state === 'walk' || b.state === 'run' ? 14 : 3);
      b.flap = Math.max(0, b.flap - dt);
      // The nearest one who scares it: the player, or a running child.
      let px = who[0].x;
      let py = who[0].y;
      for (const s of who) {
        if (Math.hypot(b.x - s.x, b.y - s.y) < Math.hypot(b.x - px, b.y - py)) {
          px = s.x;
          py = s.y;
        }
      }
      const dp = Math.hypot(b.x - px, b.y - py);
      // Scared: run straight away from the player, flapping.
      if (dp < SCARE && b.state !== 'run') {
        const k = 70 / Math.max(1, dp);
        b.tx = b.x + (b.x - px) * k;
        b.ty = b.y + (b.y - py) * k;
        b.state = 'run';
        b.flap = 0.6;
        b.t = 1.2;
      }
      if (b.state === 'walk' || b.state === 'run') {
        const sp = (b.state === 'run' ? RUN_SPEED : HEN_SPEED) * (b.kind === 'chick' ? 1.15 : 1) * dt;
        const dx = b.tx - b.x;
        const dy = b.ty - b.y;
        const d = Math.hypot(dx, dy);
        if (Math.abs(dx) > 0.5) b.face = dx < 0 ? -1 : 1;
        if (d <= sp || b.t <= 0) {
          b.state = fl.r() < 0.6 ? 'peck' : 'idle';
          b.t = 0.8 + fl.r() * 2.2;
        } else {
          const nx = b.x + (dx / d) * sp;
          const ny = b.y + (dy / d) * sp;
          if (fl.free(nx, ny)) {
            b.x = nx;
            b.y = ny;
          } else {
            b.state = 'idle';
            b.t = 0.3;
          }
        }
      } else if (b.t <= 0) {
        // Pick the next place: chicks follow their mother, the others wander around home.
        let p;
        if (b.kind === 'chick' && b.mom) p = { x: b.mom.x + (fl.r() - 0.5) * 30, y: b.mom.y + 6 + (fl.r() - 0.5) * 16 };
        else p = fl.spot(b.f);
        b.tx = p.x;
        b.ty = p.y;
        b.state = 'walk';
        b.t = 6;
        if (b.kind === 'rooster' && fl.r() < 0.15) b.flap = 0.8;
      }
    }
  }
  function drawBird(g, b) {
    const f = b.face;
    const x = b.x;
    const walking = b.state === 'walk' || b.state === 'run';
    const bob = walking ? Math.abs(Math.sin(b.phase)) * 1.6 : 0;
    const y = b.y - bob;
    if (b.kind === 'chick') {
      g.fillStyle(0x1a160f, 0.2).fillEllipse(b.x, b.y + 4, 8, 3);
      g.fillStyle(0xf1d36a, 1).fillCircle(x, y, 3.6);
      g.fillStyle(0xf6df86, 1).fillCircle(x + f * 2.5, y - 3, 2.6);
      g.fillStyle(0xd9822e, 1).fillTriangle(x + f * 4.5, y - 3.5, x + f * 6.5, y - 3, x + f * 4.5, y - 2.3);
      g.fillStyle(0x2a241d, 1).fillRect(x + f * 3 - 0.5, y - 4, 1.2, 1.2);
      return;
    }
    const big = b.kind === 'rooster' ? 1.18 : 1;

  function faceTo(p, dx, dy) {
    const ax = Math.abs(dx);
    const ay = Math.abs(dy);
    if (ax < 0.01 && ay < 0.01) return;
    if (ay < ax * 0.4) {
      p.dir = 'side';
      p.flip = dx < 0;
    } else if (ax < ay * 0.4) p.dir = dy < 0 ? 'up' : 'down';
    else {
      p.dir = dy < 0 ? 'uside' : 'dside';
      p.flip = dx < 0;
    }
  }
  function folkSpot(st, home) {
    for (let k = 0; k < 30; k++) {
      const ang = st.r() * Math.PI * 2;
      const d = Math.sqrt(st.r()) * home[2] * T;
      const x = home[0] * T + Math.cos(ang) * d;
      const y = home[1] * T + Math.sin(ang) * d;
      if (st.free(x, y)) return { x, y };
    }
    return { x: home[0] * T, y: home[1] * T };
  }
  // Move p toward (tx, ty). Returns true when it got there (or is stuck).
  function folkStep(st, p, sp, px, py) {
    const dx = p.tx - p.x;
    const dy = p.ty - p.y;
    const d = Math.hypot(dx, dy);
    if (d <= sp) {
      p.x = p.tx;
      p.y = p.ty;
      return true;
    }
    const nx = p.x + (dx / d) * sp;
    const ny = p.y + (dy / d) * sp;
    faceTo(p, dx, dy);
    if (Math.hypot(nx - px, ny - py) < 14) return false; // never walk through the player: wait
    if (!st.free(nx, ny)) return true;
    p.x = nx;
    p.y = ny;
    p.walking = true;
    return false;
  }

    const g = sc.store && sc.store.game;
    const want = workMode(g && typeof g.minutes === 'number' ? g.minutes : 480);
    const [door, sx, sy] = v.work;
    if (!p.mode) {
      // First frame in the village: put them where the clock says, without walking there.
      p.mode = want;
      if (want === 'field') {
        p.x = p.tx = sx * T;
        p.y = p.ty = sy * T;
      } else if (want === 'home') {
        p.x = p.tx = (door + 0.5) * T;
        p.y = p.ty = DOOR_Y * T;
        p.gone = true;
      }
    }
    if (!p.route && p.mode !== want) {
      p.route = workRoute(p, p.mode, want);
      p.goal = want;
      p.gone = false; // morning: they come out of their door
      p.chaser = false;
    }
    p.away = !!p.route || p.mode !== 'village';
    if (p.route) {
      const pt = p.route[0];
      p.tx = pt[0] * T;
      p.ty = pt[1] * T;
      if (workStep(p, (WALK[v.kind] || 32) * dt, px, py)) {
        p.route.shift();
        if (!p.route.length) {
          p.route = null;
          p.mode = p.goal;
          if (p.mode === 'home') p.gone = true;
          if (p.mode === 'village') {
            p.away = false;
            p.leg = 0;
            p.t = 0;
          }
        }
      }
      return true;
    }
    if (p.mode === 'home') return true;
    if (p.mode === 'field') {
      // Planting: step slowly along the row, bending down again and again to push the seedlings in.
      p.plant = (p.plant || 0) + dt;
      p.x = sx * T + Math.sin(p.plant * 0.25 + p.i) * 0.9 * T;
      p.y = sy * T;
      p.dir = 'down';
      p.planting = true;
      return true;
    }
    return false;
  }

  function folkTick(st, sc, dt) {
    const px = sc.player.x;
    const py = sc.player.y;
    const folk = st.folk;
    const taggers = folk.filter((p) => p.v.play === 'tag' && !p.away);
    if (taggers.length && !taggers.some((p) => p.chaser)) taggers[0].chaser = true;
    for (const p of folk) {
      const v = p.v;
      p.walking = false;
      p.planting = false;
      if (v.after) {
        // Only home after the Day of Glory (the young soldier).
        const d = data();
        p.gone = !(d && d.done);
        if (p.gone) continue;
      }
      p.t -= dt;
      p.freeze = Math.max(0, p.freeze - dt);
      if (p.gone && p.mode === 'home' && workMode(sc.store.game.minutes) === 'home') continue;
      const dp = Math.hypot(p.x - px, p.y - py);
      // Grown-ups stop and look at the player who comes close; children keep playing.
      if (!isKid(v.kind) && dp < 48) {
        faceTo(p, px - p.x, py - p.y);
        continue;
      }
      if (workTick(sc, p, dt, px, py)) continue;
      if (v.play === 'tag') {
        const home = v.home;
        const sp = TAG_RUN * dt;
        if (p.freeze > 0) continue;
        if (p.chaser) {
          // Run after the nearest other child; touching them makes them "it".
          let best = null;
          let bd = 1e9;
          for (const o of taggers) {
            if (o === p || o.freeze > 0) continue;
            const d = Math.hypot(o.x - p.x, o.y - p.y);
            if (d < bd) {
              bd = d;
              best = o;
            }
          }
          if (!best) continue;
          if (bd < 11) {
            p.chaser = false;
            best.chaser = true;
            best.freeze = 1.1; // the new "it" counts a little before running
            p.freeze = 0.4;
            continue;
          }
          p.tx = best.x;
          p.ty = best.y;
          folkStep(st, p, sp * 0.92, px, py);
        } else {
          const ch = taggers.find((o) => o.chaser);
          const dc = ch ? Math.hypot(ch.x - p.x, ch.y - p.y) : 1e9;
          if (ch && dc < 56 && ch.freeze <= 0) {
            // Run away from "it", but stay in the square.
            let tx = p.x + ((p.x - ch.x) / Math.max(1, dc)) * 40;
            let ty = p.y + ((p.y - ch.y) / Math.max(1, dc)) * 40;
            if (Math.hypot(tx - home[0] * T, ty - home[1] * T) > home[2] * T || !st.free(tx, ty)) {
              const s = folkSpot(st, home);
              tx = s.x;
              ty = s.y;
            }
            if (p.t <= 0 || Math.hypot(p.tx - p.x, p.ty - p.y) < 4) {
              p.tx = tx;
              p.ty = ty;
              p.t = 0.6;
            }
            folkStep(st, p, sp, px, py);
          } else {
            if (p.t <= 0 || Math.hypot(p.tx - p.x, p.ty - p.y) < 3) {
              const s = folkSpot(st, home);
              p.tx = s.x;
              p.ty = s.y;
              p.t = 1 + st.r() * 1.5;
            }
            folkStep(st, p, sp * 0.55, px, py);
          }
        }
        continue;
      }
      const sp = (WALK[v.kind] || 32) * dt;
      if (v.follow != null) {
        // A little one next to her mother: a step behind and to the side.
        const m = folk[v.follow];
        p.tx = m.x + 10;
        p.ty = m.y + 6;
        if (Math.hypot(p.tx - p.x, p.ty - p.y) > 6) folkStep(st, p, sp * 1.3, px, py);
        else faceTo(p, m.x - p.x, m.y - p.y);
        continue;
      }
      if (v.path) {
        if (p.leg == null) p.leg = 0;
        if (p.t > 0) continue; // a short pause at each point
        const pt = v.path[p.leg];
        p.tx = pt[0] * T;
        p.ty = pt[1] * T;
        if (folkStep(st, p, sp, px, py)) {
          p.leg = (p.leg + 1) % v.path.length;
          p.t = 0.6 + st.r() * 1.8;
        }
        continue;
      }
      if (v.still) {
        // Stands at the spot (comes back to it if it was pushed away) and looks the same way.
        p.tx = v.home[0] * T;
        p.ty = v.home[1] * T;
        if (Math.hypot(p.tx - p.x, p.ty - p.y) > 2) folkStep(st, p, sp, px, py);
        else p.dir = v.still;
        continue;
      }
      // Wander around home, with pauses.
      if (p.t > 0 && Math.hypot(p.tx - p.x, p.ty - p.y) < 1) continue;
      if (p.t <= 0 && Math.hypot(p.tx - p.x, p.ty - p.y) < 1) {
        const s = folkSpot(st, v.home);
        p.tx = s.x;
        p.ty = s.y;
        p.t = 8;
      }
      if (folkStep(st, p, sp, px, py)) p.t = 1.5 + st.r() * 4;
    }
  }

  function folkUpdate(sc, delta) {
    let st = sc.__lqPastFolk;
    const kill = () => {
      if (!st) return;
      st.folk.forEach((p) => {
        p.img.destroy();
        if (p.bubble) p.bubble.destroy();
      });
      sc.__lqPastFolk = null;
    };
    if (!here(sc) || sc.switching || !sc.player) return kill();
    if (!st || st.area !== sc.area) {
      kill();
      const a = sc.area;
      st = sc.__lqPastFolk = { area: a, folk: [], r: rnd(29), free: (x, y) => !a.isTileBlocked(Math.floor(x / T), Math.floor(y / T)) };
      try {
        st.folk = makeFolk(sc);
      } catch (e) {
        console.warn('past villagers failed', e);
      }
    }
    const hide = !!sc.inside || filming;
    const dt = Math.min(0.05, (delta || 16) / 1000);
    if (!hide) folkTick(st, sc, dt);
    for (const p of st.folk) {
      p.img.setVisible(!hide && !p.gone);
      if (hide || p.gone) {
        if (p.bubble) p.bubble.setVisible(false);
        continue;
      }
      // Planting rice: bend down and up (origin is at the feet, so they stay on the ground).
      const h = HEIGHT[p.v.kind] || 1;
      p.img.setScale(h, p.planting ? h * (0.9 + 0.06 * Math.sin(p.plant * 2.4 + p.i)) : h);
      // Walk frames: 1 and 2 while moving (children step faster), 0 when standing.
      p.anim += dt * (p.walking ? (isKid(p.v.kind) ? 9 : 6) : 0);
      const frame = p.walking ? 1 + (Math.floor(p.anim) % 2) : 0;
      const tex = `ch_${p.key}_${p.dir}_${frame}`;
      if (p.img.texture.key !== tex) p.img.setTexture(tex);
      p.img.setFlipX(p.flip && p.dir !== 'down' && p.dir !== 'up');
      p.img.setPosition(p.x, p.y + FEET);
      p.img.setDepth(folkDepth(sc, p));
      // Talk once each time the player comes close (and stays a moment).
      const dp = Math.hypot(p.x - sc.player.x, p.y - sc.player.y);
      if (dp < 48) {
        p.near = (p.near || 0) + dt;
        if (!p.said && p.near > 0.5) {
          p.said = true;
          folkSay(sc, st, p);
        }
      } else if (dp > 70) {
        p.said = false;
        p.near = 0;
      }
      if (p.bubble) {
        p.bubbleT -= dt;
        if (p.bubbleT <= 0) {
          p.bubble.destroy();
          p.bubble = null;
        } else p.bubble.setVisible(true).setPosition(p.x, p.y + FEET - 38 * (HEIGHT[p.v.kind] || 1)).setAlpha(Math.min(1, p.bubbleT * 2));
      }
    }
  }
