  function wishBarriers(){
    const barriers=[];
    for(let y=0;y<WV_WORLD_GEOMETRY.height;y++){
      let begin=-1;
      for(let x=0;x<=WV_WORLD_GEOMETRY.width;x++){
        const blocked=x<WV_WORLD_GEOMETRY.width&&WV_WORLD_GEOMETRY.isWaterBlocked(x,y);
        if(blocked&&begin<0)begin=x;
        if(!blocked&&begin>=0){
          barriers.push({kind:WV_WORLD_GEOMETRY.isPond(begin,y)?'moonveil-water':'river',x:begin,y,w:x-begin,h:1});

        }
      }
    }
    return barriers;
  }

    const ctx=canvas.getContext('2d',{willReadFrequently:false});ctx.imageSmoothingEnabled=false;

    }

    }
