          function tick(scene, delta) {
            if (!scene || !scene.__lq9Glows || scene.switching) return;
            const q = quality();
            if (q === 0) {
              for (const g of scene.__lq9Glows) {
                g.a.setAlpha(0);
                g.b.setAlpha(0);
              }
              return;
            }
            scene.__lq9Tick = (scene.__lq9Tick || 0) + delta;
            if (scene.__lq9Tick < 90) return;
            scene.__lq9Tick = 0;
            const t = performance.now() / 1000;
            const minute = scene.store?.game?.minutes || 720;
            const dark = minute < 360 || minute > 1080;
            for (const g of scene.__lq9Glows) {
              const fade = g.interior ? 0.6 : dark ? 1 : 0.08;
              const wave = 0.87 + 0.13 * Math.sin(t * (g.color === 0xffad54 ? 5.5 : 2) + g.phase);
              g.a.setAlpha(fade * wave * (q === 2 ? 1 : 0.65));
              g.b.setAlpha(fade * (0.87 + 0.13 * Math.cos(t * 3 + g.phase)) * (q === 2 ? 1 : 0.65));
            }
            for (const p of scene.__lq9Particles || []) {
              const alive = p.interior || p.rocky || p.art || dark;
              p.node.setVisible(alive);
              if (!alive) continue;
              const sp = t * 0.65 + p.p;
              p.node.setPosition(p.x + Math.sin(sp) * 10, p.y + Math.cos(sp * 1.3) * 8);
              p.node.setAlpha((0.28 + 0.23 * Math.sin(sp * 2)) * (q === 2 ? 1 : 0.7));
            }
          }

  const items=sc.inside.spec.items ||[],unit=32*sc.inside.k;

  const allowed=new Set(['desk','table','shelf','counter','forge','stove','board','clock','paint','glasscase','rack','bed','altar','crystal','chair','schooldesk','glass','bookshelf']);
  let best=null,dist=Infinity;
  for(const item of items){if(!allowed.has(item.k))continue;
   const x=Yr+(item.x+item.w/2)*unit,y=Xr+(item.y+item.h/2)*unit;
   const d=Math.hypot(sc.player.x-x,sc.player.y-y);

  }
  return best;
 }
