  function v26AttachTownspeople(sc){
    // Re-create ambient sprites on each visit. The Phaser scene object can be
    // reused after switchArea() destroys its previous area objects.
    const previous=v26AmbientActors.get(sc)||[];

      if(old.label?.active)old.label.destroy();
    }
    v26AmbientActors.delete(sc);
    sc.__v26LastNPCFrame=null;
    const actors=[];
    for(const [i,t] of V26_WALKERS.entries()){
      const [tx,ty]=t.at;
      if(!inBounds(sc.area,tx,ty))continue;
      const xy=techPos(tx,ty);
      const sprite=sc.add.image(xy.x,xy.y,v26TownspersonTexture(sc,t.role,'down',0))
        .setOrigin(.5,.93).setDepth(9);
      const label=sc.add.text(xy.x,xy.y-51,t.name,{

        backgroundColor:'#3c3743cc',padding:{x:5,y:2}
      }).setOrigin(.5,1).setDepth(12).setVisible(false);
      sc.areaObjs.push(sprite,label);
      actors.push({t,x:xy.x,y:xy.y,sprite,label,face:'down',phase:i,route:0,

    for(const a of actors){
      a.sprite.setVisible(show);a.label.setVisible(show&&Math.hypot(px-a.x,py-a.y)<105);
      if(!show)continue;

    const agents=getRoyalAgents(sc);if(!agents||sc.switching)return;

    const dt=Math.min(80,Math.max(0,time-(sc.__royalNpcLastTime??time)));
    sc.__royalNpcLastTime=time;
    const minutes=Number(sc.store?.game?.minutes??480);
    for(const a of agents.values()){
      const list=ROYAL_SCHEDULE[a.t.id];
      const slot=list.find(v=>minutes<v.until)||list[list.length-1];
      if(a.site!==slot.site||a.lastSchedule!==slot.until){
        a.site=slot.site;a.lastSchedule=slot.until;
        a.path=actorPath(sc,a,...slot.tile);
      }
      const chatting=isDialogOpen()&&activeTask?.id===a.t.id;
      if(chatting)a.mode='TALK';
      else if(a.path.length){
        a.mode='WALK';const goal=a.path[0];
        const dx=goal.x-a.x,dy=goal.y-a.y,d=Math.hypot(dx,dy),speed=44*dt/1000;
        if(d<speed+.6){a.x=goal.x;a.y=goal.y;a.path.shift();}
        else {a.x+=dx/d*speed;a.y+=dy/d*speed;}
        a.face=Math.abs(dx)>Math.abs(dy)?dx>0?'right':'left':dy>0?'down':'up';
      }else{
        a.mode=slot.activity;
        if(time-a.lastWander>11000){
          a.lastWander=time+(a.t.id.length*179);
          const px=Math.floor(a.x/T),py=Math.floor(a.y/T);

        }
