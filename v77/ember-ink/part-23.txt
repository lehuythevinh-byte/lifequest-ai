        function aiShopping(text) {
          const t = norm(text),
            ctx = context(),
            nearShop = ctx.inside && shops[ctx.inside] ? ctx.inside : null;
          const m = t.match(/(?:mua|buy|order|purchase)\s+(?:cho toi|me|a|an|mot|one)?\s*(.*)/);
          if (m) {
            const id = Object.entries(catalog).find(
              ([key, it]) =>
                norm(it.name).includes(m[1]) ||
                norm(m[1]).includes(norm(it.name).replace(/^[^\p{L}]+/u, '')) ||
                key === norm(m[1]).replace(/\s+/g, '_'),
            )?.[0];
            if (id) {
              const it = catalog[id],
                where = nearShop || it.shops[0];
              chosenShop = where;
              show('shop', where);
              return `🛒 ${it.name} costs $${it.price}, available at ${it.shops.join(', ')}. ${canShop(where) ? 'Pick the item and press Buy to confirm the purchase.' : 'Go to the shop first; the AI never deducts money by itself.'}`;
            }
          }
          if (
            /(toi co|i have|ngan sach|budget|afford|mua duoc gi)/.test(t) &&
            /(\$|\b\d+\b|mua|buy|budget)/.test(t)
          ) {
            const n = Number(t.match(/\d+/)?.[0]) || money();
            const results = Object.values(catalog)
              .filter((x) => x.price <= n)
              .slice(0, 8)
              .map((x) => `${x.name} $${x.price}`)
              .join(' · ');
            return `💰 With $${n} you could consider: ${results}. Cash in your wallet: $${money()}. Open the Shop (B) to check prices and confirm a purchase.`;
          }
          if (/(kiem tien|lam viec|earn money|find a job)/.test(t))
            return '💼 Earn money from NPC quests or a once-a-day job at Sunny Cafe, Library, Post Office or Smithy. Go to the matching building and open the Economy window (B), then Inventory → Work.';
          if (/(tiet kiem|ngan hang|bank|deposit|withdraw|saving)/.test(t)) {
            show('bank');
            return `🏦 You have $${money()} in your wallet and $${state()?.bank || 0} in savings. Go to the Bank to deposit/withdraw. Saving does not create extra virtual money.`;
          }
          if (/(gia|bao nhieu|how much|price|cheaper|discount)/.test(t)) {
            const sample = Object.values(catalog)
              .slice(0, 4)
              .map((x) => `${x.name}: $${x.price}`)
              .join(', ');
            return `🛍️ Prices are fixed in shops, for example: ${sample}. Try asking “How much does it cost?” or “Do you have anything cheaper?”`;
          }
          return null;
        }

              render();
              return;

              return;

              return;
            }

              openLesson(b.dataset.v10Lesson);

              answerLesson(b.dataset.v10Answer);

              notice(bankAction(a, Number(value)));

              const n = Number($('v10-goal-amount').value);
              if (!validInt(n, 1, 1e6)) {
                notice('The goal must be a valid amount.');
                return;
              }
              state().goal = { name: $('v10-goal-name').value.slice(0, 60), amount: n };
              save();
              notice('🎯 Savings goal saved.');

                'To get a job, say a sentence in English (for example: Could I help you with your work, please?)',

              notice('Room updated.');

<script>
/*
 * LifeQuest loader.
 * The game used to be one huge function inside index.html. It is now split into small files under js/game/.
 * The files are plain slices of that function body, so they share one scope: we download them in order,
 * join them and run them as a single function. (This needs http://, i.e. localhost or GitHub Pages — not file://.)
 */
(function () {
  const startupInfo = document.getElementById('lq-loading');

  function displayError(message) {
    startupInfo.hidden = false;
    startupInfo.innerHTML =
      '<strong>Could not load the game</strong><p>' +
      message +
      '</p><small>Reload the page. The Phaser 4.2.1 graphics library is loaded from vendor/phaser.min.js (or a CDN if that file is missing).</small>';
  }

  function loadScript(src) {
    return new Promise((resolve, reject) => {
      const s = document.createElement('script');
      s.src = src;
      s.async = true;
      s.onload = () => resolve();
      s.onerror = () => reject(Error(src));
      document.head.appendChild(s);
    });
  }

  async function loadPhaser() {
    // The single-file build already embeds Phaser, so window.Phaser exists and nothing is downloaded.
    if (window.Phaser) return window.Phaser;
    for (const u of [
      // Local copy first (split/ and LIFEQUEST_deploy ship vendor/phaser.min.js), so the game starts without Internet.
      'vendor/phaser.min.js',
      'https://cdnjs.cloudflare.com/ajax/libs/phaser/4.2.1/phaser.min.js',
      'https://cdn.jsdelivr.net/npm/phaser@4.2.1/dist/phaser.min.js',
      'https://unpkg.com/phaser@4.2.1/dist/phaser.min.js',
    ]) {
      try {
        await loadScript(u);
        if (window.Phaser) return window.Phaser;
      } catch (e) {
        console.warn('Could not load Phaser from', u);
      }
    }
    throw Error('Could not load the Phaser library');
  }

  const PARTS = [
    'js/game/01-services.js',
    'js/game/02-core-npcs-and-schedules.js',
    'js/game/03-core-areas-and-map.js',
    'js/game/04-core-quest-helpers.js',
    'js/game/05a-core-npc-dialogue-profiles.js',
    'js/game/05b-core-npc-dialogue-profiles.js',
    'js/game/06-core-npc-chat-engine.js',
    'js/game/07a-core-ai-provider-and-world-art.js',
    'js/game/07b-core-ai-provider-and-world-art.js',
    'js/game/08-core-interiors.js',
    'js/game/09-core-graphics-v9.js',
    'js/game/10-core-characters-and-scene.js',
    'js/game/11-study-lab.js',
    'js/game/12-adventure-rpg.js',
    'js/game/13-assistant-core.js',
    'js/game/14-v8-quests-and-minimap.js',
    'js/game/15-ui-assistant-panel.js',
    'js/game/16-economy.js',
    'js/game/17-train-and-tickets.js',
    'js/game/18-economy-shop-and-npc-hooks.js',
    'js/game/19-menu-v11.js',
    'js/game/20-startup.js',
  ];

  async function loadGameSource() {
    // The single-file build (build-single.ps1) embeds the game source in the page.
    const embedded = document.getElementById('lq-game-src');
    if (embedded) return embedded.textContent + '\n//# sourceURL=lifequest-game.js';
    if (location.protocol === 'file:') {
      throw new Error(
        'Open the game over http (e.g. http://localhost:8080/) or GitHub Pages, not directly via file://.',
      );
    }
    const texts = await Promise.all(
      // 22 files at once: a simple local server (python -m http.server) sometimes drops a connection ("Failed to
      // fetch"), so each file is tried up to 3 times before giving up.
      PARTS.map(async (url) => {
        let last;
        for (let attempt = 0; attempt < 3; attempt++) {
          try {
            const res = await fetch(url, { cache: 'no-cache' });
            if (res.ok) return res.text();
            last = new Error('Could not load ' + url + ' (' + res.status + ')');
          } catch (e) {
            last = e;
          }
          await new Promise((r) => setTimeout(r, 300 * (attempt + 1)));
        }
        throw last;
      }),
    );
    return texts.join('\n') + '\n//# sourceURL=lifequest-game.js';
  }

  async function boot() {
    try {
      // Start downloading Phaser and the game while the title screen is shown.
      const phaserReady = loadPhaser();
      const sourceReady = loadGameSource();
      phaserReady.catch(() => {});
      sourceReady.catch(() => {});
      if (window.LQTitle) await window.LQTitle.show();
      // How the player plays (computer / phone / tablet): asked once, right after Start (device.js).
      if (window.LQDevice) await window.LQDevice.ask();
      if (window.LQAuth && !window.__lqVocaltPreviewMode) await window.LQAuth.start();
      const Phaser = await phaserReady;
      const source = await sourceReady;
      const AsyncFunction = Object.getPrototypeOf(async function () {}).constructor;
      const running = new AsyncFunction('Phaser', source)(Phaser);
      if (window.LQCreatorUI) window.LQCreatorUI.watch();
      await running;
    } catch (e) {
      console.error('LifeQuest HTML failed to start:', e);
      displayError('Could not start: ' + (e?.message || e));
    }
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot);
  else boot();
})();

function safeTile(sc,tx,ty,max=18){

function route(sc,start,end,limit=3200){if(!goodScene(sc)||!start||!end)return [];

 const queue=[origin],back=new Map([[origin,-1]]);let head=0,step=0;
 while(head<queue.length&&step++<limit){const k=queue[head++];if(k===goal){const nodes=[];let p=k;while(p!==origin&&nodes.length<140){nodes.push({x:((p%w)+.5)*T,y:(Math.floor(p/w)+.5)*T});p=back.get(p);if(p===undefined)return []}return nodes.reverse()}
  const x=k%w,y=Math.floor(k/w);
  for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){const nx=x+dx,ny=y+dy,kk=key(nx,ny);if(back.has(kk)||!openTile(sc,nx,ny))continue;back.set(kk,k);queue.push(kk)}

}
