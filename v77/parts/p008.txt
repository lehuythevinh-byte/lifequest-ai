  function royalRoadNetwork(g,routes){
    const drawing=(border)=>{
      for(const {path,width} of routes){
        const size=width+(border?9:0),color=border?0x9c876b:0xe6d2af;
        for(let i=1;i<path.length;i++){
          const a=path[i-1],b=path[i];
          rLine(g,color,a[0],a[1],b[0],b[1],size);
        }
        // Round the joints and termini to prevent broken joins on diagonals.
        for(const pt of path)rOval(g,color,pt[0],pt[1],size,size);
      }
    };
    drawing(true);
    drawing(false);
    // Cobble joints are intentionally small, offset and low-contrast.
    // There are no thick centre-lines crossing junctions or sharp end caps.
    for(const {path,width} of routes){
      for(let i=1;i<path.length;i++){
        const a=path[i-1],b=path[i],d=Math.hypot(b[0]-a[0],b[1]-a[1]);
        const n=Math.max(1,Math.floor(d/36));
        for(let j=1;j<n;j++){

          rRect(g,0x9f886d,x-4,y-2,8,2,.13);
        }
      }
    }
  }
