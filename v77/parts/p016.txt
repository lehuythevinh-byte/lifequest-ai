  function paintFloor(sc, a) {
    const key = 'lq_art_floor67';
    if (sc.textures.exists(key)) return key;
    const R = rnd(4242);
    const g = sc.make.graphics({ x: 0, y: 0, add: false });
    const pick = () => PAL[Math.floor(R() * PAL.length)];
    // 1. paint splatters on streets and squares (not on the buildings)
    for (let i = 0; i < 320; i++) {
      const x = R() * a.w;
      const y = R() * a.h;
      if (onBuilding(a, x, y, 0.4)) continue;
      const c = pick();
      const r = 3 + R() * 9;
      g.fillStyle(c, 0.18 + R() * 0.22).fillCircle(x * T, y * T, r);
      for (let k = 0; k < 5; k++) g.fillCircle(x * T + (R() - 0.5) * r * 4, y * T + (R() - 0.5) * r * 4, 1 + R() * 2.5);
    }
    // 2. paint dripping down from the bottom of every facade
    for (const b of a.buildings) {
      const by = (b.y + b.h) * T;
      for (let x = b.x * T + 6; x < (b.x + b.w) * T - 6; x += 7 + R() * 16) {
        const c = pick();
        const len = 6 + R() * 26;
        g.fillStyle(c, 0.75).fillRect(x, by - 2, 3, len);
        g.fillCircle(x + 1.5, by - 2 + len, 2.6);
      }
    }
    // 3. rainbow crosswalks where the streets cross
    for (const v of a.roadsV)
      for (const h of a.roadsH) {
        if (h.y < (v.y0 ?? 0) || h.y > (v.y1 ?? a.h)) continue;
        for (let i = 0; i < 6; i++) {
          g.fillStyle(PAL[i], 0.55).fillRect(v.x * T + 4, (h.y - 1) * T + 4 + i * 5, v.w * T - 8, 3);
          g.fillRect((v.x - 1) * T + 4 + i * 5, h.y * T + 4, 3, h.h * T - 8);
        }
      }
    // 4. little cyan stars leading from the tunnel up to the plaza
    for (let y = 47; y > 44; y -= 2) sparkle(g, 35.5 * T + (y % 4 ? 10 : -10), y * T, 6, 0x35c8d0, 0.85);
    // 5. the big floor mural of the three marks (scaled up around its centre)
    g.save();
    g.translateCanvas(MURAL.x, MURAL.y);
    g.scaleCanvas(MURAL.s, MURAL.s);
    mural(g, 0, 0);
    g.restore();
    // 6. V67 art sidewalks and planters; don't overpaint dedicated new facades.
    if (window.LQArtV67?.paintStreetDetails) window.LQArtV67.paintStreetDetails(g, a);
    else for (const b of a.buildings) graffiti(g, b, R);
    window.LQArtV69?.paintStreets?.(g,a);
    // 7. Fully accessible Living Canvas and the new walk-through Dream Gate.
    site(g, R);
    g.generateTexture(key, a.w * T, a.h * T);
    g.destroy();
    return key;
  }

  // ---- lights -----------------------------------------------------------------------------------------------------------
  // String lights hang over the streets, from post to post (every 6 tiles), with a little sag.
  function strings(a) {
    const out = [];
    const span = (x1, y1, x2, y2) => out.push([x1 * T, y1 * T, x2 * T, y2 * T]);
    for (const h of a.roadsH) for (let x = (h.x0 ?? 0) + 1; x <= (h.x1 ?? a.w) - 7; x += 6) span(x, h.y + 0.2, x + 6, h.y + 0.2);
    for (const v of a.roadsV) for (let y = (v.y0 ?? 0) + 3; y < (v.y1 ?? a.h) - 6; y += 6) span(v.x - 0.2, y, v.x - 0.2, y + 6);
    return out;
  }
  function makeLights(sc, a) {
    const st = {
      floor: sc.add.image(0, 0, paintFloor(sc, a)).setOrigin(0).setDepth(0.5),
      wires: sc.add.graphics().setDepth(12),
      bulbs: sc.add.graphics().setDepth(12.1).setBlendMode(ADD),
      glow: sc.add.graphics().setDepth(0.6).setBlendMode(ADD),
      motes: [],
      spans: strings(a),
      area: a,
    };
    const R = rnd(9);
    // the wires never change: draw them once
    for (const [x1, y1, x2, y2] of st.spans) {
      st.wires.lineStyle(1, 0x101018, 0.9);
      st.wires.beginPath();
      for (let k = 0; k <= 12; k++) {
        const t = k / 12;
        const sag = Math.sin(t * Math.PI) * 10;
        const x = x1 + (x2 - x1) * t + (x1 === x2 ? sag : 0);
        const y = y1 + (y2 - y1) * t + (y1 === y2 ? sag : 0);
        if (k) st.wires.lineTo(x, y);
        else st.wires.moveTo(x, y);
      }
      st.wires.strokePath();
    }
    for (let i = 0; i < 126; i++) st.motes.push({ x: R() * a.w * T, y: R() * a.h * T, v: 10 + R() * 22, c: MOTE_PAL[i % MOTE_PAL.length], p: R() * 6, r: 1.6 + R() * 1.9 });
    return st;
  }
  function drawLights(st, time, dt, cam) {
    const b = st.bulbs;
    b.clear();
    const view = cam.worldView;
    st.spans.forEach(([x1, y1, x2, y2], si) => {
      if (Math.max(x1, x2) < view.x - 40 || Math.min(x1, x2) > view.right + 40 || Math.max(y1, y2) < view.y - 40 || Math.min(y1, y2) > view.bottom + 40) return;
      for (let k = 1; k < 8; k++) {
        const t = k / 8;
        const sag = Math.sin(t * Math.PI) * 10;
        const x = x1 + (x2 - x1) * t + (x1 === x2 ? sag : 0);
        const y = y1 + (y2 - y1) * t + (y1 === y2 ? sag : 0) + 3;
        const c = PAL[(si + k) % PAL.length];
        const tw = 0.55 + 0.45 * Math.sin(time * 0.004 + si * 1.7 + k * 0.9);
        b.fillStyle(c, 0.12 * tw).fillCircle(x, y, 11);
        b.fillStyle(c, 0.5 + 0.5 * tw).fillCircle(x, y, 2.6);
      }
    });
    // specks of light drifting up, then starting again at the bottom
    for (const m of st.motes) {
      m.y -= m.v * dt;
      m.p += dt;
      if (m.y < 0) m.y = st.area.h * T;
      const x = m.x + Math.sin(m.p * 0.8) * 10;
      if (x < view.x || x > view.right || m.y < view.y || m.y > view.bottom) continue;
      const tw = 0.5 + 0.5 * Math.sin(m.p * 2.3);
      b.fillStyle(m.c, 0.14 * tw).fillCircle(x, m.y, 4 + m.r * 3);

/*

 * while the player's look has aura === 'infinity' (Gojo). It hooks into the scene's update loop once per scene.

(function () {
  'use strict';

  const getScene = () => (typeof window.__lifequestGetScene === 'function' ? window.__lifequestGetScene() : null);

  function draw(sc, time) {
    const st = sc.__lqAura;
    if (!st) return;

    g.clear();

    const p = sc.player;
    if (!look || look.aura !== 'infinity' || !p || !p.visible) return;

    if (st.lastOrigin === undefined || Math.abs(p.originY - st.lastOrigin) > 1e-6) st.baseOrigin = p.originY;
    const bob = Math.sin(time * 0.0016) * 1.5;

    p.setOrigin(0.5, st.lastOrigin);

    const pulse = 0.5 + 0.5 * Math.sin(time * 0.0025);

    const layers = 18;
    const breathe = 0.85 + 0.15 * pulse;

      const k = i / (layers - 1);

      g.fillEllipse(cx, cy, r * 1.5, r * 2.3);
