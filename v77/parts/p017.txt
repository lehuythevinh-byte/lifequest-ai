        }

    }
    if (placeGoal && !ins) {

      if (b) here.push({ kind: 'place', icon: '🏠', label: b.name, x: b.door * 32 + 16, y: (b.y + b.h) * 32 + 4, k: 1, wall: false, b });
    }
    for (const [id, why] of people) {
      const n = sc.npcs && sc.npcs.find((x) => x.def.id === id);
      if (!n) {
        away.otherTown++; // villagers of the other town are not in this scene
        continue;

      const name = (n.def.name || id).split(' (')[0];
      if (!n.sprite.visible) {

  // ---- drawing ------------------------------------------------------------------------------------------------
  // While the eyes are open the world sinks under a dark layer (depth DIM). Only the player and what the Six Eyes
  // see are lifted above it, each with a glowing outline that follows its shape; nothing is labelled.
  const DIM = 59;
  const DIM_ALPHA = 0.85; // how dark the world gets
  const D = { glow: 59.2, aura: 59.4, player: 59.5, npc: 59.6, npcLabel: 59.65, marks: 59.7, markText: 59.75, top: 59.8 };
  const BASE = { aura: 9.5, player: 10, npc: 9, npcLabel: 11, marks: 8.8, markText: 8.9 };

  // Lift (or put back) the game objects that should shine through the dark layer.
  function lift(sc, s, on, npcs) {
    const set = (o, d) => o && o.depth !== d && o.setDepth(d);
    set(sc.player, on ? D.player : BASE.player);
    set(sc.__lqAura && sc.__lqAura.g, on ? D.aura : BASE.aura);
    set(sc.__lqFingers && sc.__lqFingers.g, on ? D.marks : BASE.marks);
    const ts = sc.__lqTenShadows;
    if (ts) {
      set(ts.g, on ? D.marks : BASE.marks);
      ts.texts.forEach((t) => set(t, on ? D.markText : BASE.markText));
    }
    const want = new Set(on ? npcs : []);
    for (const n of s.lifted) if (!want.has(n)) set(n.sprite, BASE.npc), set(n.label, BASE.npcLabel);
    for (const n of want) set(n.sprite, D.npc), set(n.label, D.npcLabel);
    s.lifted = want;
  }

  // A pixel-crisp outline: four copies of the sprite, filled with one colour, nudged one pixel each way.
  function outline(sc, s, spr, color, alpha) {
    const off = Math.max(1, spr.scaleX || 1);
    for (const [dx, dy] of [[-off, 0], [off, 0], [0, -off], [0, off]]) {
      let img = s.glows[s.gi++];
      if (!img) s.glows.push((img = sc.add.image(0, 0, spr.texture.key).setDepth(D.glow)));
      img
        .setTexture(spr.texture.key, spr.frame.name)
        .setOrigin(spr.originX, spr.originY)
        .setScale(spr.scaleX, spr.scaleY)
        .setFlipX(spr.flipX)
        .setPosition(spr.x + dx, spr.y + dy)
        .setAlpha(alpha)
        .setVisible(true);
      fillTint(img, color);
    }
  }

  // Solid-colour tint: Phaser 4 uses setTint + FILL mode (setTintFill was removed); Phaser 3 has setTintFill.
  function fillTint(img, color) {
    const P = window.Phaser;
    if (img.setTintMode && P && P.TintModes) img.setTint(color).setTintMode(P.TintModes.FILL);
    else img.setTintFill(color);
  }

  // Soft glow with no hard edge: stacked faint discs.
  function halo(g, x, y, r, color, a) {
    for (let i = 0; i < 6; i++) {
      g.fillStyle(color, a);
      g.fillCircle(x, y, r * (1 - i * 0.14));
    }
  }

  function drawWorld(sc, s, seen, time, e) {
    const cam = sc.cameras.main;
    const v = cam.worldView;
    const z = cam.zoom || 1;
    const pulse = 0.5 + 0.5 * Math.sin(time * 0.005);
    s.dim.fillStyle(0x030812, DIM_ALPHA * e);
    s.dim.fillRect(v.x - 4, v.y - 4, v.width + 8, v.height + 8);
    const g = s.g;
    const top = s.top.setAlpha(e);
    s.gi = 0;
    const px = sc.player.x,
      py = sc.player.y;
    const onScreen = (t) => t.x > v.x && t.x < v.right && t.y > v.y && t.y < v.bottom;
    const inView = [],
      outView = [];
    for (const t of seen.here) {
      t.d = Math.hypot(t.x - px, t.y - py);
      (onScreen(t) || (t.b && t.b.x * 32 < v.right && (t.b.x + t.b.w) * 32 > v.x && t.b.y * 32 < v.bottom && (t.b.y + t.b.h) * 32 > v.y) ? inView : outView).push(t);
    }
    // The player keeps a faint blue rim so Gojo still stands out.
    outline(sc, s, sc.player, 0x8fe8ff, 0.55 * e);
    lift(sc, s, true, inView.filter((t) => t.npc).map((t) => t.npc));
    inView.forEach((t, i) => {
      const c = COLORS[t.kind];
      const k = t.k;
      if (t.kind === 'npc') {
        halo(g, t.npc.sprite.x, t.npc.sprite.y - 14 * k, 26 * k, c, 0.035 * e);
        outline(sc, s, t.npc.sprite, c, (0.7 + 0.3 * pulse) * e);
      } else if (t.kind === 'place' && t.b) {
        // The whole building is traced in light.
        const x = t.b.x * 32,
          y = t.b.y * 32,
          w = t.b.w * 32,
          h = t.b.h * 32;
        [[7, 0.12], [4, 0.25], [1.5, 0.9]].forEach(([lw, a]) => {
          g.lineStyle(lw, c, a * (0.75 + 0.25 * pulse) * e);
          g.strokeRect(x, y, w, h);
        });
      } else if (t.kind === 'place') {
        g.lineStyle(2, c, (0.6 + 0.4 * pulse) * e);
        g.strokeEllipse(t.x, t.y + 6, 34, 12);
        halo(g, t.x, t.y, 22, c, 0.04 * e);
      } else {
        halo(g, t.x, t.y, 30 * k, c, (0.05 + 0.03 * pulse) * e);
        if (t.wall) {
          // Seen through the roof: the object itself, drawn over the building.
          if (t.kind === 'finger' && window.LQFingers && window.LQFingers.draw) window.LQFingers.draw(top, t.x, t.y, 1, time, i);
          if (t.kind === 'shadow' && window.LQTenShadows && window.LQTenShadows.drawMark) window.LQTenShadows.drawMark(top, t.x, t.y, 1, time, i);
        }
      }
    });
    for (let i = s.gi; i < s.glows.length; i++) s.glows[i].setVisible(false);
    // Glowing chevrons on the edge of the view for what lies beyond it (nearest first, no text).
    const m = 22 / z;
    outView
      .sort((a, b) => a.d - b.d)
      .slice(0, MAX_ARROWS)
      .forEach((t) => {
        const c = COLORS[t.kind];
        const cx = Math.min(v.right - m, Math.max(v.x + m, t.x));
        const cy = Math.min(v.bottom - m, Math.max(v.y + m, t.y));
        const ang = Math.atan2(t.y - cy, t.x - cx);
        const chevron = (size, a) => {
          const L = size / z,
            W2 = size * 0.7 / z;
          const tipX = cx + Math.cos(ang) * L,
            tipY = cy + Math.sin(ang) * L;
          const bx = cx - Math.cos(ang) * L * 0.5,
            by = cy - Math.sin(ang) * L * 0.5;
          top.fillStyle(c, a);
          top.fillTriangle(tipX, tipY, bx + Math.cos(ang + Math.PI / 2) * W2, by + Math.sin(ang + Math.PI / 2) * W2, bx + Math.cos(ang - Math.PI / 2) * W2, by + Math.sin(ang - Math.PI / 2) * W2);
        };
        chevron(17, 0.15);
        chevron(13, 0.3);
        chevron(9, 0.95);
      });
  }

  // One short line: time left, and what is hidden elsewhere (indoors, outdoors or in the other town).
  function hudText(seen, now) {
    const left = Math.max(0, Math.ceil((st.until - now) / 1000));
    const a = seen.away;
    const bits = [];
    if (a.finger) bits.push(`👆${a.finger}`);
    if (a.shadow) bits.push(`🌑${a.shadow}`);
    if (a.npc.length + a.otherTown) bits.push(`🗣${a.npc.length + a.otherTown}`);
    return `<b>👁 ${left}s</b>${bits.length ? ` · elsewhere ${bits.join(' ')}` : ''}`;
  }

  let lastHud = 0;
  function frame(sc, time, delta) {
    const s = sc.__lqSixEyes;
    if (!s) return;
    s.g.clear();
    s.dim.clear();
    s.top.clear();

    const step = Math.min(1, (delta || 16) / 350);
    st.e = target > st.e ? Math.min(1, st.e + step) : Math.max(0, st.e - step);
    // Button label: ready / open / resting.
    const rest = now < st.readyAt ? Math.ceil((st.readyAt - now) / 1000) : 0;
    const label = st.active ? '👁 Six Eyes ✦' : rest ? `👁 ${rest}s` : '👁 Six Eyes';
    if (btn.textContent !== label) btn.textContent = label;
    btn.classList.toggle('on', st.active);
    btn.classList.toggle('rest', !st.active && !!rest);
    veil.style.opacity = String(st.e);
    // The eyes: blindfold off, glowing blue, while the Six Eyes are open.
    if (st.e > 0 && able) {
      if (s.look !== look) {
        s.look = look;
        s.key = window.LQCharacter.ensure(sc, Object.assign({}, look, { blindfold: false, eyeGlow: true, eye: EYE_BLUE }));
      }
      sc.playerKey = s.key;
      s.worn = true;
    } else if (s.worn) {
      s.worn = false;
      try {
        sc.applyRankLook();
      } catch (e) {
        /* the scene is being rebuilt */
      }
    }
    if (st.e <= 0 || !able) {
      if (s.on) {
        s.on = false;
        lift(sc, s, false, []);
        s.glows.forEach((img) => img.setVisible(false));
      }
      if (now < st.msgUntil) {
        hud.style.display = 'block';
        hud.innerHTML = st.msg;
      } else hud.style.display = 'none';
      return;
    }
    const seen = sense(sc);
    s.on = true;
    drawWorld(sc, s, seen, time, st.e);
    if (now - lastHud > 200) {
      lastHud = now;
      hud.style.display = 'block';
      hud.innerHTML = hudText(seen, now);
    }
  }

  function attach(sc) {
    if (sc.__lqSixEyes) return;
    sc.__lqSixEyes = {
      dim: sc.add.graphics().setDepth(DIM),
      g: sc.add.graphics().setDepth(D.glow),

      key: null,
      look: null,

    sc.events.on('update', (time, delta) => frame(sc, time, delta));
    sc.events.once('shutdown', () => {

    });
  }

        attach(sc);
      } catch (e) {
        console.warn('six eyes attach failed', e);
      }

  }, 1000);

  // Ctrl tap (pressed and released alone), the same key as Megumi's Divine Dog; each skill checks its own look.

  window.LQGojo = {
    greet,

    isSixEyesOpen: () => st.active,

/*
 * Device choice: right after Start on the title screen, the player must say how they play: 💻 Computer, 📱 Phone or
 * 📲 iPad / tablet. The likely answer is pre-selected (from the screen size and touch support), but nothing goes on
 * until the player confirms: there is no close button and Esc does nothing. The answer is remembered in localStorage
 * and only asked once; it can be changed later from the Menu (see LQDevice.change()).
 *
 * The choice is written to <html data-device="pc|phone|tablet"> so CSS and other scripts can adapt (touch controls
 * only appear on phone and tablet), and a 'lifequest:device' event is sent whenever it changes.
 *
 * window.LQDevice.ask() returns a promise that resolves once a device is chosen (at once if it was chosen before).
 */
(function () {
  'use strict';

  const KEY = 'lifequest.device';
  const DEVICES = [
    { id: 'pc', icon: '💻', title: 'Computer', desc: 'Keyboard: WASD or arrow keys to move, E to talk.' },
    { id: 'phone', icon: '📱', title: 'Phone', desc: 'On-screen joystick and buttons, sized for small screens.' },
    { id: 'tablet', icon: '📲', title: 'iPad / tablet', desc: 'On-screen joystick and larger buttons.' },
  ];
  const valid = (id) => DEVICES.some((d) => d.id === id);

  function read() {
    try {
      const v = localStorage.getItem(KEY);
      return valid(v) ? v : null;
    } catch (e) {
      return null;
    }
  }

  // Best guess, only used to pre-select an answer. The primary pointer decides (a touchscreen laptop still has a
  // mouse or trackpad as its main pointer); iPads that report themselves as a Mac are caught by 'MacIntel' + touch.
  function guess() {
    const mq = (q) => !!(window.matchMedia && matchMedia(q).matches);
    const ipadAsMac = navigator.platform === 'MacIntel' && (navigator.maxTouchPoints || 0) > 1;
    if (ipadAsMac) return 'tablet';
    if (!mq('(pointer: coarse)')) return 'pc';
    const short = Math.min(screen.width || innerWidth, screen.height || innerHeight);
    return short >= 600 ? 'tablet' : 'phone';
  }

  let current = read();
  function apply(id) {
    current = id;
    document.documentElement.dataset.device = id;
    window.dispatchEvent(new CustomEvent('lifequest:device', { detail: { device: id } }));
  }
  if (current) apply(current);

  const CSS = `
#lq-device{position:fixed;inset:0;z-index:260;display:flex;align-items:center;justify-content:center;padding:16px;box-sizing:border-box;background:#070a16f2;color:#eef2ff;font-family:Inter,system-ui,-apple-system,"Segoe UI",Roboto,sans-serif;animation:lqdev-in .35s ease}

#lq-device .opt:hover{border-color:#6b82c9}
#lq-device .opt.on{border-color:#ffd36b;background:#1e2748}
#lq-device .opt .ic{font-size:34px;line-height:1}
#lq-device .opt>div{display:flex;flex-direction:column;gap:4px}

        <p class="note">You can change this later in Menu → Settings.</p>
      </div>`;
      document.body.appendChild(el);
      let pick = pre;
      const opts = [...el.querySelectorAll('.opt')];
      const select = (id) => {
        pick = id;
        opts.forEach((o) => {
          o.classList.toggle('on', o.dataset.dev === id);
          o.setAttribute('aria-checked', String(o.dataset.dev === id));

      // Esc must not close it, and keys must not reach the game underneath.

        if (ev.key === 'Escape') ev.preventDefault();
        ev.stopPropagation();
      };
      addEventListener('keydown', onKey, true);
      go.addEventListener('click', () => {
        if (!valid(pick)) return;
        try {
          localStorage.setItem(KEY, pick);
        } catch (e) {
          /* private mode: the choice lasts for this visit only */
        }
        removeEventListener('keydown', onKey, true);
        el.remove();
        st.remove();
        apply(pick);
        resolve(pick);
      });
      setTimeout(() => go.focus(), 50);
    });
  }

  window.LQDevice = {
    list: DEVICES,
    get: () => current,
    guess,
    // First visit: ask (cannot be skipped). Later visits: resolve at once with the saved choice.
    // index.html?device=ask always asks (handy for testing the first-visit screen).
    ask: () => (current && !/[?&]device=ask\b/.test(location.search) ? Promise.resolve(current) : open()),
    // From the Menu: ask again, with the current choice pre-selected.
    change: () => open(),
    isTouch: () => current === 'phone' || current === 'tablet',
  };
})();
