var dr = class {
            primary;
            fallback;
            constructor(e, t) {
              ((this.primary = e), (this.fallback = t));
            }
            get label() {
              return this.primary.label;
            }
            // Rate-limit cool-down: after a 429 every Gemini call is skipped until this time (ms), so the free quota
            // is not burnt on requests that would fail anyway. NPCs answer offline meanwhile.
            coolUntil = 0;
            coolSecondsLeft() {
              return Math.max(0, Math.ceil((this.coolUntil - Date.now()) / 1e3));
            }
            noteError(err) {
              if (err && err.status === 429) {
                let secs = Math.min(120, Math.max(20, err.retryAfterSec || 30));
                this.coolUntil = Date.now() + secs * 1e3;
              }
            }
            // Only network problems, timeouts and server errors (5xx) are worth one more try. A 429, a bad key or a
            // blocked request fails again straight away and only costs another request.
            isRetryable(err) {
              let s = err && err.status;
              return !s || s >= 500;
            }
            async offlineWithNotice(e, t, n, r, i, notice) {
              let o = await this.fallback.respond(e, t, n, r, i);
              try {
                o && typeof o == `object` && (o.notice = notice + (o.notice ? ` | ${o.notice}` : ``));
              } catch {}
              return o;
            }
            restingNotice() {
              return `The AI is resting for about ${this.coolSecondsLeft()}s (free Gemini limit reached), so this reply is offline. You can keep chatting.`;
            }
            async respond(e, t, n, r, i) {
              if (this.coolSecondsLeft() > 0) return this.offlineWithNotice(e, t, n, r, i, this.restingNotice());
              try {
                return await this.primary.respond(e, t, n, r, i);
              } catch (first) {
                let err = first;
                if (this.isRetryable(first)) {
                  try {
                    await new Promise((e) => setTimeout(e, 500));
                    return await this.primary.respond(e, t, n, r, i);
                  } catch (second) {
                    err = second;
                  }
                }
                console.error(`[LIFEQUEST] Gemini error:`, err);
                this.noteError(err);
                let notice =
                  err && err.status === 429
                    ? this.restingNotice()
                    : `Gemini error: ${(err && err.message) || err}. Open Settings > Test to check your key.`;
                return this.offlineWithNotice(e, t, n, r, i, notice);
              }
            }
            async greet(e, t) {
              if (!this.primary.greet || this.coolSecondsLeft() > 0) return ``;
              try {
                return await this.primary.greet(e, t);
              } catch (err) {
                this.noteError(err);
                throw err;
              }
            }
            async summarize(e, t, n, r) {
              if (!this.primary.summarize || this.coolSecondsLeft() > 0) return ``;
              try {
                return await this.primary.summarize(e, t, n, r);
              } catch (err) {
                this.noteError(err);
                throw err;
              }
            }
            async check(e) {
              try {
                if (this.primary.check) return await this.primary.check(e);
              } catch {}
              return this.fallback.check(e);
            }
            explain(e) {
              return this.primary.explain(e);
            }
