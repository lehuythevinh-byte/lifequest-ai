  // Plate armour (no helmet) for Lord Gusest. The game's character art has no armour, so we draw him with marker colours
  // and repaint every frame: magenta top -> steel (same light and shade), green legs -> dark steel greaves,
  // a gold crest on the chest, and from behind a red cloak over the back.
  const ARMOR_DIRS = ['down', 'up', 'side', 'uside', 'dside'];
  function armorize(sc, key, inPlace) {
    const out = inPlace ? key : `${key}_armor`;
    for (const dir of ARMOR_DIRS) {
      for (let f = 0; f < 3; f++) {
        const src = `ch_${key}_${dir}_${f}`;
        const dst = `ch_${out}_${dir}_${f}`;
        if (!sc.textures.exists(src) || (!inPlace && sc.textures.exists(dst))) continue;
        const img = sc.textures.get(src).getSourceImage();
        const cv = document.createElement('canvas');
        cv.width = img.width;
        cv.height = img.height;
        const cx = cv.getContext('2d');
        cx.drawImage(img, 0, 0);
        const id = cx.getImageData(0, 0, cv.width, cv.height);
        const px = id.data;
        let x0 = 1e9;
        let x1 = -1;
        let y0 = 1e9;
        let y1 = -1;
        const back = dir === 'up';
        for (let k = 0; k < px.length; k += 4) {
          const R = px[k];
          const G = px[k + 1];
          const B = px[k + 2];
          if (!px[k + 3]) continue;
          const top = R > G + 60 && B > G + 60; // magenta marker (any shade)
          const leg = G > R + 60 && G > B + 60; // green marker
          if (!top && !leg) continue;
          const l = (top ? (R + B) / 2 : G) / 255; // keep the shading of the original
          if (top && back) {
            // the cloak, seen from behind
            px[k] = 60 + l * 90;
            px[k + 1] = 22 + l * 26;
            px[k + 2] = 24 + l * 26;
          } else if (top) {
            // darker steel than white cloth, so it still reads as metal under the sepia filter of the village
            px[k] = 58 + l * 78;
            px[k + 1] = 64 + l * 82;
            px[k + 2] = 78 + l * 92;
          } else {
            px[k] = 46 + l * 60;
            px[k + 1] = 50 + l * 62;
            px[k + 2] = 60 + l * 68;
          }
          if (top) {
            const x = (k / 4) % cv.width;
            const y = Math.floor(k / 4 / cv.width);
            x0 = Math.min(x0, x);
            x1 = Math.max(x1, x);
            y0 = Math.min(y0, y);
            y1 = Math.max(y1, y);
          }
        }
        cx.putImageData(id, 0, 0);
        if (x1 >= 0) {
          const w = x1 - x0 + 1;
          if (!back) {
            // the red cloak shows behind the arms, from the shoulders down to the knees
            cx.fillStyle = '#7d2e2e';
            cx.fillRect(x0 - 1, y0 + 1, 1, y1 - y0 + 4);
            cx.fillRect(x1 + 1, y0 + 1, 1, y1 - y0 + 4);
            // plate lines across the breastplate (lames), and a bright rim on top
            cx.fillStyle = '#2b3038';
            for (let y = y0 + 4; y < y1; y += 3) cx.fillRect(x0 + 1, y, w - 2, 1);
            cx.fillStyle = '#e6ebf2';
            cx.fillRect(x0 + 1, y0 + 1, w - 2, 1);
          }
          // pauldrons: round steel plates a little wider than the shoulders
          cx.fillStyle = '#8d96a3';
          cx.fillRect(x0 - 1, y0, 3, 3);
          cx.fillRect(x1 - 1, y0, 3, 3);
          cx.fillStyle = '#dfe5ec';
          cx.fillRect(x0 - 1, y0, 3, 1);
          cx.fillRect(x1 - 1, y0, 3, 1);
          cx.fillStyle = '#2b3038';
          cx.fillRect(x0 - 1, y0 + 3, 3, 1);
          cx.fillRect(x1 - 1, y0 + 3, 3, 1);
          const mx = Math.round((x0 + x1) / 2);
          if (dir === 'down' || dir === 'dside') {
            // the Oriaris crest: a little gold shield with a brown tower
            cx.fillStyle = '#e0b440';
            cx.fillRect(mx - 1, y0 + 2, 3, 3);

            cx.fillStyle = '#6b4a2a';

            cx.fillRect(x0, y1 + 1, x1 - x0 + 1, 4);
            cx.fillStyle = '#7d2e2e';

          }
        }
        if (inPlace) sc.textures.remove(src);
        sc.textures.addCanvas(dst, cv);
      }
    }
    return out;
  }

  function faceTo(p, dx, dy) {
    const ax = Math.abs(dx);
    const ay = Math.abs(dy);
    if (ax < 0.01 && ay < 0.01) return;
    if (ay < ax * 0.4) {
      p.dir = 'side';
      p.flip = dx < 0;
    } else if (ax < ay * 0.4) p.dir = dy < 0 ? 'up' : 'down';
    else {
      p.dir = dy < 0 ? 'uside' : 'dside';
      p.flip = dx < 0;
    }
  }
  function folkSpot(st, home) {
    for (let k = 0; k < 30; k++) {
      const ang = st.r() * Math.PI * 2;
      const d = Math.sqrt(st.r()) * home[2] * T;
      const x = home[0] * T + Math.cos(ang) * d;
      const y = home[1] * T + Math.sin(ang) * d;
      if (st.free(x, y)) return { x, y };
    }
    return { x: home[0] * T, y: home[1] * T };
  }
  // Move p toward (tx, ty). Returns true when it got there (or is stuck).
  function folkStep(st, p, sp, px, py) {
    const dx = p.tx - p.x;
    const dy = p.ty - p.y;
    const d = Math.hypot(dx, dy);
    if (d <= sp) {
      p.x = p.tx;
      p.y = p.ty;
      return true;
    }
    const nx = p.x + (dx / d) * sp;
    const ny = p.y + (dy / d) * sp;
    faceTo(p, dx, dy);
    if (Math.hypot(nx - px, ny - py) < 14) return false; // never walk through the player: wait
    if (!st.free(nx, ny)) return true;
    p.x = nx;
    p.y = ny;
    p.walking = true;
    return false;
  }
  // ---- the working day: children worked too in those days ---------------------------------------------------------------
  // Village clock (game minutes): 05:30–11:30 and 13:00–17:30 planting rice in the south paddies, back in the village at
  // noon and in the evening, and from 19:30 home: they walk to their own door and go inside (hidden) until morning.
  // They go out through the south gate and along the dike, so the routes never cross the wall.
  const HUB = [18.5, 30.5]; // just inside the south gate
  const DOOR_Y = 28.8; // in front of the cottage doors
  function workMode(min) {
    const m = ((min % 1440) + 1440) % 1440;
    if (m < 330 || m >= 1170) return 'home';
    if (m < 690 || (m >= 780 && m < 1050)) return 'field';
    return 'village';
  }
  function workBase(v) {
    if (v.home) return v.home;
    if (v.path) return v.path[0];
    return FOLK[v.follow].home;
  }
  function workRoute(p, from, to) {
    const [door, sx, sy] = p.v.work;
    const base = workBase(p.v);
    const low = (y) => y > 26.5; // south of the cottages: go along the lane behind them
    const out = [];
    if (from === 'field') out.push([sx, 41.5], [18.5, 41.5], [18.5, 33], HUB);
    else if (from === 'home') out.push([door + 0.5, 30.5], HUB);
    else out.push(low(p.y / T) ? [p.x / T, 30.5] : [18.5, 20.5], HUB);
    if (to === 'field') out.push([18.5, 33], [18.5, 41.5], [sx, 41.5], [sx, sy]);
    else if (to === 'home') out.push([door + 0.5, 30.5], [door + 0.5, DOOR_Y]);
    else out.push(low(base[1]) ? [base[0], 30.5] : [18.5, 20.5], [base[0], base[1]]);
    return out;
  }
  // Like folkStep, but on a planned route (gates, dikes): no wall check. Still never walks through the player.
  function workStep(p, sp, px, py) {
    const dx = p.tx - p.x;
    const dy = p.ty - p.y;
    const d = Math.hypot(dx, dy);
    if (d <= sp) {
      p.x = p.tx;
      p.y = p.ty;
      return true;
    }
    faceTo(p, dx, dy);
    const nx = p.x + (dx / d) * sp;
    const ny = p.y + (dy / d) * sp;
    if (Math.hypot(nx - px, ny - py) < 14) return false;
    p.x = nx;
    p.y = ny;
    p.walking = true;
    return false;
  }
  // Returns true when the schedule moved this villager (their usual village life is skipped this frame).
  function workTick(sc, p, dt, px, py) {
    const v = p.v;
    if (!v.work) return false;
    const g = sc.store && sc.store.game;
    const want = workMode(g && typeof g.minutes === 'number' ? g.minutes : 480);
    const [door, sx, sy] = v.work;
    if (!p.mode) {
      // First frame in the village: put them where the clock says, without walking there.
      p.mode = want;
      if (want === 'field') {
        p.x = p.tx = sx * T;
        p.y = p.ty = sy * T;
      } else if (want === 'home') {
        p.x = p.tx = (door + 0.5) * T;
        p.y = p.ty = DOOR_Y * T;
        p.gone = true;
      }
    }
    if (!p.route && p.mode !== want) {
      p.route = workRoute(p, p.mode, want);
      p.goal = want;
      p.gone = false; // morning: they come out of their door
      p.chaser = false;
    }
    p.away = !!p.route || p.mode !== 'village';
    if (p.route) {
      const pt = p.route[0];
      p.tx = pt[0] * T;
      p.ty = pt[1] * T;
      if (workStep(p, (WALK[v.kind] || 32) * dt, px, py)) {
        p.route.shift();
        if (!p.route.length) {
          p.route = null;
          p.mode = p.goal;
          if (p.mode === 'home') p.gone = true;
          if (p.mode === 'village') {
            p.away = false;
            p.leg = 0;
            p.t = 0;
          }
        }
      }
      return true;
    }
    if (p.mode === 'home') return true;
    if (p.mode === 'field') {
      // Planting: step slowly along the row, bending down again and again to push the seedlings in.
      p.plant = (p.plant || 0) + dt;
      p.x = sx * T + Math.sin(p.plant * 0.25 + p.i) * 0.9 * T;
      p.y = sy * T;
      p.dir = 'down';
      p.planting = true;
      return true;
    }
    return false;
  }

  function folkTick(st, sc, dt) {
    const px = sc.player.x;
    const py = sc.player.y;
    const folk = st.folk;
    const taggers = folk.filter((p) => p.v.play === 'tag' && !p.away);
    if (taggers.length && !taggers.some((p) => p.chaser)) taggers[0].chaser = true;
    for (const p of folk) {
      const v = p.v;
      p.walking = false;
      p.planting = false;
      if (v.after) {
        // Only home after the Day of Glory (the young soldier).
        const d = data();
        p.gone = !(d && d.done);
        if (p.gone) continue;
      }
      p.t -= dt;
      p.freeze = Math.max(0, p.freeze - dt);
      if (p.gone && p.mode === 'home' && workMode(sc.store.game.minutes) === 'home') continue;
      const dp = Math.hypot(p.x - px, p.y - py);
      // Grown-ups stop and look at the player who comes close; children keep playing.
      if (!isKid(v.kind) && dp < 48) {
        faceTo(p, px - p.x, py - p.y);
        continue;
      }
      if (workTick(sc, p, dt, px, py)) continue;
      if (v.play === 'tag') {
        const home = v.home;
        const sp = TAG_RUN * dt;
        if (p.freeze > 0) continue;
        if (p.chaser) {
          // Run after the nearest other child; touching them makes them "it".
          let best = null;
          let bd = 1e9;
          for (const o of taggers) {
            if (o === p || o.freeze > 0) continue;
            const d = Math.hypot(o.x - p.x, o.y - p.y);
            if (d < bd) {
              bd = d;
              best = o;
            }
          }
          if (!best) continue;
          if (bd < 11) {
            p.chaser = false;
            best.chaser = true;
            best.freeze = 1.1; // the new "it" counts a little before running
            p.freeze = 0.4;
            continue;
          }
          p.tx = best.x;
          p.ty = best.y;
          folkStep(st, p, sp * 0.92, px, py);
        } else {
          const ch = taggers.find((o) => o.chaser);
          const dc = ch ? Math.hypot(ch.x - p.x, ch.y - p.y) : 1e9;
          if (ch && dc < 56 && ch.freeze <= 0) {
            // Run away from "it", but stay in the square.
            let tx = p.x + ((p.x - ch.x) / Math.max(1, dc)) * 40;
            let ty = p.y + ((p.y - ch.y) / Math.max(1, dc)) * 40;
            if (Math.hypot(tx - home[0] * T, ty - home[1] * T) > home[2] * T || !st.free(tx, ty)) {
              const s = folkSpot(st, home);
              tx = s.x;
              ty = s.y;
            }
            if (p.t <= 0 || Math.hypot(p.tx - p.x, p.ty - p.y) < 4) {
              p.tx = tx;
              p.ty = ty;
              p.t = 0.6;
            }
            folkStep(st, p, sp, px, py);
          } else {
            if (p.t <= 0 || Math.hypot(p.tx - p.x, p.ty - p.y) < 3) {
              const s = folkSpot(st, home);
              p.tx = s.x;
              p.ty = s.y;
              p.t = 1 + st.r() * 1.5;
            }
            folkStep(st, p, sp * 0.55, px, py);
          }
        }
        continue;
      }
      const sp = (WALK[v.kind] || 32) * dt;
      if (v.follow != null) {
        // A little one next to her mother: a step behind and to the side.
        const m = folk[v.follow];
        p.tx = m.x + 10;
        p.ty = m.y + 6;
        if (Math.hypot(p.tx - p.x, p.ty - p.y) > 6) folkStep(st, p, sp * 1.3, px, py);
        else faceTo(p, m.x - p.x, m.y - p.y);
        continue;
      }
      if (v.path) {
        if (p.leg == null) p.leg = 0;
        if (p.t > 0) continue; // a short pause at each point
        const pt = v.path[p.leg];
        p.tx = pt[0] * T;
        p.ty = pt[1] * T;
        if (folkStep(st, p, sp, px, py)) {
          p.leg = (p.leg + 1) % v.path.length;
          p.t = 0.6 + st.r() * 1.8;
        }
        continue;
      }
      if (v.still) {
        // Stands at the spot (comes back to it if it was pushed away) and looks the same way.
        p.tx = v.home[0] * T;
        p.ty = v.home[1] * T;
        if (Math.hypot(p.tx - p.x, p.ty - p.y) > 2) folkStep(st, p, sp, px, py);
        else p.dir = v.still;
        continue;
      }
      // Wander around home, with pauses.
      if (p.t > 0 && Math.hypot(p.tx - p.x, p.ty - p.y) < 1) continue;
      if (p.t <= 0 && Math.hypot(p.tx - p.x, p.ty - p.y) < 1) {
        const s = folkSpot(st, v.home);
        p.tx = s.x;
        p.ty = s.y;
        p.t = 8;
      }
      if (folkStep(st, p, sp, px, py)) p.t = 1.5 + st.r() * 4;
    }
  }
  // Depth: in front of / behind the player, and behind a house or tree when standing north of its base.
  function folkDepth(sc, p) {
    let dep = (p.y < sc.player.y ? PLAYER_DEPTH - 0.1 : PLAYER_DEPTH + 0.1) + p.y * 1e-5;
    const tall = sc.__lqPastTall && sc.__lqPastTall.items;
    if (tall) {
      for (const it of tall) {
        const im = it.img;
        if (p.x < im.x || p.x > im.x + im.width || p.y + FEET < im.y || p.y > it.base + 24) continue;
        if (p.y < it.base) dep = Math.min(dep, im.depth - 0.05);
        else dep = Math.max(dep, im.depth + 0.05);
      }
    }
    return dep;
  }

  function folkUpdate(sc, delta) {
    let st = sc.__lqPastFolk;
    const kill = () => {
      if (!st) return;

      try {
        st.folk = makeFolk(sc);
      } catch (e) {
        console.warn('past villagers failed', e);
      }
    }
    const hide = !!sc.inside || filming;

    if (!hide) folkTick(st, sc, dt);

      p.img.setVisible(!hide && !p.gone);
      if (hide || p.gone) {
        if (p.bubble) p.bubble.setVisible(false);
        continue;
      }
      // Planting rice: bend down and up (origin is at the feet, so they stay on the ground).
      const h = HEIGHT[p.v.kind] || 1;
      p.img.setScale(h, p.planting ? h * (0.9 + 0.06 * Math.sin(p.plant * 2.4 + p.i)) : h);
      // Walk frames: 1 and 2 while moving (children step faster), 0 when standing.
      p.anim += dt * (p.walking ? (isKid(p.v.kind) ? 9 : 6) : 0);
      const frame = p.walking ? 1 + (Math.floor(p.anim) % 2) : 0;
      const tex = `ch_${p.key}_${p.dir}_${frame}`;
      if (p.img.texture.key !== tex) p.img.setTexture(tex);
      p.img.setFlipX(p.flip && p.dir !== 'down' && p.dir !== 'up');
      p.img.setPosition(p.x, p.y + FEET);
      p.img.setDepth(folkDepth(sc, p));
      // Talk once each time the player comes close (and stays a moment).
      const dp = Math.hypot(p.x - sc.player.x, p.y - sc.player.y);
      if (dp < 48) {
        p.near = (p.near || 0) + dt;
        if (!p.said && p.near > 0.5) {
          p.said = true;
          folkSay(sc, st, p);
        }
      } else if (dp > 70) {
        p.said = false;
        p.near = 0;
      }
      if (p.bubble) {
        p.bubbleT -= dt;
        if (p.bubbleT <= 0) {
          p.bubble.destroy();
          p.bubble = null;
        } else p.bubble.setVisible(true).setPosition(p.x, p.y + FEET - 38 * (HEIGHT[p.v.kind] || 1)).setAlpha(Math.min(1, p.bubbleT * 2));
      }
    }
