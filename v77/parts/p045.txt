        function getQuiz(level) {
          const q = samples[level] || samples.B1,
            raw = q[Math.floor(Math.random() * q.length)];
          const choices = [raw[1], raw[2], raw[3]];
          const answer = choices[{ a: 0, b: 1, c: 2 }[raw[4]]];
          for (let i = choices.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [choices[i], choices[j]] = [choices[j], choices[i]];
          }
          const item = [raw[0], ...choices, ['a', 'b', 'c'][choices.indexOf(answer)], raw[5]];
          return { item, tries: 0 };

              /^(tu vung|tra tu|nghia cua|what does|meaning of|vocabulary|explain word|luyen hoi thoai|roleplay|practice conversation|luyen noi|speaking)/.test(
                s,

        function sound(type = 'click') {
          if (!preferences.audio) return;
          try {
            const C = window.AudioContext || window.webkitAudioContext;
            if (!C) return;
            audioCtx = audioCtx || new C();
            if (audioCtx.state === 'suspended') audioCtx.resume();
            const osc = audioCtx.createOscillator(),
              gain = audioCtx.createGain(),
              t = audioCtx.currentTime;
            const hz = type === 'door' ? 440 : type === 'quest' ? 690 : type === 'step' ? 185 : 520;
            osc.type = type === 'step' ? 'triangle' : 'sine';
            osc.frequency.setValueAtTime(hz, t);
            if (type === 'quest') osc.frequency.exponentialRampToValueAtTime(920, t + 0.16);
            gain.gain.setValueAtTime(0.0001, t);
            gain.gain.exponentialRampToValueAtTime(type === 'step' ? 0.006 : 0.045, t + 0.015);
            gain.gain.exponentialRampToValueAtTime(0.0001, t + (type === 'quest' ? 0.32 : 0.12));
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start(t);
            osc.stop(t + (type === 'quest' ? 0.33 : 0.13));
          } catch {}
        }

        function onEvent(kind, payload = {}) {
          const q = currentQuest();
          if (!q || done(q)) return false;
          const step = nextStep();
          if (!step || step.kind !== kind) return false;
          const scene = getScene();
          if (kind === 'npc') {
            if (payload.npc !== step.npc) return false;
            const msg = clean(payload.text);
            if (!/^[\x20-\x7e]+$/.test(msg) || msg.split(/\s+/).length < 3) return false;
            const input = normalize(msg);
            if (!step.words.some((w) => input.includes(normalize(w)))) return false;
            // Protect against a repeated keypress or duplicate NPC callback.
            const id = 'npc:' + step.npc + ':' + progress(q),
              at = Date.now();
            if (onEvent.lastId === id && at - (onEvent.lastAt || 0) < 900) return false;
            onEvent.lastId = id;
            onEvent.lastAt = at;
          } else {
            if (step.building !== payload.building) return false;
            if (step.floor !== undefined && step.floor !== (Number(payload.floor) || 0))
              return false;
            if (step.objects && !step.objects.includes(payload.object)) return false;
            if (step.area) {
              let a = payload.area || identifyArea(scene);
              if (step.area === 'minerals' && !isMinerals(a)) return false;
            }
          }
          state.progress[q.id] = progress(q) + 1;
          store();
          sound('click');
          if (progress(q) >= q.steps.length) reward(q);
          else {
            markActivity('Quest ' + q.title + ': ' + progress(q) + '/' + q.steps.length);
            toast('✅ Step completed ' + progress(q) + '/' + q.steps.length);
            renderQuest();
          }
          return true;
        }

        function target() {
          const q = currentQuest();

          const step = nextStep();
          if (!step || step.noMark) return null;
          const scene = getScene();
          if (!scene) return null;
          if (step.kind === 'enter' || step.kind === 'floor' || step.kind === 'inspect') {
            const b = scene.area.buildings.find((x) => x.name === step.building);
            if (!b) return destinationGate(scene, step.building);
            return { x: b.door * 32 + 16, y: (b.y + b.h) * 32 + 16, name: b.name };
          }
          if (step.kind === 'npc') {
            const n = scene.npcs.find((x) => x.def.id === step.npc);

              return { remote: n.def.name + ' (on their daily schedule)', area: 'near' };
            return { x: n.sprite.x, y: n.sprite.y, name: n.def.name };
          }
          return null;
