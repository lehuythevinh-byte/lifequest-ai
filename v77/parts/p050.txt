function safeTile(sc,tx,ty,max=18){

function route(sc,start,end,limit=3200){if(!goodScene(sc)||!start||!end)return [];
 const sx=Math.floor(start.x/T),sy=Math.floor(start.y/T),ex=Math.floor(end.x/T),ey=Math.floor(end.y/T);

 const queue=[origin],back=new Map([[origin,-1]]);let head=0,step=0;
 while(head<queue.length&&step++<limit){const k=queue[head++];if(k===goal){const nodes=[];let p=k;while(p!==origin&&nodes.length<140){nodes.push({x:((p%w)+.5)*T,y:(Math.floor(p/w)+.5)*T});p=back.get(p);if(p===undefined)return []}return nodes.reverse()}
  const x=k%w,y=Math.floor(k/w);
  for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){const nx=x+dx,ny=y+dy,kk=key(nx,ny);if(back.has(kk)||!openTile(sc,nx,ny))continue;back.set(kk,k);queue.push(kk)}

}
