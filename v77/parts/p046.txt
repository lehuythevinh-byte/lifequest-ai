            const cx = c.getContext('2d');
            cx?.clearRect(0, 0, c.width, c.height);

            weatherType === 'rain' ? (preferences.performance === 'high' ? 65 : 36) : 18;
          while (fxParticles.length < count)
            fxParticles.push({
              x: Math.random() * c.width,
              y: Math.random() * c.height,
              speed: 1 + Math.random() * 2.4,
            });
          if (fxParticles.length > count) fxParticles.length = count;
          ctx.strokeStyle =
            weatherType === 'rain' ? 'rgba(164,209,250,0.45)' : 'rgba(234,244,255,0.16)';
          ctx.lineWidth = weatherType === 'rain' ? 1.2 : 8;
          for (const p of fxParticles) {
            p.x += weatherType === 'rain' ? 2 : 0.3;
            p.y += weatherType === 'rain' ? 6 * p.speed : 0.2 * p.speed;
            if (p.y > c.height) {
              p.y = -15;
              p.x = Math.random() * c.width;
            }
            if (p.x > c.width) p.x = 0;
            ctx.beginPath();
            ctx.moveTo(p.x, p.y);
            ctx.lineTo(
              p.x + (weatherType === 'rain' ? 3 : 17),
              p.y + (weatherType === 'rain' ? 14 : 7),
            );
            ctx.stroke();
          }
        }

        function importBackup(file) {
          if (!file || file.size > 3e6) {
            toast('⚠ Backup file is too large or invalid.');
            return;
          }
          const reader = new FileReader();
          reader.onload = () => {
            try {
              const obj = JSON.parse(reader.result);
              // Accept the original V1–V7 Journal/J export format for safe cross-file migration.
              if (
                obj.app === 'english-town' &&
                obj.game &&
                typeof obj.game === 'object' &&
                ['day', 'money', 'rep', 'minutes'].every(
                  (k) => typeof obj.game[k] === 'number' && Number.isFinite(obj.game[k]),
                )
              ) {
                if (!confirm('Importing LIFEQUEST V7 data will overwrite your current progress. Continue?'))
                  return;
                localStorage.setItem(
                  'english-town-save-v1',
                  JSON.stringify({
                    game: obj.game,
                    mem:
                      obj.mem && typeof obj.mem === 'object' && !Array.isArray(obj.mem)
                        ? obj.mem
                        : {},
                  }),
                );
                location.reload();
                return;
              }
              if (
                obj.format !== 'LIFEQUEST_V8_BACKUP' ||
                !obj.data ||
                typeof obj.data !== 'object' ||
                Array.isArray(obj.data)
              )
                throw Error('Invalid format');
              const entries = Object.entries(obj.data);
              if (entries.length > 100) throw Error('Too many entries');
              for (const [k, v] of entries) {
                if (
                  !/^(english-town-save-v1|lifequest[_-][a-z0-9_-]{1,100})$/i.test(k) ||
                  /(?:key|secret|token|credential)/i.test(k) ||
                  typeof v !== 'string' ||
                  v.length > 2e6
                )
                  throw Error('Invalid data: ' + k);
                JSON.parse(v);
              }
              if (
                !confirm(
                  'Importing a backup will overwrite game progress on this device. Continue?',
                )
              )
                return;
              for (const [k, v] of entries) localStorage.setItem(k, v);
              location.reload();
            } catch (e) {
              toast('⚠ Could not import backup: ' + e.message);
            }
          };
          reader.readAsText(file, 'utf-8');
        }
