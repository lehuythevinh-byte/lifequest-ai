        function findMention(s, entries, aliases) {
          const hits = [];
          for (const x of entries || []) {
            let names = [
              x.id,
              x.name,
              String(x.name)
                .replace(/\([^)]*\)/g, '')
                .replace(/^(mr|ms|dr|officer|father)\.? /i, ''),
              ...(aliases[x.id] || []),
              ...(aliases[x.name] || []),
            ];
            for (const name of names) {
              const n = fold(name);
              if (n.length > 2 && has(s, n)) hits.push({ x, len: n.length });
            }
          }
          hits.sort((a, b) => b.len - a.len);
          return hits[0]?.x || null;
        }
        function detectAction(input, ctx, mode) {
          const s = fold(input),
            plain = s.replace(/[.!?]+$/, '');
          const isStop =
            /^(dung|dung lai|stop|stop moving|cancel|huy lenh|huy di chuyen|thoi dung|pause)( ngay| lai| di| nhe)?$/.test(
              plain,

          if (isStop) return { kind: 'stop' };
          if (mode === 'chat' || mode === 'learn') return null;
          const isGuide =
            /^(help|giup do|huong dan|cach choi|lenh nao|commands|ban lam duoc gi|nhung lenh nao)/.test(
              s,

          if (isGuide) return { kind: 'help' };

            /^(toi|minh|nhan vat|player|i) (dang )?o dau|^(vi tri|toa do|where am i|what time is it|may gio roi|gio trong game)/.test(

            /noi chuyen|tro chuyen|bat chuyen|chat with|talk to|speak to|giao tiep voi|noi voi|say hello to/.test(

            place = findMention(s, ctx.places, placeAliases);

          }
          if (place && motion && !asking) {

              ? 'left'

                ? 'right'
                : /\b(len|tren|up|north)\b/.test(s)
                  ? 'up'
                  : /\b(xuong|duoi|down|south)\b/.test(s)
