    const r = await client.from('saves').select('data,updated_at').eq('user_id', user.id).maybeSingle();

    return r.data || null;

  function sameData(a, b) {
    const ka = Object.keys(a).sort(),
      kb = Object.keys(b).sort();
    return ka.length === kb.length && ka.every((k, i) => k === kb[i] && a[k] === b[k]);
  }

  async function push(opts = {}) {
    if (!user || !client || busy) return false;
    busy = true;
    setStatus('sync');
    try {
      const data = snapshot();
      if (!data[SAVE_KEY]) {
        setStatus('ok');
        return true;
      }
      const body = { v: 1, data, savedAt: new Date().toISOString() };
      if (JSON.stringify(body).length > MAX_BYTES) throw new Error('Save data is too large');
      const meta = readMeta();
      if (!opts.force) {
        const row = await fetchRow();
        if (row && row.updated_at !== meta.lastCloudAt && !sameData(row.data.data || {}, data)) {
          busy = false;
          const pick = await chooseSave(data, row, 'Server progress was changed from another device.');
          if (pick === 'cloud') {
            applySnapshot(row.data.data || {});
            writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false });
            hardReload();
            return false;
          }
          return await push({ force: true });
        }
      }
      const r = await client.from('saves').upsert({ user_id: user.id, data: body }).select('updated_at').single();
      if (r.error) throw r.error;
      writeMeta({ userId: user.id, lastCloudAt: r.data.updated_at, dirty: false, guest: false });
      setStatus('ok');
      return true;
    } catch (e) {
      console.warn('cloud save failed', e);
      setStatus('bad', friendly(e));
      return false;
    } finally {
      busy = false;
    }
  }

  // Best-effort save while the tab is closing (fetch with keepalive survives page unload).
  function pushOnLeave() {
    if (!user || !token || !readMeta().dirty) return;
    const data = snapshot();
    if (!data[SAVE_KEY]) return;
    const body = JSON.stringify({ user_id: user.id, data: { v: 1, data, savedAt: new Date().toISOString() } });
    if (body.length > 60000) return; // keepalive requests are limited to 64 KB
    try {
      fetch(URL_ + '/rest/v1/saves?on_conflict=user_id', {
        method: 'POST',
        keepalive: true,
        headers: { apikey: ANON, Authorization: 'Bearer ' + token, 'Content-Type': 'application/json', Prefer: 'resolution=merge-duplicates,return=minimal' },
        body,
      });
      writeMeta({ dirty: false, lastCloudAt: '' });
    } catch (e) {
      /* ignore */
    }
  }

  function markDirty() {

    }
  }

  /** Decide which copy to keep after signing in. Returns 'applied' when this device's data was replaced. */
  async function reconcile() {
    const meta = readMeta();
    const row = await fetchRow();
    const local = snapshot();
    const hasLocal = !!local[SAVE_KEY];
    if (!row) {
      if (hasLocal) await push({ force: true });
      else writeMeta({ userId: user.id, lastCloudAt: '', dirty: false, guest: false });
      return 'kept';
    }
    const cloudData = row.data.data || {};
    if (hasLocal && sameData(cloudData, local)) {
      writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false, guest: false });
      return 'kept';
    }
    if (!hasLocal) {
      applySnapshot(cloudData);
      writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false, guest: false });
      return 'applied';
    }
    if (meta.userId === user.id && row.updated_at === meta.lastCloudAt) {
      if (meta.dirty) await push({ force: true });
      return 'kept';
    }
    if (meta.userId === user.id && !meta.dirty) {
      applySnapshot(cloudData);
      writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false });
      return 'applied';
    }
    const why = meta.userId === user.id ? 'Both places have unsynced changes.' : 'This device has game progress (guest or another account), and your account also has progress on the server.';
    const pick = await chooseSave(local, row, why);
    if (pick === 'cloud') {
      applySnapshot(cloudData);
      writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false, guest: false });
      return 'applied';
    }
    await push({ force: true });
    return 'kept';
  }

  async function checkRemote() {
    if (!user || busy || Date.now() - lastCheck < 20000) return;
    lastCheck = Date.now();
    try {
      const row = await fetchRow();

    }

  function installHooks() {
    Storage.prototype.setItem = function (k, v) {
      if (this === store && blockWrites && isSyncKey(k)) return;
      nativeSet.call(this, k, v);
      if (this === store && !suppress && isSyncKey(k)) markDirty();
    };
    Storage.prototype.removeItem = function (k) {
      if (this === store && blockWrites && isSyncKey(k)) return;
      nativeRemove.call(this, k);
      if (this === store && !suppress && isSyncKey(k)) markDirty();
    };

      if (document.visibilityState === 'hidden') pushOnLeave();
      else checkRemote();

    } catch (e) {
      console.warn('Đăng nhập không khả dụng:', e);
      boot();
      return;
    }
    installHooks();
    client.auth.onAuthStateChange((evt, session) => {
      token = session ? session.access_token : '';
      if (session) user = session.user;
      if (evt === 'SIGNED_OUT') user = null;
    });
    try {
      const s = await client.auth.getSession();
      if (s.data && s.data.session) {
        user = s.data.session.user;
        token = s.data.session.access_token;
      }
      if (!user) {

        if (r === 'guest') writeMeta({ guest: true });

      if (user) await reconcile();
    } catch (e) {
      console.warn('Không đồng bộ được lúc khởi động:', e);
      setStatus('bad', friendly(e));
    }
    boot();
  }

  function applyPreset(i) {
    const sc = getScene();
    if (!sc || !sc.store || !sc.store.game) return;
    const preset = PRESETS[i];
    if (!preset || isLocked(preset)) return;
    const g = sc.store.game;
    if (!g.rpg || typeof g.rpg !== 'object') g.rpg = {};
    const cur = g.rpg.look;
    const next = Object.assign(presetLook(preset), { name: ((cur && cur.name) || '').trim() || preset.name });
    g.rpg.look = JSON.parse(JSON.stringify(next));
    sc.store.save();
    try {
      sc.applyRankLook();
    } catch (e) {
      console.warn('applyRankLook failed', e);
    }
    window.dispatchEvent(new CustomEvent('lifequest:lookchanged', { detail: g.rpg.look }));
    drawQuick();

  // A save that already wears a locked character (e.g. Gojo picked before the lock existed) is switched to the first
  // unlocked preset, keeping the player's name.
  function enforceLocks() {
    const cur = currentLook();
    if (!cur) return;
    const locked = PRESETS.find((p) => isLocked(p) && cur.style === p.look.style && cur.hair === p.look.hair && !!cur.uniform === !!p.look.uniform);
    if (!locked) return;
    const i = PRESETS.findIndex((p) => !isLocked(p));
    if (i >= 0) {
      applyPreset(i);
      if (window.LQAchievements) window.LQAchievements.hint(locked.lock);
    }
  }

  // Open the creator by itself the first time a player enters the game.

    let tries = 0;

      tries++;
      const sc = getScene();
      if (sc && sc.player && sc.store && sc.store.game) {
        clearInterval(timer);
        mountQuick();
        enforceLocks();
        if (!currentLook()) open({ mandatory: true });
      } else if (tries > 240) clearInterval(timer);
    }, 500);
  }

  let toastEl = null,
    toastTimer = 0;

  function toast(kicker, title, text) {
    if (!toastEl) {
      const st = document.createElement('style');
      st.textContent = TOAST_CSS;
      document.head.appendChild(st);
      toastEl = document.createElement('div');
      toastEl.id = 'lq-ach-toast';
      toastEl.setAttribute('role', 'status');
      toastEl.addEventListener('click', () => {
        toastEl.classList.remove('on');
        if (window.LQ11_UI) window.LQ11_UI.open('achievements');
      });
      document.body.appendChild(toastEl);
    }
    toastEl.innerHTML = `<small>${esc(kicker)}</small><b>${esc(title)}</b><p>${esc(text)}</p>`;
    requestAnimationFrame(() => toastEl.classList.add('on'));
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => toastEl.classList.remove('on'), 7000);
  }

  function hint(id) {
    const a = byId(id);
    if (a) toast('🔒 Achievement required', a.title, 'See how to unlock it in Menu → Achievements.');
  }

  function complete(a) {
    const st = state(true);
    if (!st || st.done[a.id]) return;
    st.done[a.id] = Date.now();
    persist();
    window.dispatchEvent(new CustomEvent('lifequest:achievements', { detail: { id: a.id } }));
    const reward = a.reward && a.reward.character ? `Unlocked the character ${a.reward.character}! Open Menu → Achievements to wear it.` : 'You completed the achievement.';
    toast('🏆 Achievement completed', a.title, reward);
    if (window.LQ11_UI && document.getElementById('l11-shell') && !document.getElementById('l11-shell').hidden)
      window.LQ11_UI.page('achievements');
  }

  // ---- progress tracking --------------------------------------------------------------------------------------
  function mark(a, sid) {
    const st = state(true);
    const key = a.id + ':' + sid;
    if (!st || st.steps[key]) return false;
    st.steps[key] = Date.now();
    persist();
    return true;
  }

  let greeting = false;
  function tick() {
    const sc = getScene();
    if (!sc || !sc.area || !sc.store || !sc.store.game) return;
    const area = sc.area.id;
    const a = byId('mineral-trip');
    if (!a || isUnlocked(a.id) || greeting) return;
    // Progress is recorded silently; the how-to lives only on the Menu → Achievements page.
    if (area === MINE) mark(a, 'visit');
