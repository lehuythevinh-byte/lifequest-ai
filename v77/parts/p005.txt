  // Everything is drawn with code (no image files). The world is shown through a pale filter (applyFilter), so the
  // drawing relies on light/shadow more than on colour: every object gets a soft shadow, a lit top and a dark base.
  function rnd(seed) {
    let s = seed | 0;
    return () => {
      s = (s + 0x6d2b79f5) | 0;
      let t = Math.imul(s ^ (s >>> 15), 1 | s);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }
  // Multiply a colour (k < 1 darker, k > 1 lighter).
  const shade = (c, k) => {
    const f = (v) => Math.max(0, Math.min(255, Math.round(v * k)));
    return (f((c >> 16) & 255) << 16) | (f((c >> 8) & 255) << 8) | f(c & 255);
  };

  // Stone wall: irregular blocks, a lit top, crenels on the outer rows and a shadow on the ground below.
  function wallTile(g, o) {
    const x = o.x * T;
    const y = o.y * T;
    const r = rnd(o.x * 131 + o.y * 977);
    const STONES = [0x8e897f, 0x98938a, 0x86817a, 0xa19c92];
    g.fillStyle(0x1a160f, 0.18).fillRect(x, y + T, T, 6);
    g.fillStyle(0x5d5850, 1).fillRect(x, y, T, T);
    const rows = [[1, 9], [11, 9], [21, 10]];
    rows.forEach(([ry, rh], i) => {
      let cx = x + (i % 2 ? -6 : 0);
      while (cx < x + T) {
        const sw = 9 + Math.floor(r() * 9);
        const sx = Math.max(x, cx + 1);
        const ex = Math.min(x + T, cx + sw - 1);
        if (ex - sx > 2) {
          const c = STONES[Math.floor(r() * STONES.length)];
          g.fillStyle(c, 1).fillRoundedRect(sx, y + ry, ex - sx, rh - 1, 2);
          g.fillStyle(shade(c, 1.12), 1).fillRect(sx + 1, y + ry, ex - sx - 2, 2);
          g.fillStyle(shade(c, 0.82), 1).fillRect(sx + 1, y + ry + rh - 3, ex - sx - 2, 2);
        }
        cx += sw;
      }
    });
    // walkway cap and crenels on top
    g.fillStyle(0xb2ada2, 1).fillRect(x, y - 3, T, 4);
    if ((o.x + o.y) % 2 === 0) {
      g.fillStyle(0x9a958b, 1).fillRect(x + 4, y - 10, T - 8, 8);
      g.fillStyle(0xb8b3a8, 1).fillRect(x + 4, y - 10, T - 8, 2);
    }
    // a little moss
    if (r() < 0.35) g.fillStyle(0x6f7d4c, 0.8).fillEllipse(x + r() * T, y + T - 3, 8, 4);
  }

  function building(g, b) {
    const x = b.x * T;
    const y = b.y * T;
    const w = b.w * T;
    const h = b.h * T;
    shadow(g, x + w / 2 + 8, y + h + 2, w + 20, 18, 0.26);
    if (b.kind === 'hall') return hall(g, b, x, y, w, h);
    if (b.kind === 'barn') return barn(g, b, x, y, w, h);
    if (b.kind === 'wood') return woodHouse(g, b, x, y, w, h);
    return thatchHouse(g, b, x, y, w, h);
  }

  // Straw-roofed cottage: thick layered thatch, daub walls with a timber frame, stone footing.
  function thatchHouse(g, b, x, y, w, h) {
    const wallTop = y + h * 0.42;
    const by = y + h;
    // walls
    g.fillStyle(0xddd1b4, 1).fillRect(x + 2, wallTop, w - 4, by - wallTop);
    g.fillStyle(0xcfc2a2, 1).fillRect(x + 2, by - 16, w - 4, 10);
    // timber frame: posts, beam, braces
    g.fillStyle(0x5a4429, 1);
    [x + 2, x + w / 2 - 2, x + w - 6].forEach((px) => g.fillRect(px, wallTop, 4, by - wallTop));
    g.fillRect(x + 2, wallTop + 4, w - 4, 3);
    g.lineStyle(3, 0x5a4429, 1).lineBetween(x + 6, by - 8, x + w / 2 - 2, wallTop + 8).lineBetween(x + w - 6, by - 8, x + w / 2 + 2, wallTop + 8);
    foundation(g, x + 2, by, w - 4);
    // window on the side away from the door
    const dx = b.door * T + 16;
    const winX = dx < x + w / 2 ? x + w - 34 : x + 18;
    windowBox(g, winX, wallTop + 14, 14, 12, true);
    g.fillStyle(0x6f7d4c, 1).fillRect(winX - 2, wallTop + 29, 18, 4);
    g.fillStyle(0xd9cfa0, 1).fillCircle(winX + 2, wallTop + 29, 2).fillCircle(winX + 9, wallTop + 28, 2);
    door(g, dx, by - 2, 18, 26);
    // thatch roof: rounded mass, layered bands with ragged edges, bound ridge
    const ry = y - 10;
    const rh = h * 0.56 + 10;
    g.fillStyle(0x1a160f, 0.25).fillRoundedRect(x - 8, wallTop - 2, w + 16, 10, 5);
    g.fillStyle(0x9a8044, 1).fillRoundedRect(x - 10, ry, w + 20, rh, 18);
    const r = rnd(x * 3 + y);
    const bands = 6;
    for (let i = bands - 1; i >= 0; i--) {
      const yy = ry + 6 + (i * (rh - 10)) / bands;
      const c = i % 2 ? 0xc8ae6c : 0xb79c58;
      g.fillStyle(c, 1).fillRoundedRect(x - 9 + i * 0.5, yy, w + 18 - i, (rh - 10) / bands + 4, 8);
      for (let xx = x - 8; xx < x + w + 8; xx += 5) {
        const len = 4 + r() * 5;
        g.fillStyle(shade(c, 0.8), 1).fillRect(xx, yy + (rh - 10) / bands + 1, 2, len - 2);
        g.lineStyle(1, shade(c, 1.12), 0.9).lineBetween(xx + 1, yy + 2, xx + 1, yy + 6);
      }
    }
    g.fillStyle(0x8a7038, 1).fillRoundedRect(x - 4, ry - 4, w + 8, 10, 5);
    for (let xx = x; xx < x + w; xx += 12) g.fillStyle(0x6b5a30, 1).fillRect(xx, ry - 4, 2, 10);
    g.fillStyle(0xd8c07e, 0.6).fillRoundedRect(x - 2, ry - 3, w + 4, 3, 2);
    chimney(g, x + w - 22, ry - 12);
  }

  // Plank house: vertical boards, a shingle roof with a lit ridge, a little porch roof over the door.
  function woodHouse(g, b, x, y, w, h) {
    const wallTop = y + h * 0.44;
    const by = y + h;
    const PL = [0x7f6142, 0x8a6a48, 0x76593c];
    for (let xx = x + 2; xx < x + w - 2; xx += 7) {
      g.fillStyle(PL[Math.floor(((xx - x) / 7) % 3)], 1).fillRect(xx, wallTop, 7, by - wallTop);
      g.fillStyle(0x4a3826, 1).fillRect(xx + 6, wallTop, 1, by - wallTop);
      g.fillStyle(0x9a7a52, 0.6).fillRect(xx + 1, wallTop, 1, by - wallTop);
    }
    g.fillStyle(0x5a4429, 1).fillRect(x + 2, wallTop, 4, by - wallTop).fillRect(x + w - 6, wallTop, 4, by - wallTop);
    g.fillStyle(0x5a4429, 1).fillRect(x + 2, wallTop + 2, w - 4, 3);
    foundation(g, x + 2, by, w - 4);
    const dx = b.door * T + 16;
    const winX = dx < x + w / 2 ? x + w - 34 : x + 18;
    windowBox(g, winX, wallTop + 13, 14, 12, true);
    door(g, dx, by - 2, 18, 26);
    // porch roof over the door
    poly(g, 0x5d4a3a, 1, [[dx - 18, by - 30], [dx + 18, by - 30], [dx + 14, by - 38], [dx - 14, by - 38]]);
    g.fillStyle(0x4a3826, 1).fillRect(dx - 17, by - 30, 3, 28).fillRect(dx + 14, by - 30, 3, 28);
    // shingle roof
    const ry = y - 8;
    const rh = h * 0.56 + 8;
    g.fillStyle(0x1a160f, 0.25).fillRect(x - 6, wallTop - 2, w + 12, 8);
    g.fillStyle(shade(b.color, 0.8), 1).fillRect(x - 8, ry, w + 16, rh);
    const rows = 7;
    for (let i = 0; i < rows; i++) {
      const yy = ry + 4 + (i * (rh - 6)) / rows;
      const off = i % 2 ? 6 : 0;
      for (let xx = x - 8 - off; xx < x + w + 8; xx += 12) {
        const sx = Math.max(x - 8, xx);
        const ex = Math.min(x + w + 8, xx + 11);
        if (ex <= sx) continue;
        const c = shade(b.color, 0.92 + ((xx * 7 + i * 3) % 5) * 0.05);
        g.fillStyle(c, 1).fillRect(sx, yy, ex - sx, (rh - 6) / rows + 2);
        g.fillStyle(shade(c, 0.7), 1).fillRect(sx, yy + (rh - 6) / rows, ex - sx, 2);
      }
    }
    g.fillStyle(shade(b.color, 0.6), 1).fillRect(x - 8, ry, 3, rh).fillRect(x + w + 5, ry, 3, rh);
    g.fillStyle(shade(b.color, 1.3), 1).fillRect(x - 6, ry - 4, w + 12, 6);
    g.fillStyle(shade(b.color, 0.75), 1).fillRect(x - 6, ry + 1, w + 12, 2);
    chimney(g, x + 20, ry - 10);
  }

  // Big stone hall with an arched door, tall windows, buttresses, a slate roof and a square tower with a banner.

    g.fillStyle(0x4c535b, 1).fillRect(x - 8, ry, w + 16, rh);

      const yy = ry + 2 + i * (rh / 6);

  function buildTall(sc) {
    const items = tallList(sc.area).map((t) => {
      const w = Math.ceil(t.x1 - t.x0);
      const h = Math.ceil(t.y1 - t.y0);
      if (!sc.textures.exists(t.key)) {
        const g = sc.add.graphics();
        g.translateCanvas(-t.x0, -t.y0);
        t.draw(g);
        g.generateTexture(t.key, w, h);
        g.destroy();
      }
      return { img: sc.add.image(t.x0, t.y0, t.key).setOrigin(0).setDepth(PLAYER_DEPTH - 0.6), base: t.base };
    });
    const st = { area: sc.area, items };
    st.update = () => {
      if (sc.area !== st.area) return clearTall(sc);
      if (!sc.player) return;
      const hide = !!sc.inside;
      const py = sc.player.y;
      for (const it of items) {
        it.img.setVisible(!hide);
        it.img.setDepth(py < it.base ? PLAYER_DEPTH + 0.6 : PLAYER_DEPTH - 0.6);
      }
    };
    sc.events.on('update', st.update);
    sc.__lqPastTall = st;
    st.update();
  }
  function clearTall(sc) {
    const st = sc && sc.__lqPastTall;
    if (!st) return;
    sc.events.off('update', st.update);
    st.items.forEach((it) => it.img.destroy());
    sc.__lqPastTall = null;
  }
  function syncTall(sc) {
    const st = sc.__lqPastTall;
    if (here(sc) && !sc.switching) {
      if (st && st.area !== sc.area) clearTall(sc);
      if (!sc.__lqPastTall) {
        try {
          buildTall(sc);
        } catch (e) {
          console.warn('past tall sprites failed', e);
          sc.__lqPastTall = { area: sc.area, items: [], update() {} };
        }
      }
    } else if (st) clearTall(sc);
  }

  // Each group stays near its home (tile cx, cy, radius r). Drawn every frame into one Graphics just below the player.
  const FLOCKS = [

  const HEN_SPEED = 26;
  const RUN_SPEED = 95;
  const SCARE = 46;
  function makeFlock(a) {
    const r = rnd(2024);
    const birds = [];
    const free = (x, y) => !a.isTileBlocked(Math.floor(x / T), Math.floor(y / T));
    const spot = (f) => {
      for (let i = 0; i < 40; i++) {
        const ang = r() * Math.PI * 2;
        const d = Math.sqrt(r()) * f.r * T;
        const x = (f.cx + 0.5) * T + Math.cos(ang) * d;
        const y = (f.cy + 0.5) * T + Math.sin(ang) * d;
        if (free(x, y)) return { x, y };
      }
      return { x: (f.cx + 0.5) * T, y: (f.cy + 0.5) * T };
    };
    FLOCKS.forEach((f) => {
      const add = (kind) => {
        const p = spot(f);
        birds.push({ kind, f, x: p.x, y: p.y, tx: p.x, ty: p.y, state: 'idle', t: r() * 2, face: r() < 0.5 ? -1 : 1, phase: r() * 10, flap: 0, mom: null });
      };
      for (let i = 0; i < f.hens; i++) add('hen');
      if (f.rooster) add('rooster');
      const moms = birds.filter((b) => b.f === f && b.kind === 'hen');
      for (let i = 0; i < (f.chicks || 0); i++) {
        add('chick');
        birds[birds.length - 1].mom = moms[i % moms.length] || null;
      }
    });
    return { birds, free, spot, r };
  }
  function henTick(fl, sc, dt) {
    const who = scarers(sc);
    for (const b of fl.birds) {
      b.t -= dt;
      b.phase += dt * (b.state === 'walk' || b.state === 'run' ? 14 : 3);
      b.flap = Math.max(0, b.flap - dt);
      // The nearest one who scares it: the player, or a running child.
      let px = who[0].x;
      let py = who[0].y;
      for (const s of who) {
        if (Math.hypot(b.x - s.x, b.y - s.y) < Math.hypot(b.x - px, b.y - py)) {
          px = s.x;
          py = s.y;
        }
      }
      const dp = Math.hypot(b.x - px, b.y - py);
      // Scared: run straight away from the player, flapping.
      if (dp < SCARE && b.state !== 'run') {
        const k = 70 / Math.max(1, dp);
        b.tx = b.x + (b.x - px) * k;
        b.ty = b.y + (b.y - py) * k;
        b.state = 'run';
        b.flap = 0.6;
        b.t = 1.2;
      }
      if (b.state === 'walk' || b.state === 'run') {
        const sp = (b.state === 'run' ? RUN_SPEED : HEN_SPEED) * (b.kind === 'chick' ? 1.15 : 1) * dt;
        const dx = b.tx - b.x;
        const dy = b.ty - b.y;
        const d = Math.hypot(dx, dy);
        if (Math.abs(dx) > 0.5) b.face = dx < 0 ? -1 : 1;
        if (d <= sp || b.t <= 0) {
          b.state = fl.r() < 0.6 ? 'peck' : 'idle';
          b.t = 0.8 + fl.r() * 2.2;
        } else {
          const nx = b.x + (dx / d) * sp;
          const ny = b.y + (dy / d) * sp;
          if (fl.free(nx, ny)) {
            b.x = nx;
            b.y = ny;
          } else {
            b.state = 'idle';
            b.t = 0.3;
          }
        }
      } else if (b.t <= 0) {
        // Pick the next place: chicks follow their mother, the others wander around home.
        let p;
        if (b.kind === 'chick' && b.mom) p = { x: b.mom.x + (fl.r() - 0.5) * 30, y: b.mom.y + 6 + (fl.r() - 0.5) * 16 };
        else p = fl.spot(b.f);
        b.tx = p.x;
        b.ty = p.y;
        b.state = 'walk';
        b.t = 6;
        if (b.kind === 'rooster' && fl.r() < 0.15) b.flap = 0.8;
      }
    }
  }
  function drawBird(g, b) {
    const f = b.face;
    const x = b.x;
    const walking = b.state === 'walk' || b.state === 'run';
    const bob = walking ? Math.abs(Math.sin(b.phase)) * 1.6 : 0;
    const y = b.y - bob;
    if (b.kind === 'chick') {
      g.fillStyle(0x1a160f, 0.2).fillEllipse(b.x, b.y + 4, 8, 3);
      g.fillStyle(0xf1d36a, 1).fillCircle(x, y, 3.6);
      g.fillStyle(0xf6df86, 1).fillCircle(x + f * 2.5, y - 3, 2.6);
      g.fillStyle(0xd9822e, 1).fillTriangle(x + f * 4.5, y - 3.5, x + f * 6.5, y - 3, x + f * 4.5, y - 2.3);
      g.fillStyle(0x2a241d, 1).fillRect(x + f * 3 - 0.5, y - 4, 1.2, 1.2);
      return;
    }
    const big = b.kind === 'rooster' ? 1.18 : 1;
    const body = b.kind === 'rooster' ? 0xc0743c : 0xefe9dc;
    const wing = b.kind === 'rooster' ? 0x9a5a2e : 0xd9d2c2;
    const pecking = b.state === 'peck' && Math.sin(b.phase * 2.2) > 0.2;
    g.fillStyle(0x1a160f, 0.2).fillEllipse(b.x, b.y + 6 * big, 14 * big, 4);
    // legs (step when walking)
    const step = walking ? Math.sin(b.phase) * 2 : 0;
    g.lineStyle(1.3, 0xd9a13a, 1).lineBetween(x - 1.5, y + 2, x - 1.5 + step, y + 6 * big).lineBetween(x + 1.5, y + 2, x + 1.5 - step, y + 6 * big);
    // tail
    if (b.kind === 'rooster') {
      g.fillStyle(0x2f4a3a, 1).fillTriangle(x - f * 4, y - 1, x - f * 11, y - 11, x - f * 9, y + 1);
      g.fillStyle(0x23382c, 1).fillTriangle(x - f * 5, y, x - f * 12, y - 6, x - f * 8, y + 3);
    } else g.fillStyle(0xe2dccd, 1).fillTriangle(x - f * 4, y - 1, x - f * 9, y - 7, x - f * 8, y + 1);
    // body + wing (wing lifts when flapping)
    g.fillStyle(body, 1).fillEllipse(x, y, 13 * big, 9 * big);
    if (b.flap > 0) {
      const up = Math.sin(b.flap * 40) * 4;
      g.fillStyle(wing, 1).fillTriangle(x - f * 2, y - 2, x - f * 8, y - 7 - up, x + f * 3, y - 3);
    } else g.fillStyle(wing, 1).fillEllipse(x - f * 1, y + 0.5, 7 * big, 5 * big);
    // head: down to the ground when pecking
    const hx = x + f * (pecking ? 7 : 5) * big;
    const hy = y - (pecking ? -1 : 5) * big;
    g.fillStyle(body === 0xefe9dc ? 0xf7f3ea : 0xd08446, 1).fillCircle(hx, hy, 3.4 * big);
    g.fillStyle(0xc8432f, 1).fillRect(hx - 1.5, hy - 4.6 * big, 3 * big, 2 * big);
    if (b.kind === 'rooster') g.fillStyle(0xc8432f, 1).fillCircle(hx + f * 1, hy - 4.5 * big, 1.6).fillCircle(hx - f * 1, hy - 5 * big, 1.6).fillCircle(hx + f * 2.5, hy + 2.5, 1.3);
    g.fillStyle(0xd9a13a, 1).fillTriangle(hx + f * 3, hy - 0.8, hx + f * 6, hy, hx + f * 3, hy + 1);
    g.fillStyle(0x2a241d, 1).fillRect(hx + f * 1.2 - 0.6, hy - 1.4, 1.3, 1.3);
  }
