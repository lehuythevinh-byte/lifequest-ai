    <script>
/*
 * Shrine cutscene: plays when the 20th finger is picked up (see fingers.js).
 *
 * 1. Four huge red parallelogram eyes fill a white, speckled screen.
 * 2. The camera pulls back: the eyes shrink up into a blood-red cloudy sky while the white clears, revealing the
 *    domain - a shrine rising out of cyan, rippling water with its reflection.
 * 3. Sukuna is a tiny figure on the shrine roof, dead centre between the eyes, so the domain feels vast around him.
 * 4. The picture holds for a moment, then fades out.
 *
 * It is drawn at 480 x 270 "pixels" and scaled up without smoothing for the pixel-art look.
 * window.LQShrine.play({ freeze }) returns a promise that resolves when the cutscene is over. `freeze` (ms) pins the
 * picture at that moment and never fades; it is only used to take screenshots while tuning the artwork.
 */
(function () {
  'use strict';

  const VW = 480;
  const VH = 270;
  const T_EYES = 600; // eyes fill the screen
  const T_PULL = 900; // camera pulls back, white clears
  const T_HOLD = 1100; // the domain, held
  const T_FADE = 1500; // fade out
  const RED = '#ed1c24';
  // Lead-in before the flash: the screen goes dark and a line is typed out letter by letter.
  const LEAD_TEXT = 'King of Curses has arrived';
  const LEAD_FONT = 'Cinzel, "Trajan Pro", "Times New Roman", Georgia, serif';
  const LEAD_FADE = 400; // fade to black
  const LEAD_CHAR = 55; // ms per letter
  const LEAD_HOLD = 800; // pause on the finished line before the cut
  const T_LEAD = LEAD_FADE + LEAD_TEXT.length * LEAD_CHAR + LEAD_HOLD;
  const HORIZON = 150;
  const CX = VW / 2;

  // The four eyes at full-screen size (upper pair large, lower pair smaller); right side is the mirror image.
  const EYES_BIG = [
    [[34, 52], [212, 96], [212, 150], [34, 106]],
    [[62, 176], [200, 176], [186, 236], [48, 236]],
  ];
  const EYES = [...EYES_BIG, ...EYES_BIG.map((e) => e.map(([x, y]) => [VW - x, y]))];
  const BIG_CENTER = [CX, 135];
  const SKY_CENTER = [CX, 50];
  const SKY_SCALE = 0.38;

  const ease = (x) => 1 - Math.pow(1 - Math.min(1, Math.max(0, x)), 3);
  const gauss = () => (Math.random() + Math.random() + Math.random() - 1.5) / 0.75;

  function seeded(seed) {
    let s = seed | 0;
    return () => {
      s = (s + 0x6d2b79f5) | 0;
      let t = Math.imul(s ^ (s >>> 15), 1 | s);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  function canvas(w, h) {
    const c = document.createElement('canvas');
    c.width = w;
    c.height = h;
    return c;
  }

  // ---- static backdrops, painted once ---------------------------------------------------------------------------
  function paintSky() {
    const c = canvas(VW, HORIZON);
    const g = c.getContext('2d');
    const rnd = seeded(7);
    const bands = ['#1a0610', '#220813', '#2b0a16', '#340c1a', '#3d0f1e', '#481222', '#531526'];
    const bh = HORIZON / bands.length;
    bands.forEach((col, i) => {
      g.fillStyle = col;
      g.fillRect(0, Math.floor(i * bh), VW, Math.ceil(bh) + 1);
    });
    const shades = ['#2a0b15', '#3a0f1c', '#56142a', '#6c1a2e', '#7d2234'];
    for (let i = 0; i < 130; i++) {
      const x = rnd() * VW;
      const y = rnd() * HORIZON;
      const rx = 18 + rnd() * 52;
      const ry = 5 + rnd() * 15;
      g.fillStyle = shades[Math.floor(rnd() * shades.length)];
      for (const ox of [-VW, 0, VW]) {
        g.beginPath();
        g.ellipse(x + ox, y, rx, ry, 0, 0, Math.PI * 2);
        g.fill();
      }
    }
    // checkerboard dithering over the clouds
    g.fillStyle = 'rgba(0,0,0,0.22)';
    for (let y = 0; y < HORIZON; y += 2) for (let x = (y / 2) % 2 ? 1 : 0; x < VW; x += 4) g.fillRect(x, y, 1, 1);
    return c;
  }

  function paintWater() {
    const h = VH - HORIZON;
    const c = canvas(VW, h);
    const g = c.getContext('2d');
    const rnd = seeded(21);
    const bands = ['#0f5a6a', '#0d4e5d', '#0b4351', '#09394a', '#073040', '#052837', '#04212e'];
    const bh = h / bands.length;
    bands.forEach((col, i) => {
      g.fillStyle = col;
      g.fillRect(0, Math.floor(i * bh), VW, Math.ceil(bh) + 1);
    });
    for (let i = 0; i < 520; i++) {
      const y = rnd() * h;
      const w = 4 + rnd() * 26 * (0.4 + y / h);
      g.fillStyle = rnd() < 0.5 ? 'rgba(53,198,209,0.28)' : 'rgba(0,0,0,0.28)';
      g.fillRect(rnd() * VW, y, w, 1);
    }
    return c;
  }

  // ---- the shrine and Sukuna (pixel sprites; feet of the shrine on y = 156) --------------------------------------
  const BASE = 156;
  function shrine(g) {
    const r = (x, y, w, h, col) => {
      g.fillStyle = col;
      g.fillRect(x, y, w, h);
    };
    r(213, 120, 8, BASE - 120, '#c9341a'); // pillars
    r(259, 120, 8, BASE - 120, '#c9341a');
    r(215, 120, 2, BASE - 120, '#e2501f');
    r(261, 120, 2, BASE - 120, '#e2501f');
    r(221, 124, 38, BASE - 124, '#15090e'); // dark body
    r(208, 126, 64, 3, '#8e1f12'); // lower beam
    // roof: dark gable with a red trim on top
    g.fillStyle = '#1d0e16';
    g.beginPath();
    g.moveTo(204, 114);
    g.lineTo(224, 102);
    g.lineTo(256, 102);
    g.lineTo(276, 114);
    g.closePath();
    g.fill();
    r(224, 101, 32, 1, '#7a1a1e');
    r(200, 114, 80, 5, '#e2501f'); // upper beam, ends swept up
    r(200, 114, 80, 1, '#ff8a3a');
    r(200, 110, 6, 5, '#e2501f');
    r(274, 110, 6, 5, '#e2501f');
    // the dark body is a gaping maw with teeth (the shrine itself has no eyes)
    for (let x = 226; x < 254; x += 2) r(x, 146 + ((x / 2) % 2), 1, 2, '#d9d2bd');
  }

  function sukuna(g) {
    const r = (x, y, w, h, col) => {
      g.fillStyle = col;
      g.fillRect(x, y, w, h);
    };
    // soft red glow: a few faint stacked ellipses instead of one hard disc
    g.fillStyle = 'rgba(255,40,40,0.045)';
    for (let i = 0; i < 6; i++) {
      g.beginPath();
      g.ellipse(240, 95, 7 + i * 3, 6 + i * 3, 0, 0, Math.PI * 2);
      g.fill();
    }
    r(237, 88, 6, 3, '#ea8fa0'); // hair
    r(236, 87, 1, 2, '#ea8fa0');
    r(243, 87, 1, 2, '#ea8fa0');
    r(239, 86, 2, 1, '#ea8fa0');
    r(238, 91, 4, 3, '#f6d2b0'); // face
    r(238, 92, 1, 1, '#c02a2a');
    r(241, 92, 1, 1, '#c02a2a');
    r(236, 94, 8, 8, '#e9e4d6'); // kimono
    r(236, 94, 1, 8, '#233a63');
    r(243, 94, 1, 8, '#233a63');
    r(237, 97, 6, 1, '#cfc9b8'); // folded arms
  }

  // ---- animated parts -------------------------------------------------------------------------------------------
  function ripples(g, t) {
    const cy = BASE + 4;
    for (let i = 0; i < 7; i++) {
      const r = ((t * 0.028 + i * 17) % 119) + 4;
      const a = 1 - r / 125;
      g.fillStyle = `rgba(150,235,240,${(0.55 * a).toFixed(3)})`;
      const steps = Math.floor(r * 4);
      for (let s = 0; s < steps; s++) {
        const ang = (s / steps) * Math.PI * 2;
        g.fillRect(Math.round(CX + Math.cos(ang) * r * 1.6), Math.round(cy + Math.sin(ang) * r * 0.34), 1, 1);
      }
    }
  }

  function sparkles(g, t) {
    const rnd = seeded(Math.floor(t / 140) + 3);
    for (let i = 0; i < 90; i++) {
      g.fillStyle = rnd() < 0.5 ? 'rgba(120,230,235,0.8)' : 'rgba(255,255,255,0.6)';
      g.fillRect(Math.floor(rnd() * VW), HORIZON + 4 + Math.floor(rnd() * (VH - HORIZON - 4)), 1, 1);
    }
  }

  function ash(g, t) {
    const rnd = seeded(99);
    for (let i = 0; i < 70; i++) {
      const x0 = rnd() * VW;
      const y0 = rnd() * HORIZON;
      const sp = 0.012 + rnd() * 0.02;
      const y = (((y0 - t * sp) % HORIZON) + HORIZON) % HORIZON;
      g.fillStyle = rnd() < 0.5 ? 'rgba(255,90,60,0.75)' : 'rgba(255,176,138,0.6)';
      g.fillRect(Math.floor(x0 + Math.sin(t / 700 + i) * 3), Math.floor(y), 1, 1);
    }
  }

  function scene(g, t, sky, water) {
    const off = (t * 0.006) % VW;
    g.drawImage(sky, -off, 0);
    g.drawImage(sky, VW - off, 0);
    g.fillStyle = 'rgba(160,30,44,0.7)';
    g.fillRect(0, HORIZON - 1, VW, 1);
    g.drawImage(water, 0, HORIZON);
    // reflection of the shrine
    g.save();
    g.globalAlpha = 0.32;
    g.translate(0, BASE * 2);
    g.scale(1, -1);
    shrine(g);
    g.restore();
    ripples(g, t);
    sparkles(g, t);
    shrine(g);
    sukuna(g);
    ash(g, t);
  }

  function eyes(g, e, t) {
    const lw = 22 + (7 - 22) * e;
    const flick = e > 0.98 ? 0.93 + 0.07 * Math.sin(t / 90) : 1;
    g.save();
    g.globalAlpha = flick;
    g.lineJoin = 'round';
    for (const eye of EYES) {
      const pts = eye.map(([x, y]) => {
        const sx = SKY_CENTER[0] + (x - BIG_CENTER[0]) * SKY_SCALE;
        const sy = SKY_CENTER[1] + (y - BIG_CENTER[1]) * SKY_SCALE;
        return [x + (sx - x) * e, y + (sy - y) * e];
      });
      g.beginPath();
      g.moveTo(pts[0][0], pts[0][1]);
      for (let i = 1; i < pts.length; i++) g.lineTo(pts[i][0], pts[i][1]);
      g.closePath();
      g.fillStyle = RED;
      g.strokeStyle = RED;
      g.lineWidth = lw;
      g.fill();
      g.stroke();
      g.strokeStyle = '#fff';
      g.lineWidth = 1;
      g.stroke();
    }
    g.restore();
  }

  function speckle(g, alpha) {
    if (alpha <= 0.02) return;
    g.globalAlpha = alpha;
    const centers = [[110, 100], [370, 100], [120, 205], [360, 205]];
    for (let i = 0; i < 9000; i++) {
      let x = Math.random() * VW;
      let y = Math.random() * VH;
      if (Math.random() < 0.7) {
        const c = centers[Math.floor(Math.random() * 4)];
        x = c[0] + gauss() * 70;
        y = c[1] + gauss() * 55;
      }
      g.fillStyle = Math.random() < 0.45 ? RED : '#000';
      g.fillRect(Math.floor(x), Math.floor(y), 1, 1);
    }
    g.globalAlpha = 1;
  }

  function playSound() {
    try {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) return;
      const ac = new AC();
      const t = ac.currentTime;
      const o = ac.createOscillator();
      const g = ac.createGain();
      o.type = 'sine';
      o.frequency.setValueAtTime(110, t);
      o.frequency.exponentialRampToValueAtTime(38, t + 0.9);
      g.gain.setValueAtTime(0.0001, t);
      g.gain.exponentialRampToValueAtTime(0.5, t + 0.03);
      g.gain.exponentialRampToValueAtTime(0.0001, t + 1.4);
      o.connect(g).connect(ac.destination);
      o.start(t);
      o.stop(t + 1.5);
      setTimeout(() => ac.close().catch(() => {}), 2000);
    } catch (e) {
      /* sound is optional */
    }
  }

  let playing = false;

  // Start fetching the lead-in font as soon as the script loads, so it is ready by the time the 20th finger is found.
  try {
    if (document.fonts && document.fonts.load) document.fonts.load('800 40px Cinzel').catch(() => {});
  } catch (e) {
    /* the serif fallback is used instead */
  }

  function play(opts) {
    const freeze = opts && typeof opts.freeze === 'number' ? opts.freeze : null;
    const leadAt = opts && typeof opts.leadAt === 'number' ? opts.leadAt : null; // screenshots of the lead-in only
    if (playing) return Promise.resolve();
    playing = true;
    return new Promise((resolve) => {
      const cv = document.createElement('canvas');
      cv.setAttribute('aria-hidden', 'true');
      cv.style.cssText = 'position:fixed;inset:0;width:100vw;height:100vh;z-index:9800;pointer-events:none;background:#1a0610';
      document.body.appendChild(cv);
      const W = (cv.width = Math.max(1, window.innerWidth));
      const H = (cv.height = Math.max(1, window.innerHeight));
      const ctx = cv.getContext('2d');
      const lo = canvas(VW, VH);
      const g = lo.getContext('2d');
      const sky = paintSky();
      const water = paintWater();
      // wide screens are filled edge to edge; tall ones keep the whole picture and show dark bars
      const scale = W / H >= 1.4 ? Math.max(W / VW, H / VH) : Math.min(W / VW, H / VH);
      const reduce = window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;
      // The dark lead-in: the line is typed in the middle of the screen (split over two lines on narrow screens).
      const drawLead = (el) => {
        cv.style.opacity = String(Math.min(1, el / LEAD_FADE));
        ctx.setTransform(1, 0, 0, 1, 0, 0);
        ctx.fillStyle = '#050207';
        ctx.fillRect(0, 0, W, H);
        const vg = ctx.createRadialGradient(W / 2, H / 2, Math.min(W, H) * 0.1, W / 2, H / 2, Math.max(W, H) * 0.7);
        vg.addColorStop(0, 'rgba(120,12,24,0.28)');
        vg.addColorStop(1, 'rgba(0,0,0,0)');
        ctx.fillStyle = vg;
        ctx.fillRect(0, 0, W, H);
        const shown = Math.max(0, Math.min(LEAD_TEXT.length, Math.floor((el - LEAD_FADE) / LEAD_CHAR)));
        const size = Math.max(18, Math.min(W * 0.04, 52));
        ctx.font = `800 ${size}px ${LEAD_FONT}`;
        if ('letterSpacing' in ctx) ctx.letterSpacing = `${(size * 0.1).toFixed(1)}px`; // wide, monumental spacing
        ctx.textBaseline = 'middle';
        ctx.textAlign = 'left';
        let lines = [LEAD_TEXT];
        if (ctx.measureText(LEAD_TEXT).width > W * 0.88) {
          const words = LEAD_TEXT.split(' ');
          const mid = Math.ceil(words.length / 2);
          lines = [words.slice(0, mid).join(' '), words.slice(mid).join(' ')];
        }
        const lh = size * 1.5;
        let left = shown;
        lines.forEach((ln, i) => {
          const part = ln.slice(0, Math.max(0, Math.min(ln.length, left)));
          left -= ln.length + 1;
          const w = ctx.measureText(ln).width;
          const x = (W - w) / 2;
          const y = H / 2 + (i - (lines.length - 1) / 2) * lh;
          ctx.shadowColor = '#ff1c2c';
          ctx.shadowBlur = size * 0.5;
          ctx.fillStyle = '#f4e9e9';
          ctx.fillText(part, x, y);
          ctx.shadowBlur = 0;
          // blinking caret at the end of what has been typed so far
          if (part.length < ln.length || (i === lines.length - 1 && Math.floor(el / 380) % 2 === 0)) {
            if (part.length > 0 || i === 0) {
              ctx.fillStyle = RED;
              ctx.fillRect(x + ctx.measureText(part).width + 3, y - size * 0.55, Math.max(2, size * 0.06), size * 1.1);
            }
          }
        });
      };
      let soundDone = false;
      const start = performance.now();
      const frame = (now) => {
        const el = now - start;
        if (leadAt !== null || (freeze === null && el < T_LEAD)) {
          drawLead(leadAt !== null ? leadAt : el);
          requestAnimationFrame(frame);
          return;
        }
        if (!soundDone && freeze === null) {
          soundDone = true;
          playSound();
        }
        const t = freeze !== null ? freeze : el - T_LEAD;
        const total = T_EYES + T_PULL + T_HOLD + T_FADE;
        if (freeze === null && t >= total) {
          cv.remove();
          playing = false;
          resolve();
          return;
        }
        const fadeStart = T_EYES + T_PULL + T_HOLD;
        cv.style.opacity = freeze !== null || t < fadeStart ? '1' : String(Math.max(0, 1 - (t - fadeStart) / T_FADE));
        const e = reduce ? 1 : ease((t - T_EYES) / T_PULL);
        // the domain, with the camera easing back from a slight zoom
        g.setTransform(1, 0, 0, 1, 0, 0);
        g.fillStyle = '#1a0610';
        g.fillRect(0, 0, VW, VH);
        g.save();
        const zs = 1.22 - 0.22 * e;
        g.translate(CX, HORIZON);
        g.scale(zs, zs);
        g.translate(-CX, -HORIZON);
        scene(g, t, sky, water);
        g.restore();
        // white screen clears away
        g.fillStyle = `rgba(255,255,255,${(1 - e).toFixed(3)})`;
        g.fillRect(0, 0, VW, VH);
