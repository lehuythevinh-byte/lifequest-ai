 function unionArea(rects,W,H){
   const a=rects.map(q=>({x:clamp(q.x,0,W),r:clamp(q.x+q.w,0,W),y:clamp(q.y,0,H),b:clamp(q.y+q.h,0,H)})).filter(q=>q.r>q.x&&q.b>q.y);
   if(!a.length)return 0;
   const X=[...new Set(a.flatMap(q=>[q.x,q.r]))].sort((u,v)=>u-v);
   let total=0;
   for(let i=0;i<X.length-1;i++){
     let segments=a.filter(q=>q.x<X[i+1]&&q.r>X[i]).map(q=>[q.y,q.b]).sort((u,v)=>u[0]-v[0]);
     let lo=-1,hi=-1,covered=0;
     for(const [u,v] of segments){if(lo<0){lo=u;hi=v;}else if(u>hi){covered+=hi-lo;lo=u;hi=v;}else hi=Math.max(hi,v)}
     if(lo>=0)covered+=hi-lo;
     total+=(X[i+1]-X[i])*covered;
   }
   return total;
 }
 function occupancyGrid(p,c=12,rw=9){
   let bits=new Uint8Array(c*rw);
   for(let row=0;row<rw;row++)for(let col=0;col<c;col++){
     let x0=p.W*col/c,y0=p.H*row/rw,x1=p.W*(col+1)/c,y1=p.H*(row+1)/rw;
     const q=p.paths||[];
     bits[row*c+col]=q.some(k=>Math.min(k.x+k.w,x1)-Math.max(k.x,x0)>0.2*(x1-x0)&&Math.min(k.y+k.h,y1)-Math.max(k.y,y0)>0.2*(y1-y0))?1:0;
   }
   return bits;
 }
 function similarity(a,b){
   if(!a||!b||a.area!==b.area)return 0;
   const ga=occupancyGrid(a),gb=occupancyGrid(b);let intersection=0,union=0;
   for(let i=0;i<ga.length;i++){if(ga[i]&&gb[i])intersection++;if(ga[i]||gb[i])union++;}
   const spatial=union?intersection/union:0;
   const size=Math.max(0,1-(Math.abs(a.W-b.W)+Math.abs(a.H-b.H))/(Math.max(a.W,b.W)+Math.max(a.H,b.H)));
   const typ=a.type===b.type?1:0,dom=a.dominant===b.dominant?1:0;
   return Math.round(100*(spatial*.58+size*.24+typ*.12+dom*.06));
 }
 function report(){
   const rooms=[...roomStore.values()],issues=[],similar=[],byMap={};
   for(const p of rooms){
     byMap[p.area]=(byMap[p.area]||0)+1;
     const occupied=unionArea(p.paths,p.W,p.H),ratio=occupied/(p.W*p.H);
     const doorClear=p.paths.every(q=>!(q.x<p.W/2+24&&q.x+q.w>p.W/2-24&&q.y+q.h>p.H-60));
     if(ratio>.42)issues.push({room:p.key,kind:'dense',coverage:Math.round(ratio*100)});
     if(ratio<.035)issues.push({room:p.key,kind:'sparse',coverage:Math.round(ratio*100)});
     if(!doorClear)issues.push({room:p.key,kind:'door-review',coverage:Math.round(ratio*100)});
   }
   for(let i=0;i<rooms.length;i++)for(let j=0;j<i;j++){
