  function setDirs(dx, dy) {
    const c = controls();
    // 8 directions: a component counts once it is a good part of the push (about 22° either side of an axis).
    const len = Math.hypot(dx, dy);
    const on = len > 0.25;
    const ux = on ? dx / len : 0,
      uy = on ? dy / len : 0;
    c.right = on && ux > 0.38;
    c.left = on && ux < -0.38;
    c.down = on && uy > 0.38;
    c.up = on && uy < -0.38;
  }
  function release() {
    pointer = null;
    joy.classList.remove('held');
    knob.style.transform = '';
    setDirs(0, 0);
  }
  function bindJoystick() {
    const move = (ev) => {
      const r = joy.getBoundingClientRect();
      const R = r.width / 2;
      let dx = (ev.clientX - (r.left + R)) / R,
        dy = (ev.clientY - (r.top + R)) / R;
      const len = Math.hypot(dx, dy);
      if (len > 1) (dx /= len), (dy /= len);
      knob.style.transform = `translate(${dx * R * 0.62}px,${dy * R * 0.62}px)`;
      setDirs(dx, dy);
    };
    joy.addEventListener('pointerdown', (ev) => {
      ev.preventDefault();
      pointer = ev.pointerId;
      try {
        joy.setPointerCapture(ev.pointerId);
      } catch (e) {
        /* capture is optional */
      }
      joy.classList.add('held');
      move(ev);
    });
    joy.addEventListener('pointermove', (ev) => {
      if (ev.pointerId === pointer) move(ev);
    });
    for (const t of ['pointerup', 'pointercancel', 'lostpointercapture'])
      joy.addEventListener(t, (ev) => {
        if (ev.pointerId === pointer) release();
      });

      function train(now) {
        const total = 50 + 4 + 50 + 4 + 57;
        const left = Math.round(W / 2 - total / 2);
        const jig = (k) => (speed > 8 && Math.sin(now / 70 + k * 1.7) > 0.7 ? 1 : 0);
        const y = GY - 2;
        carriage(left, y - jig(0), 0, now, -1);
        carriage(left + 54, y - jig(1), 1, now, 2);
        locomotive(left + 108, y - jig(2));
        // smoke
        // Puffs while moving; once stopped, only a thin wisp that fades quickly.
        const moving = speed > 8;
        smokeT += moving ? 1 : 0.12;
        if (smokeT > 7) {
          smokeT = 0;
          smoke.push({ x: left + 108 + 44, y: GY - 37, r: moving ? 2.5 : 1.5, life: 0, max: moving ? 2.4 : 1.1 });
        }
        for (const p of smoke) {
          p.life += 1 / 60;
          p.x -= (speed / 60) * 0.85 + 0.15;
          p.y -= 0.32;
          p.r += 0.07;
        }
        for (let k = smoke.length - 1; k >= 0; k--) if (smoke[k].life > smoke[k].max) smoke.splice(k, 1);
        for (const p of smoke) disc(p.x, p.y, p.r, `rgba(255,255,255,${Math.max(0, 0.75 * (1 - p.life / p.max)).toFixed(2)})`);

      function show(i) {

        if (!typing) return;

        viEl.classList.add('show');

      function next() {
        if (ended) return;
        if (li < 0) {
          show(0);
          return;
        }
        if (typing) {
          finishTyping();
          return;
        }
        if (li < lines.length - 1) show(li + 1);
        else end();
      }
      function end() {
        if (ended) return;
        ended = true;
        root.classList.remove('on');
        removeEventListener('keydown', onKey, true);
        setTimeout(() => {
          cancelAnimationFrame(raf);
          removeEventListener('resize', size);
          root.remove();
          try {
            if (paused && sc && sc.scene) sc.scene.resume();
          } catch (e) {}
          playing = null;
          resolve();
        }, 650);
      }
      const onKey = (ev) => {
        // The scene owns the keyboard while it plays: the game must not move or open menus behind it.
        ev.preventDefault();
        ev.stopImmediatePropagation();
        if (ev.repeat) return;
        if (ev.key === ' ' || ev.key === 'Enter' || ev.key === 'e' || ev.key === 'E' || ev.key === 'ArrowRight') next();
      };
      addEventListener('keydown', onKey, true);
      root.addEventListener('click', (ev) => {
        if (ev.target.closest('.skip')) {
          end();
          return;
        }
        next();
      });

      function frame(now) {
        if (!document.body.contains(root)) return;
        const dt = Math.min(0.05, (now - last) / 1000);
        last = now;
        if (brake) {
          const p = Math.min(1, (now - brake.t0) / 1000 / brake.T);
          cam = brake.c0 + brake.D * (1 - (1 - p) * (1 - p));
          speed = CRUISE * (1 - p);
        } else cam += speed * dt;
        if (li < 0 && now - t0 > 2500) show(0);
        if (typing) {
          typing.t += dt;
          const n = Math.min(typing.text.length, Math.floor(typing.t * 40));
          if (n !== typing.n) {
            typing.n = n;
            enEl.textContent = typing.text.slice(0, n);
          }
          if (n >= typing.text.length) finishTyping();
        }
        draw(now);
        raf = requestAnimationFrame(frame);
      }
      raf = requestAnimationFrame(frame);
    });
    return playing;
  }

      el.querySelector('.vi').textContent = (note ? note.vi + ' ' : '') + shownStep.vi;

  function advance() {
    const g = game();
    if (!g || !g.tut) return;
    const list = steps();
    g.tut.step = Math.min(list.length, (g.tut.step | 0) + 1);
    base = null;
    if (g.tut.step >= list.length) finish(false);
    else save();
  }
  function jumpTo(id) {
    const g = game();
    const i = stepIndex(id);
    if (!g || !g.tut || i < 0 || (g.tut.step | 0) >= i) return;
    g.tut.step = i;
    base = null;
    save();
  }

    else toast('🎓 Tutorial complete', 'Welcome to English Town!', 'Finish “First Day in Town” on the quest card.');
  }

    const msgs = (g.stats && g.stats.messages) || 0;
    if (!base || base.i !== i) base = { i, x: sc.player.x, y: sc.player.y, msgs };
    const chatId = sc.chat && sc.chat.isOpen ? sc.chat.currentId : null;

    // Talking to Mia early skips the steps before it.
    if (chatId === 'kid' && i < stepIndex('say')) return jumpTo('say');

    const mia = sc.npcs && sc.npcs.find((n) => n.def && n.def.id === 'kid');
    const miaHere = !!(mia && mia.sprite && mia.sprite.visible && !!mia.inside === !!sc.inside);
    const dMia = miaHere ? Math.hypot(sc.player.x - mia.sprite.x, sc.player.y - mia.sprite.y) : Infinity;

    let shownStep = s;
    let note = null;
    if (s.id === 'move' && Math.hypot(sc.player.x - base.x, sc.player.y - base.y) > 80) return advance();
    if (s.id === 'find' && dMia < 90) return advance();
    if (s.id === 'talk' && dMia > 220) shownStep = list[stepIndex('find')];
    if (s.id === 'say') {
      if (chatId !== 'kid') {
        shownStep = list[stepIndex('talk')];
        note = { en: 'Talk to Mia again.', vi: 'Hãy nói chuyện lại với Mia.' };
        if (dMia > 220) shownStep = list[stepIndex('find')];
      } else if (msgs > base.msgs) return advance();
    }
    if (s.id === 'tips' && !chatId) return advance();
    if (s.id === 'menu' && shown('l11-shell')) return advance();

    // Hide behind big panels, and while chatting with someone other than Mia (or chatting at all on the last steps).
    const chatOk = s.id === 'say' || s.id === 'tips';
    if (COVERS.some(shown) || (chatId && (!chatOk || chatId !== 'kid'))) return hideCoach();
    renderCoach(i, list, shownStep, note);
    // On small screens the chat is tall: the card moves up over the minimap so it does not hide the chat's ✕.
    coachEl.classList.toggle('chatting', !!chatId);
  }
