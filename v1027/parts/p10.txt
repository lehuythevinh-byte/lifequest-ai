        function findMention(s, entries, aliases) {
          const hits = [];
          for (const x of entries || []) {
            let names = [

              x.name,
              String(x.name)
                .replace(/\([^)]*\)/g, '')

              ...(aliases[x.name] || []),

              if (n.length > 2 && has(s, n)) hits.push({ x, len: n.length });
            }

            return '⚠️ Gemini rejected the request (400). Check the model name and try a shorter message.';

        function paintWeather(now) {
          const c = $('v8-weather-canvas');
          if (!c) return;
          if (!preferences.weather || preferences.performance === 'low') {
            const cx = c.getContext('2d');
            cx?.clearRect(0, 0, c.width, c.height);
            return;
          }
          const sc = getScene();
          if (!sc || sc.inside || weatherType === 'clear') {
            c.getContext('2d')?.clearRect(0, 0, c.width, c.height);
            return;
          }
          const fps = preferences.performance === 'high' ? 30 : 20;
          if (now - fxFrame < 1000 / fps) return;
          fxFrame = now;
          const ctx = c.getContext('2d');
          if (!ctx) return;
          ctx.clearRect(0, 0, c.width, c.height);
          const count =
            weatherType === 'rain' ? (preferences.performance === 'high' ? 65 : 36) : 18;
          while (fxParticles.length < count)
            fxParticles.push({
              x: Math.random() * c.width,
              y: Math.random() * c.height,
              speed: 1 + Math.random() * 2.4,
            });
          if (fxParticles.length > count) fxParticles.length = count;
          ctx.strokeStyle =
            weatherType === 'rain' ? 'rgba(164,209,250,0.45)' : 'rgba(234,244,255,0.16)';
          ctx.lineWidth = weatherType === 'rain' ? 1.2 : 8;
          for (const p of fxParticles) {
            p.x += weatherType === 'rain' ? 2 : 0.3;
            p.y += weatherType === 'rain' ? 6 * p.speed : 0.2 * p.speed;
            if (p.y > c.height) {
              p.y = -15;
              p.x = Math.random() * c.width;
            }
            if (p.x > c.width) p.x = 0;
            ctx.beginPath();
            ctx.moveTo(p.x, p.y);
            ctx.lineTo(
              p.x + (weatherType === 'rain' ? 3 : 17),
              p.y + (weatherType === 'rain' ? 14 : 7),
            );
            ctx.stroke();
          }
        }

        function transactBuy(shop, id, qty) {
          const sc = scene(),
            s = state(),
            def = catalog[id],
            g = game();
          if (!sc || !s || !g) return { ok: false, message: 'The game is not ready yet.' };
          if (busy) return { ok: false, message: 'A transaction is already being processed.' };
          const problem = checkShop(shop);
          if (problem) return { ok: false, message: problem };
          if (!def || !def.shops.includes(shop))
            return { ok: false, message: 'This item is not sold at this shop.' };
          const n = amount(qty);
          if (!Number.isFinite(n) || n > 10)
            return { ok: false, message: 'You can only buy 1–10 items at a time.' };
          if (def.effect === 'room' && n !== 1)
            return { ok: false, message: 'You can only rent one room at a time.' };
          const sig = `${shop}:${id}:${n}`;
          if (lastPurchase.signature === sig && Date.now() - lastPurchase.at < 700)
            return {
              ok: false,
              message: 'Your previous transaction was received. Please wait a moment before buying again.',
            };
          const total = def.price * n;
          if (!validInt(total) || !validInt(g.money) || g.money < total)
            return {
              ok: false,
              message: `You need $${Math.max(0, total - g.money)} more. Complete quests to earn money.`,
            };
          if (def.unique && count(id) > 0)
            return { ok: false, message: 'You already own this item.' };
          if (def.unique && n !== 1)
            return { ok: false, message: 'Upgrade items can only be bought one at a time.' };
          if (def.category !== 'service' && usedSpace() + n > capacity())
            return {
              ok: false,
              message: `Your bag only has ${Math.max(0, capacity() - usedSpace())} free slots. Use some items or buy a backpack.`,
            };
          busy = true;
          try {
            g.money -= total;
            if (def.category === 'service' && def.effect === 'room') {
              s.flags.roomUntilDay = g.day;
              s.flags.roomShop = shop;
              s.flags.roomName = id;
            } else if (def.category === 'service' && def.effect === 'mail') {
              s.flags.pendingMail = (Number(s.flags.pendingMail) || 0) + n;
            } else if (def.category !== 'service') {
              inventory()[id] = count(id) + n;
            }
            if (def.effect === 'lesson') s.unlocked[def.lesson] = true;
            log('purchase', -total, `${def.name} ×${n}`, { shop, item: id, qty: n });
            lastPurchase = { signature: sig, at: Date.now() };
            save();
            return {
              ok: true,
              message: `✅ Bought ${def.name} ×${n} at ${shop}. -$${total}. “Could I have a receipt, please?”`,
            };
          } finally {
            busy = false;
          }
        }
        function bankAction(kind, raw) {
          const c = context(),
            g = game(),
            s = state();
          if (!g || !s) return 'The game is not ready yet.';
          if (!canShop('Bank')) return 'You must go to the Bank to deposit or withdraw money.';
          const n = Number(raw);
          if (!validInt(n, 1, 1000000)) return 'Enter a whole amount from 1 to 1,000,000.';
          if (kind === 'deposit') {
            if (!validInt(g.money) || g.money < n) return 'You do not have enough cash in your wallet.';
            if (s.bank + n > 1000000000) return 'This exceeds the account limit.';
            g.money -= n;
            s.bank += n;
            log('deposit', -n, `Savings deposit $${n}`);
          } else if (kind === 'withdraw') {
            if (s.bank < n || g.money + n > 1000000000)
              return 'Insufficient balance or over the safe limit.';
            s.bank -= n;
            g.money += n;
            log('withdraw', n, `Savings withdrawal $${n}`);
          } else return 'Invalid command.';
          save();
          return `🏦 Success! Wallet $${g.money}; savings $${s.bank}. Your total assets are unchanged because money just moved between two accounts.`;
        }

        function consume(id) {
          const def = catalog[id],
            sc = scene(),
            g = game(),
            s = state();
          if (!def || !sc || count(id) < 1) return 'You do not own this item.';
          if (def.effect === 'energy') {
            g.rpg.energy = Math.min(100, (Number(g.rpg.energy) || 0) + def.energy);
            moveItem(id, -1);
            log('use', 0, `Used ${def.name}`);
            save();
            return `✨ ${def.name}: +${def.energy} energy (max 100).`;
          }
          if (def.effect === 'equip') {
            s.flags[id + 'Equipped'] = !s.flags[id + 'Equipped'];
            save();
            return s.flags[id + 'Equipped'] ? `✅ Equipped ${def.name}.` : `Put away ${def.name}.`;
          }
          if (def.effect === 'lesson') {
            openTab = 'learning';
            show('learning');
            return `📖 Opened lesson ${def.lesson || 'English'}.`;
          }
          if (def.effect === 'upgrade') {
            s.flags.largeBag = true;
            save();
            return '🎒 Large backpack: 48 inventory slots.';
          }
          if (def.effect === 'gift') {
            const npc = context().near[0];
            if (!npc) return 'Go near an NPC and open your inventory to give a gift.';
            const day = sc.store.game.day;
            const key = `${npc}_${day}`;
            if (s.gifts[key]) return 'You already gave this NPC a gift today.';
            s.gifts[key] = true;
            moveItem(id, -1);
            if (g.rpg.relationships?.[npc])
              g.rpg.relationships[npc].trust = Math.min(
                100,
                (Number(g.rpg.relationships[npc].trust) || 0) + 5,
              );
            sc.store.get(npc).affinity = Math.min(100, (sc.store.get(npc).affinity || 0) + 4);
            log('gift', 0, `Gave ${def.name} to ${npc}`);
            save();
            return `🎁 Gave ${def.name}! Your relationship with ${npc} improved.`;
          }
          if (def.effect === 'travel') {
            return 'Go to platform 1 and step onto the front of the train. The conductor will check your ticket on board.';
          }
          if (false) {
            const c = context();
            if (!canShop('Rail Station')) return 'Go to the Rail Station to use your train ticket.';
            const portals = sc.area?.portals || {};
            const trainRoutes = {
              'english-town': {
                to: 'mining-town',
                spawn: { x: 30, y: 50 },
                label: 'Minerals Central Station',
              },
              'mining-town': {
                to: 'english-town',
                spawn: { x: 30, y: 50 },
                label: 'English Town Central Station',
              },
            };
            const target =
              trainRoutes[sc.area?.id] ||
              (sc.area?.warps || []).find((p) => p?.to && p.to !== sc.area.id) ||
              Object.values(portals).find((p) => p?.to && p.to !== sc.area.id);
            if (!target) return 'The station has no valid route from this area; your ticket is kept.';
            moveItem(id, -1);
            log('travel', 0, `Train trip: ${sc.area?.name || ''}`);
            save();
            try {
              sc.switchArea(target.to, target.spawn, target.label || 'Rail Station');
              return '🚂 Traveling to the next town.';
            } catch {
              moveItem(id, 1);
              save();
              return 'Travel failed; your ticket was refunded.';
            }
          }
          if (def.effect === 'repair') {
            if (context().inside !== 'Smithy') return 'Go to the Smithy to use the repair kit.';
            moveItem(id, -1);
            const day = g.day;
            const key = `repair-${day}`;
            if (!s.jobs[key]) {
              s.jobs[key] = true;
              g.rpg.xp = (Number(g.rpg.xp) || 0) + 20;
              log('work', 0, 'Repair practice +20 XP');
            }
            save();
            return '🧰 You practiced repairing tools. +20 XP if this is your first time today.';
          }
          if (def.effect === 'decorate') {
            if (context().inside !== "Hero's Home")
              return 'Go to Hero’s Home to place the items you bought.';
            const slots = [0, 1, 2];
            const index = Number(s.homeSlots[id] ?? 0);
            s.homeSlots[id] = slots[(index + 1) % slots.length];
            renderHomeDecor();
            save();
            return `🏠 Moved ${def.name} (position ${s.homeSlots[id] + 1}/3).`;
          }
          if (def.effect === 'display') {
            return `💎 ${def.name} is kept as a collectible. You can show it off to NPCs.`;
          }
          return `${def.name}: find the right place to use it.`;
        }
        function restAction() {
          const sc = scene(),
            s = state(),
            g = game();
          const place = context().inside;
          if (!['Hotel', 'Mining Inn'].includes(place))
            return 'Go inside the Hotel or Mining Inn to rest overnight.';
          if (!s || s.flags.roomShop !== place || Number(s.flags.roomUntilDay) !== g.day)
            return 'You need to rent a valid room at the inn desk first.';
          if (sc.sleeping) return 'You are already resting.';
          sc.sleeping = true;
          try {
            g.minutes = 420;
            sc.newDay();
            sc.snapNpcs();
            s.flags.roomUntilDay = -1;
            s.flags.roomShop = '';
            log('rest', 0, `Overnight rest at ${place}`);
            save();
            return '🌅 Good morning! You rested overnight at the inn.';
          } finally {
            sc.sleeping = false;
          }
        }

        const RAIL_CLERKS = ['ticketclerk', 'etclerk'],

        function railPop(icon, text) {
          let n = $('lq-rail-pop');
          if (!n) {
            n = document.createElement('div');
            n.id = 'lq-rail-pop';
            document.body.appendChild(n);
          }
          n.innerHTML = '<span>' + icon + '</span><b>' + esc(text) + '</b>';
          n.classList.add('shown');
          clearTimeout(railPop.t);
          railPop.t = setTimeout(() => n.classList.remove('shown'), 4200);
          railBadge();
        }
        function railBadge() {
          const b = $('lq-rail-badge');
          if (!b) return;
          const p = count('boarding_pass'),
            k = count('rail_ticket');
          if (p > 0) {
            b.hidden = false;
            b.className = 'ok';
            b.textContent = '✅ Ticket checked · have a safe trip';
          } else if (k > 0) {
            b.hidden = false;
            b.className = '';
            b.textContent = '🎫 Train ticket · go to platform 1 and step onto the front of the train to board';
          } else b.hidden = true;
        }

        function railFlow(id, message) {
          const g = game();
          if (!g || !state()) return [];
          const clerk = RAIL_CLERKS.includes(id),
            checker = RAIL_CHECKERS.includes(id);
          if (!clerk && !checker) return [];
          if (!canShop('Rail Station')) return [];
          const text = norm(message);
          if (clerk) {
            if (!/(ticket|tickets|travel|train|go to|trip|minerals|english town)/.test(text))
              return [];
            if (count('boarding_pass') > 0 || count('rail_ticket') > 0)
              return [
                '🎫 You already have a ticket. Go to platform 1 and step onto the front of the train; the conductor will check your ticket on board. “Please board the train at platform one.”',
              ];
            if (usedSpace() >= capacity())
              return ['🎒 Your bag is full. Free up some space, then ask for the ticket again.'];
            moveItem('rail_ticket', 1);
            log('travel', 0, 'Received a free train ticket');
            save();
            railPop('🎫', 'You received: Train ticket');
            return [
              '🎫 Your free train ticket is in your bag. “Here is your ticket. Please board the train at platform one.” Step onto the front of the train at platform 1; your ticket will be checked on board.',
            ];
          }
          if (!/(ticket|train|platform|board|check|inspect|here|please)/.test(text)) return [];
          if (count('rail_ticket') > 0 || count('boarding_pass') > 0)
            return [
              '🚆 Your ticket is good. Step onto the front of the train at platform 1; the conductor will check your ticket on board. “The conductor will check your ticket on the train.”',
            ];
          return [
            '⚠️ You do not have a ticket yet. Ask for one at the ticket window (window one) in English first. “Tickets are at window one.”',
          ];
        }
