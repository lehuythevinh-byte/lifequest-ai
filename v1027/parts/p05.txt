          setAutoSpeak(auto) {
            this.settings.autoSpeak = auto;
            this.saveSettings();
          }
          setRate(rate) {
            this.settings.rate = Math.max(0.6, Math.min(1.4, rate));
            this.saveSettings();
          }
          isAvailable() {
            return typeof window !== 'undefined' && 'speechSynthesis' in window;
          }
          isSpeaking() {
            return this.isSpeakingState;
          }
          /**
           * Speak an English sentence.
           * Strips out markdown, emoji, and formatting so the speech output sounds completely natural.
           */
          speak(text, callbacks) {
            if (!this.synth || !this.settings.voiceEnabled) return;
            // Stop current speech
            this.stop();
            // Clean text for speech synthesis
            const cleanText = text
              .replace(/https?:\/\/\S+/g, '') // remove URLs
              .replace(/[*_#`~[\]]/g, '') // remove markdown
              .replace(
                /[\u{1F600}-\u{1F64F}\u{1F300}-\u{1F5FF}\u{1F680}-\u{1F6FF}\u{1F1E0}-\u{1F1FF}]/gu,
                '',
              ) // remove emojis
              .trim();
            if (!cleanText) return;
            const utterance = new SpeechSynthesisUtterance(cleanText);
            utterance.lang = this.selectedVoice?.lang || 'en-US';
            if (this.selectedVoice) {
              utterance.voice = this.selectedVoice;
            }
            utterance.rate = this.settings.rate;
            utterance.pitch = this.settings.pitch;
            utterance.onstart = () => {
              this.isSpeakingState = true;
              callbacks?.onStart?.();
              this.notify();
            };
            utterance.onend = () => {
              this.isSpeakingState = false;
              this.currentUtterance = null;
              callbacks?.onEnd?.();
              this.notify();
            };
            utterance.onerror = (e) => {
              this.isSpeakingState = false;
              this.currentUtterance = null;
              if (e.error !== 'interrupted' && e.error !== 'canceled') {
                console.warn('Speech synthesis error:', e);
                callbacks?.onError?.(e);
              }
              this.notify();
            };
            this.currentUtterance = utterance;
            try {
              this.synth.speak(utterance);
            } catch (err) {
              console.warn('Failed to speak utterance:', err);
            }
          }
          stop() {
            if (this.synth) {
              try {
                this.synth.cancel();
              } catch (e) {}
              this.isSpeakingState = false;
              this.currentUtterance = null;
              this.notify();
            }
          }
          subscribe(listener) {
            this.listeners.push(listener);
            return () => {
              this.listeners = this.listeners.filter((l) => l !== listener);
            };
          }
          notify() {
            this.listeners.forEach((l) => l());
          }
        }
        const speechService = new SpeechService();

        return { speechService };
      })();

      const {
        attachTownScene,
        detachTownScene,
        updateTownAutopilot,
        cancelTownAutopilot,
        executeTownCommand,
        getTownContext,

        /**
         * Non-visual bridge between LIFEQUEST AI and the existing Phaser scene.
         * Do not change any terrain, sprites, drawings, or world-generation rules here.
         */
        let scene = null;
        let route = null;

        function notify(text) {
          if (typeof window !== 'undefined') {
            window.dispatchEvent(new CustomEvent('lifequest:status', { detail: { text } }));
          }
        }

        function attachTownScene(current) {
          scene = current;
          window.__lifequestGetScene = () => scene;
          window.__lifequestServices = { speech: speechService, cefr: cefrService };
          route = null;
        }

        function detachTownScene(current) {
          if (scene !== current) return;
          scene = null;
          route = null;
        }

        function getTownContext() {
          if (!scene || !scene.player || !scene.area) return null;
          const { area, player, npcs, inside, store } = scene;
          return {
            area: area.name,
            areaId: area.id,
            inside: inside?.b?.name || null,
            day: store.game.day,
            time:
              Math.floor(store.game.minutes / 60)
                .toString()
                .padStart(2, '0') +
              ':' +
              (store.game.minutes % 60).toString().padStart(2, '0'),
            position: { x: Math.floor(player.x / 32), y: Math.floor(player.y / 32) },
            nearby: npcs
              .filter((n) => n.sprite.visible && !!n.inside === !!inside)
              .filter(
                (n) =>
                  Math.hypot(n.sprite.x - player.x, n.sprite.y - player.y) < (inside ? 95 : 80),
              )
              .map((n) => ({ id: n.def.id, name: n.def.name })),
            npcs: npcs.map((n) => ({
              id: n.def.id,
              name: n.def.name,
              available: !!n.sprite.visible && !!n.inside === !!inside,
              indoors: !!n.inside,
              activity: n.slot.label,
              tile: { x: Math.floor(n.sprite.x / 32), y: Math.floor(n.sprite.y / 32) },
            })),
            places: [
              ...area.buildings.map((b) => ({
                id: b.name,
                name: b.name,
                type: 'building',
                tile: { x: b.door, y: b.y + b.h },
              })),
              ...Object.keys(area.places)
                .filter((k) => !k.startsWith('home_'))
                .map((k) => ({
                  id: k,
                  name: k,
                  type: 'landmark',
                  tile: { x: area.places[k].x, y: area.places[k].y },
                })),
            ],
            isMoving: !!route,
          };
        }

        function cancelTownAutopilot(reason = '', announce = false) {
          if (!route) return;
          route = null;
          if (reason && announce) notify(reason);
        }

        const tile = (x, y) => ({ x: Math.floor(x / 32), y: Math.floor(y / 32) });
        const center = (x, y) => ({ x: x * 32 + 16, y: y * 32 + 16 });

        function routeToPoint(targetX, targetY, maxDistanceTiles = 0, preferExact = false) {
          const { player, area } = scene;
          const target = tile(targetX, targetY);
          const candidates = [];
          for (let rad = 0; rad <= maxDistanceTiles; rad++) {
            for (let dy = -rad; dy <= rad; dy++) {
              for (let dx = -rad; dx <= rad; dx++) {
                if (Math.max(Math.abs(dx), Math.abs(dy)) !== rad) continue;
                const x = target.x + dx,
                  y = target.y + dy;
                if (area.isTileBlocked(x, y)) continue;
                const point = center(x, y);
                if (scene.blocked(point.x, point.y)) continue;
                // The original findPath returns a one-node fallback for an unreachable goal;
                // validate every step before using the route.
                const steps = area.findPath({ x: player.x, y: player.y }, { x, y });
                if (!Array.isArray(steps) || !steps.length || steps.length > 1600) continue;
                const end = steps[steps.length - 1];
                if (Math.hypot(end.x - point.x, end.y - point.y) > 2) continue;
                let last = center(Math.floor(player.x / 32), Math.floor(player.y / 32));
                if (
                  !steps.every((p) => {
                    const valid =
                      !area.isTileBlocked(Math.floor(p.x / 32), Math.floor(p.y / 32)) &&
                      !scene.blocked(p.x, p.y) &&
                      Math.hypot(p.x - last.x, p.y - last.y) <= 46;
                    last = p;
                    return valid;
                  })
                )
                  continue;
                const proximity = Math.hypot(point.x - targetX, point.y - targetY);
                if (preferExact && proximity > 23) continue;
                candidates.push({ steps, cost: steps.length + proximity / 72 });
              }
            }
          }
          candidates.sort((a, b) => a.cost - b.cost);
          return candidates[0]?.steps || null;
        }

        function beginRoute(points, label, npcId = null, autoTalk = false) {
          if (!points)
            return `Couldn't find a valid path to ${label}. Try getting closer with WASD.`;
          route = { kind: 'path', points: [...points], label, npcId, autoTalk, retries: 0 };
          return `🚶 Your character is finding a path to ${label}. Press WASD/arrow keys to take back control, or say “stop”.`;
        }

        function canAct() {
          return (
            !!scene &&
            !scene.switching &&
            !scene.sleeping &&
            !scene.chat.isOpen &&
            !scene.briefing.isOpen &&
            !scene.settings.isOpen &&
            !scene.helper.isOpen &&
            !scene.journal.isOpen &&
            !scene.quests.isOpen
          );
        }

        function executeTownCommand(command) {
          if (!scene || !scene.player) return 'The game hasn\'t finished loading. Please try again.';
          const action = command?.kind;
          if (action === 'stop') {
            cancelTownAutopilot();
            return '🛑 Autopilot stopped. You can use WASD or the arrow keys.';
          }
          if (action === 'status') {
            const data = getTownContext();
            return `📍 You are at ${data.inside || data.area}, position (${data.position.x}, ${data.position.y}), ${data.time}.`;
          }
          if (action === 'help') {
            return 'You can say: “go to the library”, “find Lily and talk to her”, “walk right for 2 seconds”, “talk to the nearest person”, “stop”, or ask how to learn English. Manual controls: WASD/arrow keys, E to chat.';
          }
          if (!canAct())
            return 'Close the NPC window, journal or in-game menu before controlling your character.';
          if (action === 'talk') {
            const npc = scene.nearestNpc(scene.inside ? 56 * scene.inside.k : 48);
            if (!npc) return 'No NPC is close enough. Walk up to an NPC and try again.';
            cancelTownAutopilot();
            scene.tryTalk();
            return `💬 Opened a conversation with ${npc.def.name}. You can type or use the microphone to speak English.`;
          }
          if (action === 'move') {
            const dirs = { up: [0, -1], down: [0, 1], left: [-1, 0], right: [1, 0] };
            if (!Object.hasOwn(dirs, command.direction)) return 'Invalid direction.';
            const seconds = Math.min(3, Math.max(0.25, Number(command.seconds) || 1));
            cancelTownAutopilot();
            route = {
              kind: 'direction',
              vector: dirs[command.direction],
              remaining: seconds * 1000,
            };
            return `🎮 Moving ${command.direction} for ${seconds} second(s).`;
          }
          if (scene.inside)
            return 'You are indoors. Go outside before using auto-navigation.';
          if (action === 'navigate_npc') {
            const npc = scene.npcs.find((n) => n.def.id === command.target);
            if (!npc) return 'This character wasn\'t found in the current area.';
            if (npc.inside)
              return `${npc.def.name} is inside a building. Enter that building first and talk in person.`;
            if (!npc.sprite.visible)
              return `${npc.def.name} is resting or not around right now. Try again at another time.`;
            const near = Math.hypot(scene.player.x - npc.sprite.x, scene.player.y - npc.sprite.y);
            if (near <= 45) {
              if (command.autoTalk) {
                scene.chat.open(npc.def);
                return `💬 Started chatting with ${npc.def.name}.`;
              }
              return `You are already next to ${npc.def.name}. Press E to chat.`;
            }
            const points = routeToPoint(npc.sprite.x, npc.sprite.y, 2);
            return beginRoute(points, npc.def.name, npc.def.id, !!command.autoTalk);
          }
          if (action === 'navigate_place') {
            const building = scene.area.buildings.find((b) => b.name === command.target);
            let x, y;
            if (building) {
              x = building.door * 32 + 16;
              y = (building.y + building.h) * 32 + 16;
            } else {
              const p = scene.area.places[command.target];
              if (!p) return 'This place wasn\'t found in the current town.';
              x = p.x * 32 + 16;
              y = p.y * 32 + 16;
            }
            if (Math.hypot(scene.player.x - x, scene.player.y - y) < 20)
              return `You are right at ${command.target}.`;
            const points = routeToPoint(x, y, building ? 1 : 2, !!building);
            return beginRoute(points, command.target);
          }
          return 'I didn\'t understand that command. Try “go to the library”, “go to Lily”, “stop”, or “help”.';
        }

        /** Called once per game update; returns movement vector, or null for manual input. */
        function updateTownAutopilot(current, delta) {
          if (!route || current !== scene) return null;
          if (!canAct()) return null;
          if (route.kind === 'direction') {
            route.remaining -= delta;
            if (route.remaining <= 0) {
              route = null;
              notify('✅ Movement command completed.');
              return null;
            }
            return { x: route.vector[0], y: route.vector[1] };
          }
          // Check destination NPC's *current* position (NPCs may wander over time).
          const npc = route.npcId && scene.npcs.find((n) => n.def.id === route.npcId);
          if (
            npc &&
            Math.hypot(scene.player.x - npc.sprite.x, scene.player.y - npc.sprite.y) <= 44
          ) {
            const label = route.label;
            const autoTalk = route.autoTalk;
            route = null;
            if (autoTalk && !scene.chat.isOpen) {
              scene.chat.open(npc.def);
              notify(`💬 Reached ${label} and opened a conversation.`);
            } else notify(`✅ Reached ${label}. Press E to chat.`);
            return null;
          }
          while (route?.points.length) {
            const waypoint = route.points[0];
            const gap = Math.hypot(waypoint.x - scene.player.x, waypoint.y - scene.player.y);
            if (gap > 5)
              return {
                x: (waypoint.x - scene.player.x) / gap,
                y: (waypoint.y - scene.player.y) / gap,
              };
            route.points.shift();
          }
          if (route) {
            const label = route.label;
            const autoTalk = route.autoTalk;
            const npcId = route.npcId;
            if (npc && route.retries < 2) {
              const points = routeToPoint(npc.sprite.x, npc.sprite.y, 2);
              if (points) {
                route = {
                  kind: 'path',
                  points,
                  label,
                  npcId,
                  autoTalk,
                  retries: route.retries + 1,
                };
                return null;
              }
            }
            route = null;
            notify(
              npc
                ? `📍 Got close to ${label}; the NPC has moved, look around.`
                : `✅ Reached ${label}.`,
            );
          }
          return null;
        }

        return {
          attachTownScene,
          detachTownScene,
          updateTownAutopilot,
          cancelTownAutopilot,
          executeTownCommand,
          getTownContext,
        };
      })();
