 function followerRoute(sc,start,goal){const W=sc.area.w,H=sc.area.h,T=TILE;
  const sx=clamp(Math.floor(start.x/T),1,W-2),sy=clamp(Math.floor(start.y/T),1,H-2);
  const gx=clamp(Math.floor(goal.x/T),1,W-2),gy=clamp(Math.floor(goal.y/T),1,H-2);
  const allowed=(x,y)=>x>=1&&y>=1&&x<W-1&&y<H-1&&Math.abs(x-sx)<=22&&Math.abs(y-sy)<=22&&!sc.blocked?.((x+.5)*T,(y+.5)*T)&&!sc.area.warpAt?.(x,y);
  if(!allowed(sx,sy))return[];
  const id=(x,y)=>y*W+x,q=[[sx,sy]],visited=new Set([id(sx,sy)]),parents=new Map();let k=0,stop=null,best=id(sx,sy),mind=Infinity;
  while(k<q.length&&k<1450){const [x,y]=q[k++],code=id(x,y),dis=Math.abs(x-gx)+Math.abs(y-gy);if(dis<mind){mind=dis;best=code}if(dis<=2){stop=code;break}
   for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){const xx=x+dx,yy=y+dy,key=id(xx,yy);if(visited.has(key)||!allowed(xx,yy))continue;visited.add(key);parents.set(key,code);q.push([xx,yy])}
  }
  const target=stop??best,origin=id(sx,sy),pts=[];let node=target;
  while(node!==origin&&parents.has(node)&&pts.length<100){pts.push({x:(node%W+.5)*T,y:(Math.floor(node/W)+.5)*T});node=parents.get(node)}
  pts.reverse();return pts;
 }
