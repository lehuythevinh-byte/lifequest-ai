  // ---- painting (Phaser Graphics, called once for the area texture) ------------------------------------------------
  // Everything is drawn with code (no image files). The world is shown through a pale filter (applyFilter), so the
  // drawing relies on light/shadow more than on colour: every object gets a soft shadow, a lit top and a dark base.
  function rnd(seed) {
    let s = seed | 0;
    return () => {
      s = (s + 0x6d2b79f5) | 0;
      let t = Math.imul(s ^ (s >>> 15), 1 | s);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }
  // Multiply a colour (k < 1 darker, k > 1 lighter).
  const shade = (c, k) => {
    const f = (v) => Math.max(0, Math.min(255, Math.round(v * k)));
    return (f((c >> 16) & 255) << 16) | (f((c >> 8) & 255) << 8) | f(c & 255);
  };
  const T = 32;
  let R = rnd(1);
  const pick = (arr) => arr[Math.floor(R() * arr.length)];
  // Things that stick up above their tile: drawn as their own sprites, in front of or behind the player.
  const TALL = new Set(['tower', 'tree', 'well', 'banner', 'torch', 'stall', 'coop', 'rack']);
  const shadow = (g, cx, cy, w, h, a = 0.22) => g.fillStyle(0x1a160f, a).fillEllipse(cx, cy, w, h);
  const poly = (g, color, alpha, pts) => g.fillStyle(color, alpha).fillPoints(pts.map(([x, y]) => ({ x, y })), true);

  function paint(g, a) {
    if (a.id !== ID && window.LQFront) return window.LQFront.paint(g, a);
    R = rnd(7070);
    const W2 = a.w * T;
    const H2 = a.h * T;
    // 1. Grass: three tones per tile, then blades, clover and a few pale flowers.
    // Soft patches instead of per-tile squares, so no grid shows through.
    g.fillStyle(0x8e9a68, 1).fillRect(0, 0, W2, H2);
    for (let i = 0; i < a.w * a.h * 0.9; i++) {
      const c = pick([0x96a271, 0x86925f, 0x9aa676, 0x8a965f]);
      g.fillStyle(c, 0.35).fillEllipse(R() * W2, R() * H2, 30 + R() * 60, 18 + R() * 36);
    }
    for (let i = 0; i < a.w * a.h * 0.25; i++) g.fillStyle(0x7f8b56, 0.25).fillEllipse(R() * W2, R() * H2, 10 + R() * 18, 6 + R() * 10);
    for (let i = 0; i < a.w * a.h * 1.6; i++) {
      const px = R() * W2;
      const py = R() * H2;
      const c = R() < 0.6 ? 0x6c7848 : 0xa6b07c;
      g.lineStyle(1.4, c, 0.9).lineBetween(px, py, px - 2, py - 5).lineBetween(px, py, px + 1, py - 6).lineBetween(px, py, px + 3, py - 4);
    }
    for (let i = 0; i < 160; i++) {
      const px = R() * W2;
      const py = R() * H2;
      g.fillStyle(R() < 0.5 ? 0xe9e2c8 : 0xd8c27a, 0.85).fillCircle(px, py, 1.6).fillCircle(px + 3, py + 1, 1.4).fillCircle(px + 1, py - 2, 1.4);
    }

    // 2. The village square: trodden earth with a soft grassy edge and pebbles.
    const sq = { x: SQUARE.x * T, y: SQUARE.y * T, w: SQUARE.w * T, h: SQUARE.h * T };
    g.fillStyle(0xb6a582, 1).fillRoundedRect(sq.x - 6, sq.y - 6, sq.w + 12, sq.h + 12, 26);
    g.fillStyle(0xbfae8b, 1).fillRoundedRect(sq.x + 10, sq.y + 8, sq.w - 20, sq.h - 16, 20);

    // 3. Dirt roads: worn middle, two wheel ruts, stones, grass creeping over the edges.
    const dirt = (x, y, w, h, vertical) => {
      g.fillStyle(0xa8966f, 1).fillRect(x, y, w, h);
      if (vertical) g.fillStyle(0xbba985, 1).fillRect(x + 8, y, w - 16, h);
      else g.fillStyle(0xbba985, 1).fillRect(x, y + 8, w, h - 16);
      const rut = (rx, ry, rw, rh) => g.fillStyle(0x92805e, 0.8).fillRect(rx, ry, rw, rh);
      if (vertical) {
        rut(x + w * 0.3, y, 4, h);
        rut(x + w * 0.66, y, 4, h);
      } else {
        rut(x, y + h * 0.32, w, 4);
        rut(x, y + h * 0.66, w, 4);
      }

        const px = x + R() * w;
        const py = y + R() * h;

        [0, 1].forEach((side) => {

    // crossings: smooth the middle again
    a.roadsH.forEach((h) => a.roadsV.forEach((v) => g.fillStyle(0xb4a27e, 1).fillRect(v.x * T + 6, h.y * T + 6, v.w * T - 12, h.h * T - 12)));

    // 4. Flat objects only. Tall things (buildings, towers, trees…) are separate sprites sorted against the player,
    //    so the player can walk behind them (see tallSprites below).
    [...a.obstacles].filter((o) => !TALL.has(o.kind)).sort((p, q) => p.y + p.h - (q.y + q.h)).forEach((o) => drawThing(g, o, a));
    // 6. The way back to today
    a.warps.forEach((w) => {
      const cx = (w.x + w.w / 2) * T;
      const cy = (w.y + 0.5) * T;
      for (let i = 0; i < 6; i++) g.fillStyle(0xffe6a8, 0.07).fillEllipse(cx, cy, w.w * T + 40 - i * 9, T + 22 - i * 4);
      g.fillStyle(0xfff6dc, 0.55).fillEllipse(cx, cy, w.w * T - 16, 12);
      for (let i = 0; i < 14; i++) g.fillStyle(0xfff6dc, 0.7).fillCircle(cx + (R() - 0.5) * w.w * T, cy - R() * 26, 1.5);
    });
  }

  // Stone wall: irregular blocks, a lit top, crenels on the outer rows and a shadow on the ground below.
  function wallTile(g, o) {
    const x = o.x * T;
    const y = o.y * T;
    const r = rnd(o.x * 131 + o.y * 977);
    const STONES = [0x8e897f, 0x98938a, 0x86817a, 0xa19c92];
    g.fillStyle(0x1a160f, 0.18).fillRect(x, y + T, T, 6);
    g.fillStyle(0x5d5850, 1).fillRect(x, y, T, T);
    const rows = [[1, 9], [11, 9], [21, 10]];
    rows.forEach(([ry, rh], i) => {
      let cx = x + (i % 2 ? -6 : 0);
      while (cx < x + T) {
        const sw = 9 + Math.floor(r() * 9);
        const sx = Math.max(x, cx + 1);
        const ex = Math.min(x + T, cx + sw - 1);
        if (ex - sx > 2) {
          const c = STONES[Math.floor(r() * STONES.length)];
          g.fillStyle(c, 1).fillRoundedRect(sx, y + ry, ex - sx, rh - 1, 2);
          g.fillStyle(shade(c, 1.12), 1).fillRect(sx + 1, y + ry, ex - sx - 2, 2);
          g.fillStyle(shade(c, 0.82), 1).fillRect(sx + 1, y + ry + rh - 3, ex - sx - 2, 2);
        }
        cx += sw;
      }
    });
    // walkway cap and crenels on top
    g.fillStyle(0xb2ada2, 1).fillRect(x, y - 3, T, 4);
    if ((o.x + o.y) % 2 === 0) {
      g.fillStyle(0x9a958b, 1).fillRect(x + 4, y - 10, T - 8, 8);
      g.fillStyle(0xb8b3a8, 1).fillRect(x + 4, y - 10, T - 8, 2);
    }
    // a little moss
    if (r() < 0.35) g.fillStyle(0x6f7d4c, 0.8).fillEllipse(x + r() * T, y + T - 3, 8, 4);
  }

  // A big rice paddy: earth dike, water with sky reflections, rows of rice (some fields ripe and golden).
  function paddy(g, x, y, w, h) {
    const r = rnd(x * 7 + y * 13);
    const ripe = r() < 0.35;
    g.fillStyle(0x1a160f, 0.15).fillRect(x + 2, y + h, w, 5);
    g.fillStyle(0x86744f, 1).fillRoundedRect(x, y, w, h, 6);

    g.fillStyle(0x96aea9, 1).fillRect(x + 8, y + 9, w - 16, 5);

    const GREEN = [0x5f8a3c, 0x72a04a, 0x679444];
    const GOLD = [0xc9a84d, 0xd8bb62, 0xb89640];
    for (let yy = y + 20; yy < y + h - 8; yy += 15)
      for (let xx = x + 14 + (Math.round((yy - y) / 15) % 2) * 6; xx < x + w - 12; xx += 13) {
        const c = ripe ? pick2(r, GOLD) : pick2(r, GREEN);

        g.lineStyle(1.8, shade(c, 0.78), 1).lineBetween(xx, yy, xx - 5, yy - 9).lineBetween(xx, yy, xx + 5, yy - 9);
        g.lineStyle(1.8, c, 1).lineBetween(xx, yy, xx - 2, yy - 12).lineBetween(xx, yy, xx + 2, yy - 12).lineBetween(xx, yy, xx, yy - 13);

        if (ripe) {

  function foundation(g, x, y, w) {

  function chimney(g, cx, top) {

  function building(g, b) {
    const x = b.x * T;
    const y = b.y * T;
    const w = b.w * T;
    const h = b.h * T;
    shadow(g, x + w / 2 + 8, y + h + 2, w + 20, 18, 0.26);
    if (b.kind === 'hall') return hall(g, b, x, y, w, h);
    if (b.kind === 'barn') return barn(g, b, x, y, w, h);
    if (b.kind === 'wood') return woodHouse(g, b, x, y, w, h);
    return thatchHouse(g, b, x, y, w, h);
  }

  // Straw-roofed cottage: thick layered thatch, daub walls with a timber frame, stone footing.
  function thatchHouse(g, b, x, y, w, h) {
    const wallTop = y + h * 0.42;
    const by = y + h;
    // walls
    g.fillStyle(0xddd1b4, 1).fillRect(x + 2, wallTop, w - 4, by - wallTop);
    g.fillStyle(0xcfc2a2, 1).fillRect(x + 2, by - 16, w - 4, 10);
    // timber frame: posts, beam, braces
    g.fillStyle(0x5a4429, 1);
    [x + 2, x + w / 2 - 2, x + w - 6].forEach((px) => g.fillRect(px, wallTop, 4, by - wallTop));
    g.fillRect(x + 2, wallTop + 4, w - 4, 3);
    g.lineStyle(3, 0x5a4429, 1).lineBetween(x + 6, by - 8, x + w / 2 - 2, wallTop + 8).lineBetween(x + w - 6, by - 8, x + w / 2 + 2, wallTop + 8);
    foundation(g, x + 2, by, w - 4);
    // window on the side away from the door
    const dx = b.door * T + 16;
    const winX = dx < x + w / 2 ? x + w - 34 : x + 18;
    windowBox(g, winX, wallTop + 14, 14, 12, true);
    g.fillStyle(0x6f7d4c, 1).fillRect(winX - 2, wallTop + 29, 18, 4);
    g.fillStyle(0xd9cfa0, 1).fillCircle(winX + 2, wallTop + 29, 2).fillCircle(winX + 9, wallTop + 28, 2);
    door(g, dx, by - 2, 18, 26);
    // thatch roof: rounded mass, layered bands with ragged edges, bound ridge
    const ry = y - 10;
    const rh = h * 0.56 + 10;
    g.fillStyle(0x1a160f, 0.25).fillRoundedRect(x - 8, wallTop - 2, w + 16, 10, 5);
    g.fillStyle(0x9a8044, 1).fillRoundedRect(x - 10, ry, w + 20, rh, 18);
    const r = rnd(x * 3 + y);
    const bands = 6;
    for (let i = bands - 1; i >= 0; i--) {
      const yy = ry + 6 + (i * (rh - 10)) / bands;
      const c = i % 2 ? 0xc8ae6c : 0xb79c58;
      g.fillStyle(c, 1).fillRoundedRect(x - 9 + i * 0.5, yy, w + 18 - i, (rh - 10) / bands + 4, 8);
      for (let xx = x - 8; xx < x + w + 8; xx += 5) {
        const len = 4 + r() * 5;
        g.fillStyle(shade(c, 0.8), 1).fillRect(xx, yy + (rh - 10) / bands + 1, 2, len - 2);
        g.lineStyle(1, shade(c, 1.12), 0.9).lineBetween(xx + 1, yy + 2, xx + 1, yy + 6);
      }
    }
    g.fillStyle(0x8a7038, 1).fillRoundedRect(x - 4, ry - 4, w + 8, 10, 5);
    for (let xx = x; xx < x + w; xx += 12) g.fillStyle(0x6b5a30, 1).fillRect(xx, ry - 4, 2, 10);
    g.fillStyle(0xd8c07e, 0.6).fillRoundedRect(x - 2, ry - 3, w + 4, 3, 2);
    chimney(g, x + w - 22, ry - 12);
  }

  // Plank house: vertical boards, a shingle roof with a lit ridge, a little porch roof over the door.
  function woodHouse(g, b, x, y, w, h) {
    const wallTop = y + h * 0.44;
    const by = y + h;
    const PL = [0x7f6142, 0x8a6a48, 0x76593c];
    for (let xx = x + 2; xx < x + w - 2; xx += 7) {
      g.fillStyle(PL[Math.floor(((xx - x) / 7) % 3)], 1).fillRect(xx, wallTop, 7, by - wallTop);
      g.fillStyle(0x4a3826, 1).fillRect(xx + 6, wallTop, 1, by - wallTop);
      g.fillStyle(0x9a7a52, 0.6).fillRect(xx + 1, wallTop, 1, by - wallTop);
    }
    g.fillStyle(0x5a4429, 1).fillRect(x + 2, wallTop, 4, by - wallTop).fillRect(x + w - 6, wallTop, 4, by - wallTop);
    g.fillStyle(0x5a4429, 1).fillRect(x + 2, wallTop + 2, w - 4, 3);
    foundation(g, x + 2, by, w - 4);
    const dx = b.door * T + 16;
    const winX = dx < x + w / 2 ? x + w - 34 : x + 18;
    windowBox(g, winX, wallTop + 13, 14, 12, true);
    door(g, dx, by - 2, 18, 26);
    // porch roof over the door
    poly(g, 0x5d4a3a, 1, [[dx - 18, by - 30], [dx + 18, by - 30], [dx + 14, by - 38], [dx - 14, by - 38]]);
    g.fillStyle(0x4a3826, 1).fillRect(dx - 17, by - 30, 3, 28).fillRect(dx + 14, by - 30, 3, 28);
    // shingle roof
    const ry = y - 8;
    const rh = h * 0.56 + 8;
    g.fillStyle(0x1a160f, 0.25).fillRect(x - 6, wallTop - 2, w + 12, 8);
    g.fillStyle(shade(b.color, 0.8), 1).fillRect(x - 8, ry, w + 16, rh);
    const rows = 7;
    for (let i = 0; i < rows; i++) {
      const yy = ry + 4 + (i * (rh - 6)) / rows;
      const off = i % 2 ? 6 : 0;
      for (let xx = x - 8 - off; xx < x + w + 8; xx += 12) {
        const sx = Math.max(x - 8, xx);
        const ex = Math.min(x + w + 8, xx + 11);
        if (ex <= sx) continue;
        const c = shade(b.color, 0.92 + ((xx * 7 + i * 3) % 5) * 0.05);
        g.fillStyle(c, 1).fillRect(sx, yy, ex - sx, (rh - 6) / rows + 2);
        g.fillStyle(shade(c, 0.7), 1).fillRect(sx, yy + (rh - 6) / rows, ex - sx, 2);
      }
    }
    g.fillStyle(shade(b.color, 0.6), 1).fillRect(x - 8, ry, 3, rh).fillRect(x + w + 5, ry, 3, rh);
    g.fillStyle(shade(b.color, 1.3), 1).fillRect(x - 6, ry - 4, w + 12, 6);
    g.fillStyle(shade(b.color, 0.75), 1).fillRect(x - 6, ry + 1, w + 12, 2);
    chimney(g, x + 20, ry - 10);
  }

  // Big stone hall with an arched door, tall windows, buttresses, a slate roof and a square tower with a banner.
  function hall(g, b, x, y, w, h) {
    const wallTop = y + h * 0.4;
    const by = y + h;
    const r = rnd(4242);
    g.fillStyle(0x8f897d, 1).fillRect(x, wallTop, w, by - wallTop);
    for (let yy = wallTop; yy < by - 4; yy += 9) {
      const off = ((yy - wallTop) / 9) % 2 ? 10 : 0;
      for (let xx = x - off; xx < x + w; xx += 20) {
        const sx = Math.max(x, xx + 1);
        const ex = Math.min(x + w, xx + 19);
        const c = pick2(r, [0xb2ab9e, 0xa9a295, 0xbab4a8]);
        g.fillStyle(c, 1).fillRect(sx, yy + 1, ex - sx, 7);
        g.fillStyle(shade(c, 1.1), 1).fillRect(sx, yy + 1, ex - sx, 1);
      }
    }
    // buttresses
    [x - 4, x + w - 8, x + w / 2 - 50, x + w / 2 + 38].forEach((bx) => {
      g.fillStyle(0x8f897d, 1).fillRect(bx, wallTop + 6, 12, by - wallTop - 6);
      g.fillStyle(0xc2bcb0, 1).fillRect(bx, wallTop + 6, 12, 3);
    });
    // tall arched windows
    [x + 26, x + w - 46].forEach((wx) => {
      g.fillStyle(0x6f6a62, 1).fillRoundedRect(wx - 2, wallTop + 10, 20, 36, { tl: 10, tr: 10, bl: 0, br: 0 });
      g.fillStyle(0x2b3236, 1).fillRoundedRect(wx, wallTop + 12, 16, 32, { tl: 8, tr: 8, bl: 0, br: 0 });

    });
    // arched double door with studs, steps
    const dx = b.door * T + 16;

    g.fillStyle(0x9a958b, 1).fillRect(dx - 26, by, 52, 4).fillRect(dx - 22, by + 4, 44, 4);

    g.fillStyle(0x4c535b, 1).fillRect(x - 8, ry, w + 16, rh);
    for (let i = 0; i < 6; i++) {
      const yy = ry + 2 + i * (rh / 6);
      for (let xx = x - 8 - (i % 2) * 5; xx < x + w + 8; xx += 10) {

        const ex = Math.min(x + w + 8, xx + 9);
        if (ex <= sx) continue;
        g.fillStyle(pick2(r, [0x5a626b, 0x535b64, 0x626a73]), 1).fillRect(sx, yy, ex - sx, rh / 6);

      }
    }
    g.fillStyle(0x7a828a, 1).fillRect(x - 6, ry - 3, w + 12, 4);

    const tx = x + w / 2 - tw / 2;

    g.fillStyle(0x9a9386, 1).fillRect(tx, tt, tw, ry - tt + 10);
    for (let yy = tt + 4; yy < ry + 8; yy += 9) g.fillStyle(0x8a8377, 1).fillRect(tx, yy, tw, 1);

    poly(g, 0x7d2e2e, 1, [[tx + tw / 2 + 2, tt - 40], [tx + tw / 2 + 30, tt - 36], [tx + tw / 2 + 24, tt - 30], [tx + tw / 2 + 30, tt - 24], [tx + tw / 2 + 2, tt - 26]]);
    crest(g, tx + tw / 2 + 13, tt - 33, 0.9);

    [x + 8, x + w - 20].forEach((bx) => {
      poly(g, 0x7d2e2e, 1, [[bx, wallTop + 4], [bx + 12, wallTop + 4], [bx + 12, wallTop + 34], [bx + 6, wallTop + 29], [bx, wallTop + 34]]);

    });
  }

  // Wooden granary: planks, a big X-braced door, a hay loft and a gambrel-style shingle roof.

    g.fillStyle(0xcdb56c, 1).fillEllipse(dx, wallTop + 16, 18, 8);

    const ry = y - 10;
    const rh = h * 0.4 + 10;
    g.fillStyle(0x1a160f, 0.25).fillRect(x - 6, wallTop - 2, w + 12, 7);
    poly(g, 0x5d4a3a, 1, [[x - 10, wallTop], [x + w + 10, wallTop], [x + w + 2, ry + rh * 0.4], [x + w - 20, ry], [x + 20, ry], [x - 2, ry + rh * 0.4]]);

  // Each tall thing is baked once into its own texture (translateCanvas moves it to 0,0), then shown as an image.

  const PLAYER_DEPTH = 10;
  function tallList(a) {
    const out = [];
    a.obstacles.forEach((o, i) => {
      if (!TALL.has(o.kind)) return;
      const x = o.x * T;
      const y = o.y * T;
      const w = o.w * T;
      const h = o.h * T;
      const pad = {
        tower: [16, 96, 16, 16],
        tree: [12, 30, 12, 14],
        well: [14, 40, 14, 22],
        banner: [4, 32, 14, 8],
        torch: [10, 24, 10, 8],
        stall: [10, 24, 10, 10],
        coop: [8, 4, 30, 18],
        rack: [6, 4, 6, 12],
      }[o.kind];
      out.push({ key: `past_o${i}_${o.kind}`, x0: x - pad[0], y0: y - pad[1], x1: x + w + pad[2], y1: y + h + pad[3], base: (o.y + o.h) * T, draw: (g) => drawThing(g, o, a) });
    });
    a.buildings.forEach((b, i) => {

      out.push({ key: `past_b${i}`, x0: x - 36, y0: y - 104, x1: x + b.w * T + 36, y1: y + b.h * T + 24, base: (b.y + b.h) * T, draw: (g) => building(g, b) });
    });

  function buildTall(sc) {
    const items = tallList(sc.area).map((t) => {
      const w = Math.ceil(t.x1 - t.x0);
      const h = Math.ceil(t.y1 - t.y0);
      if (!sc.textures.exists(t.key)) {
        const g = sc.add.graphics();
        g.translateCanvas(-t.x0, -t.y0);
        t.draw(g);
        g.generateTexture(t.key, w, h);
        g.destroy();
      }
      return { img: sc.add.image(t.x0, t.y0, t.key).setOrigin(0).setDepth(PLAYER_DEPTH - 0.6), base: t.base };
    });
    const st = { area: sc.area, items };
    st.update = () => {
      if (sc.area !== st.area) return clearTall(sc);
      if (!sc.player) return;
      const hide = !!sc.inside;
      const py = sc.player.y;
      for (const it of items) {
        it.img.setVisible(!hide);
        it.img.setDepth(py < it.base ? PLAYER_DEPTH + 0.6 : PLAYER_DEPTH - 0.6);
      }
    };
    sc.events.on('update', st.update);
    sc.__lqPastTall = st;
    st.update();
  }
  function clearTall(sc) {
    const st = sc && sc.__lqPastTall;
    if (!st) return;
    sc.events.off('update', st.update);
    st.items.forEach((it) => it.img.destroy());
    sc.__lqPastTall = null;
  }
  function syncTall(sc) {
    const st = sc.__lqPastTall;
    if (here(sc) && !sc.switching) {
      if (st && st.area !== sc.area) clearTall(sc);
      if (!sc.__lqPastTall) {
        try {
          buildTall(sc);
        } catch (e) {
          console.warn('past tall sprites failed', e);
          sc.__lqPastTall = { area: sc.area, items: [], update() {} };
        }
      }
    } else if (st) clearTall(sc);
  }

  function makeFlock(a) {
    const r = rnd(2024);
    const birds = [];
    const free = (x, y) => !a.isTileBlocked(Math.floor(x / T), Math.floor(y / T));
    const spot = (f) => {
      for (let i = 0; i < 40; i++) {
        const ang = r() * Math.PI * 2;
        const d = Math.sqrt(r()) * f.r * T;
        const x = (f.cx + 0.5) * T + Math.cos(ang) * d;
        const y = (f.cy + 0.5) * T + Math.sin(ang) * d;
        if (free(x, y)) return { x, y };
      }
      return { x: (f.cx + 0.5) * T, y: (f.cy + 0.5) * T };
    };
    FLOCKS.forEach((f) => {
      const add = (kind) => {
        const p = spot(f);
        birds.push({ kind, f, x: p.x, y: p.y, tx: p.x, ty: p.y, state: 'idle', t: r() * 2, face: r() < 0.5 ? -1 : 1, phase: r() * 10, flap: 0, mom: null });
      };
      for (let i = 0; i < f.hens; i++) add('hen');
      if (f.rooster) add('rooster');
      const moms = birds.filter((b) => b.f === f && b.kind === 'hen');
      for (let i = 0; i < (f.chicks || 0); i++) {
        add('chick');
        birds[birds.length - 1].mom = moms[i % moms.length] || null;
      }
    });
    return { birds, free, spot, r };
  }
  function henTick(fl, sc, dt) {
    const who = scarers(sc);
    for (const b of fl.birds) {
      b.t -= dt;
      b.phase += dt * (b.state === 'walk' || b.state === 'run' ? 14 : 3);
      b.flap = Math.max(0, b.flap - dt);
      // The nearest one who scares it: the player, or a running child.
      let px = who[0].x;
      let py = who[0].y;
      for (const s of who) {
        if (Math.hypot(b.x - s.x, b.y - s.y) < Math.hypot(b.x - px, b.y - py)) {
          px = s.x;
          py = s.y;
        }
      }
      const dp = Math.hypot(b.x - px, b.y - py);
      // Scared: run straight away from the player, flapping.
      if (dp < SCARE && b.state !== 'run') {
        const k = 70 / Math.max(1, dp);
        b.tx = b.x + (b.x - px) * k;
        b.ty = b.y + (b.y - py) * k;
        b.state = 'run';
        b.flap = 0.6;
        b.t = 1.2;
      }
      if (b.state === 'walk' || b.state === 'run') {
        const sp = (b.state === 'run' ? RUN_SPEED : HEN_SPEED) * (b.kind === 'chick' ? 1.15 : 1) * dt;
        const dx = b.tx - b.x;
        const dy = b.ty - b.y;
        const d = Math.hypot(dx, dy);
        if (Math.abs(dx) > 0.5) b.face = dx < 0 ? -1 : 1;
        if (d <= sp || b.t <= 0) {
          b.state = fl.r() < 0.6 ? 'peck' : 'idle';
          b.t = 0.8 + fl.r() * 2.2;
        } else {
          const nx = b.x + (dx / d) * sp;
          const ny = b.y + (dy / d) * sp;
          if (fl.free(nx, ny)) {
            b.x = nx;
            b.y = ny;
          } else {
            b.state = 'idle';
            b.t = 0.3;
          }
        }
      } else if (b.t <= 0) {
        // Pick the next place: chicks follow their mother, the others wander around home.
        let p;
        if (b.kind === 'chick' && b.mom) p = { x: b.mom.x + (fl.r() - 0.5) * 30, y: b.mom.y + 6 + (fl.r() - 0.5) * 16 };
        else p = fl.spot(b.f);
        b.tx = p.x;
        b.ty = p.y;
        b.state = 'walk';
        b.t = 6;
        if (b.kind === 'rooster' && fl.r() < 0.15) b.flap = 0.8;
      }
    }
  }
  function drawBird(g, b) {
    const f = b.face;
    const x = b.x;
    const walking = b.state === 'walk' || b.state === 'run';
    const bob = walking ? Math.abs(Math.sin(b.phase)) * 1.6 : 0;
    const y = b.y - bob;
    if (b.kind === 'chick') {
      g.fillStyle(0x1a160f, 0.2).fillEllipse(b.x, b.y + 4, 8, 3);
      g.fillStyle(0xf1d36a, 1).fillCircle(x, y, 3.6);
      g.fillStyle(0xf6df86, 1).fillCircle(x + f * 2.5, y - 3, 2.6);
      g.fillStyle(0xd9822e, 1).fillTriangle(x + f * 4.5, y - 3.5, x + f * 6.5, y - 3, x + f * 4.5, y - 2.3);
      g.fillStyle(0x2a241d, 1).fillRect(x + f * 3 - 0.5, y - 4, 1.2, 1.2);
      return;
    }
    const big = b.kind === 'rooster' ? 1.18 : 1;
    const body = b.kind === 'rooster' ? 0xc0743c : 0xefe9dc;
    const wing = b.kind === 'rooster' ? 0x9a5a2e : 0xd9d2c2;
    const pecking = b.state === 'peck' && Math.sin(b.phase * 2.2) > 0.2;
    g.fillStyle(0x1a160f, 0.2).fillEllipse(b.x, b.y + 6 * big, 14 * big, 4);
    // legs (step when walking)
    const step = walking ? Math.sin(b.phase) * 2 : 0;
    g.lineStyle(1.3, 0xd9a13a, 1).lineBetween(x - 1.5, y + 2, x - 1.5 + step, y + 6 * big).lineBetween(x + 1.5, y + 2, x + 1.5 - step, y + 6 * big);
    // tail
    if (b.kind === 'rooster') {
      g.fillStyle(0x2f4a3a, 1).fillTriangle(x - f * 4, y - 1, x - f * 11, y - 11, x - f * 9, y + 1);
      g.fillStyle(0x23382c, 1).fillTriangle(x - f * 5, y, x - f * 12, y - 6, x - f * 8, y + 3);
    } else g.fillStyle(0xe2dccd, 1).fillTriangle(x - f * 4, y - 1, x - f * 9, y - 7, x - f * 8, y + 1);
    // body + wing (wing lifts when flapping)
    g.fillStyle(body, 1).fillEllipse(x, y, 13 * big, 9 * big);
    if (b.flap > 0) {
      const up = Math.sin(b.flap * 40) * 4;
      g.fillStyle(wing, 1).fillTriangle(x - f * 2, y - 2, x - f * 8, y - 7 - up, x + f * 3, y - 3);
    } else g.fillStyle(wing, 1).fillEllipse(x - f * 1, y + 0.5, 7 * big, 5 * big);
    // head: down to the ground when pecking
    const hx = x + f * (pecking ? 7 : 5) * big;
    const hy = y - (pecking ? -1 : 5) * big;
    g.fillStyle(body === 0xefe9dc ? 0xf7f3ea : 0xd08446, 1).fillCircle(hx, hy, 3.4 * big);
    g.fillStyle(0xc8432f, 1).fillRect(hx - 1.5, hy - 4.6 * big, 3 * big, 2 * big);
    if (b.kind === 'rooster') g.fillStyle(0xc8432f, 1).fillCircle(hx + f * 1, hy - 4.5 * big, 1.6).fillCircle(hx - f * 1, hy - 5 * big, 1.6).fillCircle(hx + f * 2.5, hy + 2.5, 1.3);
    g.fillStyle(0xd9a13a, 1).fillTriangle(hx + f * 3, hy - 0.8, hx + f * 6, hy, hx + f * 3, hy + 1);
    g.fillStyle(0x2a241d, 1).fillRect(hx + f * 1.2 - 0.6, hy - 1.4, 1.3, 1.3);
  }

  function faceTo(p, dx, dy) {
    const ax = Math.abs(dx);
    const ay = Math.abs(dy);
    if (ax < 0.01 && ay < 0.01) return;
    if (ay < ax * 0.4) {
      p.dir = 'side';
      p.flip = dx < 0;
    } else if (ax < ay * 0.4) p.dir = dy < 0 ? 'up' : 'down';
    else {
      p.dir = dy < 0 ? 'uside' : 'dside';
      p.flip = dx < 0;
    }
  }
  function folkSpot(st, home) {
    for (let k = 0; k < 30; k++) {
      const ang = st.r() * Math.PI * 2;
      const d = Math.sqrt(st.r()) * home[2] * T;
      const x = home[0] * T + Math.cos(ang) * d;
      const y = home[1] * T + Math.sin(ang) * d;
      if (st.free(x, y)) return { x, y };
    }
    return { x: home[0] * T, y: home[1] * T };
  }
  // Move p toward (tx, ty). Returns true when it got there (or is stuck).
  function folkStep(st, p, sp, px, py) {
    const dx = p.tx - p.x;
    const dy = p.ty - p.y;
    const d = Math.hypot(dx, dy);
    if (d <= sp) {
      p.x = p.tx;
      p.y = p.ty;
      return true;
    }
    const nx = p.x + (dx / d) * sp;
    const ny = p.y + (dy / d) * sp;
    faceTo(p, dx, dy);
    if (Math.hypot(nx - px, ny - py) < 14) return false; // never walk through the player: wait
    if (!st.free(nx, ny)) return true;
    p.x = nx;
    p.y = ny;
    p.walking = true;
    return false;
  }
  // ---- the working day: children worked too in those days ---------------------------------------------------------------
  // Village clock (game minutes): 05:30–11:30 and 13:00–17:30 planting rice in the south paddies, back in the village at
  // noon and in the evening, and from 19:30 home: they walk to their own door and go inside (hidden) until morning.
  // They go out through the south gate and along the dike, so the routes never cross the wall.
  const HUB = [18.5, 30.5]; // just inside the south gate
  const DOOR_Y = 28.8; // in front of the cottage doors
  function workMode(min) {
    const m = ((min % 1440) + 1440) % 1440;
    if (m < 330 || m >= 1170) return 'home';
    if (m < 690 || (m >= 780 && m < 1050)) return 'field';
    return 'village';
  }
  function workBase(v) {
    if (v.home) return v.home;
    if (v.path) return v.path[0];
    return FOLK[v.follow].home;
  }
  function workRoute(p, from, to) {
    const [door, sx, sy] = p.v.work;
    const base = workBase(p.v);
    const low = (y) => y > 26.5; // south of the cottages: go along the lane behind them
    const out = [];
    if (from === 'field') out.push([sx, 41.5], [18.5, 41.5], [18.5, 33], HUB);
    else if (from === 'home') out.push([door + 0.5, 30.5], HUB);
    else out.push(low(p.y / T) ? [p.x / T, 30.5] : [18.5, 20.5], HUB);
    if (to === 'field') out.push([18.5, 33], [18.5, 41.5], [sx, 41.5], [sx, sy]);
    else if (to === 'home') out.push([door + 0.5, 30.5], [door + 0.5, DOOR_Y]);
    else out.push(low(base[1]) ? [base[0], 30.5] : [18.5, 20.5], [base[0], base[1]]);
    return out;
  }
  // Like folkStep, but on a planned route (gates, dikes): no wall check. Still never walks through the player.
  function workStep(p, sp, px, py) {
    const dx = p.tx - p.x;
    const dy = p.ty - p.y;
    const d = Math.hypot(dx, dy);
    if (d <= sp) {
      p.x = p.tx;
      p.y = p.ty;
      return true;
    }
    faceTo(p, dx, dy);
    const nx = p.x + (dx / d) * sp;
    const ny = p.y + (dy / d) * sp;
    if (Math.hypot(nx - px, ny - py) < 14) return false;
    p.x = nx;
    p.y = ny;
    p.walking = true;
    return false;
  }
  // Returns true when the schedule moved this villager (their usual village life is skipped this frame).
  function workTick(sc, p, dt, px, py) {
    const v = p.v;
    if (!v.work) return false;
    const g = sc.store && sc.store.game;
    const want = workMode(g && typeof g.minutes === 'number' ? g.minutes : 480);
    const [door, sx, sy] = v.work;
    if (!p.mode) {
      // First frame in the village: put them where the clock says, without walking there.
      p.mode = want;
      if (want === 'field') {
        p.x = p.tx = sx * T;
        p.y = p.ty = sy * T;
      } else if (want === 'home') {
        p.x = p.tx = (door + 0.5) * T;
        p.y = p.ty = DOOR_Y * T;
        p.gone = true;
      }
    }
    if (!p.route && p.mode !== want) {
      p.route = workRoute(p, p.mode, want);
      p.goal = want;
      p.gone = false; // morning: they come out of their door
      p.chaser = false;
    }
    p.away = !!p.route || p.mode !== 'village';
    if (p.route) {
      const pt = p.route[0];
      p.tx = pt[0] * T;
      p.ty = pt[1] * T;
      if (workStep(p, (WALK[v.kind] || 32) * dt, px, py)) {
        p.route.shift();
        if (!p.route.length) {
          p.route = null;
          p.mode = p.goal;
          if (p.mode === 'home') p.gone = true;
          if (p.mode === 'village') {
            p.away = false;
            p.leg = 0;
            p.t = 0;
          }
        }
      }
      return true;
    }
    if (p.mode === 'home') return true;
    if (p.mode === 'field') {
      // Planting: step slowly along the row, bending down again and again to push the seedlings in.
      p.plant = (p.plant || 0) + dt;
      p.x = sx * T + Math.sin(p.plant * 0.25 + p.i) * 0.9 * T;
      p.y = sy * T;
      p.dir = 'down';
      p.planting = true;
      return true;
    }
    return false;
  }

  function folkTick(st, sc, dt) {
    const px = sc.player.x;
    const py = sc.player.y;
    const folk = st.folk;
    const taggers = folk.filter((p) => p.v.play === 'tag' && !p.away);
    if (taggers.length && !taggers.some((p) => p.chaser)) taggers[0].chaser = true;
    for (const p of folk) {
      const v = p.v;
      p.walking = false;
      p.planting = false;
      if (v.after) {
        // Only home after the Day of Glory (the young soldier).
        const d = data();
        p.gone = !(d && d.done);
        if (p.gone) continue;
      }
      p.t -= dt;
      p.freeze = Math.max(0, p.freeze - dt);
      if (p.gone && p.mode === 'home' && workMode(sc.store.game.minutes) === 'home') continue;
      const dp = Math.hypot(p.x - px, p.y - py);
      // Grown-ups stop and look at the player who comes close; children keep playing.
      if (!isKid(v.kind) && dp < 48) {
        faceTo(p, px - p.x, py - p.y);
        continue;
      }
      if (workTick(sc, p, dt, px, py)) continue;
      if (v.play === 'tag') {
        const home = v.home;
        const sp = TAG_RUN * dt;
        if (p.freeze > 0) continue;
        if (p.chaser) {
          // Run after the nearest other child; touching them makes them "it".
          let best = null;
          let bd = 1e9;
          for (const o of taggers) {
            if (o === p || o.freeze > 0) continue;
            const d = Math.hypot(o.x - p.x, o.y - p.y);
            if (d < bd) {
              bd = d;
              best = o;
            }
          }
          if (!best) continue;
          if (bd < 11) {
            p.chaser = false;
            best.chaser = true;
            best.freeze = 1.1; // the new "it" counts a little before running
            p.freeze = 0.4;
            continue;
          }
          p.tx = best.x;
          p.ty = best.y;
          folkStep(st, p, sp * 0.92, px, py);
        } else {
          const ch = taggers.find((o) => o.chaser);
          const dc = ch ? Math.hypot(ch.x - p.x, ch.y - p.y) : 1e9;
          if (ch && dc < 56 && ch.freeze <= 0) {
            // Run away from "it", but stay in the square.
            let tx = p.x + ((p.x - ch.x) / Math.max(1, dc)) * 40;
            let ty = p.y + ((p.y - ch.y) / Math.max(1, dc)) * 40;
            if (Math.hypot(tx - home[0] * T, ty - home[1] * T) > home[2] * T || !st.free(tx, ty)) {
              const s = folkSpot(st, home);
              tx = s.x;
              ty = s.y;
            }
            if (p.t <= 0 || Math.hypot(p.tx - p.x, p.ty - p.y) < 4) {
              p.tx = tx;
              p.ty = ty;
              p.t = 0.6;
            }
            folkStep(st, p, sp, px, py);
          } else {
            if (p.t <= 0 || Math.hypot(p.tx - p.x, p.ty - p.y) < 3) {
              const s = folkSpot(st, home);
              p.tx = s.x;
              p.ty = s.y;
              p.t = 1 + st.r() * 1.5;
            }
            folkStep(st, p, sp * 0.55, px, py);
          }
        }
        continue;
      }
      const sp = (WALK[v.kind] || 32) * dt;
      if (v.follow != null) {
        // A little one next to her mother: a step behind and to the side.
        const m = folk[v.follow];
        p.tx = m.x + 10;
        p.ty = m.y + 6;
        if (Math.hypot(p.tx - p.x, p.ty - p.y) > 6) folkStep(st, p, sp * 1.3, px, py);
        else faceTo(p, m.x - p.x, m.y - p.y);
        continue;
      }
      if (v.path) {
        if (p.leg == null) p.leg = 0;
        if (p.t > 0) continue; // a short pause at each point
        const pt = v.path[p.leg];
        p.tx = pt[0] * T;
        p.ty = pt[1] * T;
        if (folkStep(st, p, sp, px, py)) {
          p.leg = (p.leg + 1) % v.path.length;
          p.t = 0.6 + st.r() * 1.8;
        }
        continue;
      }
      if (v.still) {
        // Stands at the spot (comes back to it if it was pushed away) and looks the same way.
        p.tx = v.home[0] * T;
        p.ty = v.home[1] * T;
        if (Math.hypot(p.tx - p.x, p.ty - p.y) > 2) folkStep(st, p, sp, px, py);
        else p.dir = v.still;
        continue;
      }
      // Wander around home, with pauses.
      if (p.t > 0 && Math.hypot(p.tx - p.x, p.ty - p.y) < 1) continue;
      if (p.t <= 0 && Math.hypot(p.tx - p.x, p.ty - p.y) < 1) {
        const s = folkSpot(st, v.home);
        p.tx = s.x;
        p.ty = s.y;
        p.t = 8;
      }
      if (folkStep(st, p, sp, px, py)) p.t = 1.5 + st.r() * 4;
    }
  }

  function folkUpdate(sc, delta) {
    let st = sc.__lqPastFolk;
    const kill = () => {
      if (!st) return;
      st.folk.forEach((p) => {
        p.img.destroy();
        if (p.bubble) p.bubble.destroy();
      });
      sc.__lqPastFolk = null;
    };
    if (!here(sc) || sc.switching || !sc.player) return kill();
    if (!st || st.area !== sc.area) {
      kill();
      const a = sc.area;
      st = sc.__lqPastFolk = { area: a, folk: [], r: rnd(29), free: (x, y) => !a.isTileBlocked(Math.floor(x / T), Math.floor(y / T)) };
      try {
        st.folk = makeFolk(sc);
      } catch (e) {
        console.warn('past villagers failed', e);
      }
    }
    const hide = !!sc.inside || filming;
    const dt = Math.min(0.05, (delta || 16) / 1000);
    if (!hide) folkTick(st, sc, dt);
    for (const p of st.folk) {
      p.img.setVisible(!hide && !p.gone);
      if (hide || p.gone) {
        if (p.bubble) p.bubble.setVisible(false);
        continue;
      }
      // Planting rice: bend down and up (origin is at the feet, so they stay on the ground).
      const h = HEIGHT[p.v.kind] || 1;
      p.img.setScale(h, p.planting ? h * (0.9 + 0.06 * Math.sin(p.plant * 2.4 + p.i)) : h);
      // Walk frames: 1 and 2 while moving (children step faster), 0 when standing.
      p.anim += dt * (p.walking ? (isKid(p.v.kind) ? 9 : 6) : 0);
      const frame = p.walking ? 1 + (Math.floor(p.anim) % 2) : 0;
      const tex = `ch_${p.key}_${p.dir}_${frame}`;
      if (p.img.texture.key !== tex) p.img.setTexture(tex);
      p.img.setFlipX(p.flip && p.dir !== 'down' && p.dir !== 'up');
      p.img.setPosition(p.x, p.y + FEET);
      p.img.setDepth(folkDepth(sc, p));
      // Talk once each time the player comes close (and stays a moment).
      const dp = Math.hypot(p.x - sc.player.x, p.y - sc.player.y);
      if (dp < 48) {
        p.near = (p.near || 0) + dt;
        if (!p.said && p.near > 0.5) {
          p.said = true;
          folkSay(sc, st, p);
        }
      } else if (dp > 70) {
        p.said = false;
        p.near = 0;
      }
      if (p.bubble) {
        p.bubbleT -= dt;
        if (p.bubbleT <= 0) {
          p.bubble.destroy();
          p.bubble = null;
        } else p.bubble.setVisible(true).setPosition(p.x, p.y + FEET - 38 * (HEIGHT[p.v.kind] || 1)).setAlpha(Math.min(1, p.bubbleT * 2));
      }
    }
  }

  // Inside a room the same works with NOTES (room tiles are scaled by ins.k, like fingers.js / tenshadows.js).
  let near = null;
  function nearestText(sc) {
    const ins = sc.inside;
    if (!ins) return SIGNS.find((s) => Math.hypot(sc.player.x - (s.x * 32 + 16), sc.player.y - (s.y * 32 + 16)) < READ_RADIUS);
    if (ins.floor !== 0 || !ins.objs || !ins.objs[0]) return null;
    const unit = 32 * ins.k;
    return NOTES.find((n) => {
      if (n.room !== ins.b.name) return false;
      const x = ins.objs[0].x + (n.x + 0.5) * unit;
      const y = ins.objs[0].y + (n.y + 0.5) * unit;
      return Math.hypot(sc.player.x - x, sc.player.y - y) < 1.9 * unit;
    });
  }

  }

  function runFilm(opts) {
    const sc = getScene();
    if (!sc || !here(sc) || filming) return Promise.resolve(false);
    filming = true;
    ensureCss();
    closeDayCard();
    return new Promise((resolve) => {
      const cam = sc.cameras.main;
      const zoom0 = cam.zoom;
      const kb = sc.input && sc.input.keyboard;
      if (kb) {
        kb.enabled = false;
        try {
          kb.resetKeys();
        } catch (e) {}
      }
      cam.stopFollow();
      const actors = sc.add.graphics().setDepth(9.6);
      const scene = opts.setup ? opts.setup(sc, actors) : {};
      const tick = (time) => {
        actors.clear();
        if (scene.tick) scene.tick(time);
      };
      sc.events.on('update', tick);
      const lines = opts.lines;
      const root = document.createElement('div');
      root.id = 'lq-past';
      root.innerHTML =
        '<div class="lb t"></div><div class="lb b"></div>' +
        `<div class="card"><small>${opts.card.small}</small><b>${opts.card.big}</b><span>${opts.card.sub}</span></div>` +
        '<div class="cap"><div class="who"></div><p class="en"></p><p class="vi"></p><div class="hint">Click / Space ▶</div></div>' +
        '<button class="skip" type="button">Skip ⏭</button>';
      document.body.appendChild(root);
      if (sc.game && sc.game.canvas) sc.game.canvas.classList.add('lq-main');
      document.body.classList.add('lq-filming');
      const card = root.querySelector('.card');
      const cap = root.querySelector('.cap');
      const who = root.querySelector('.who');
      const en = root.querySelector('.en');
      const vi = root.querySelector('.vi');
      let i = -1;
      let typing = 0;
      let full = '';
      let ended = false;
      const look = (shot, ms) => {
        if (shot.cam) cam.pan(shot.cam[0] * 32 + 16, shot.cam[1] * 32 + 16, ms, 'Sine.easeInOut', true);
        else cam.pan(sc.player.x, sc.player.y, ms, 'Sine.easeInOut', true);
        cam.zoomTo(zoom0 * (shot.zoom || 1), ms, 'Sine.easeInOut', true);
      };
      const show = () => {
        const s = lines[i];
        look(s, i === 0 ? 1 : 1400);
        if (s.act && scene.act) scene.act(s.act);
        who.textContent = s.who || '';
        who.style.display = s.who ? '' : 'none';
        vi.textContent = s.vi;
        full = s.en;
        en.textContent = '';
        clearInterval(typing);
        let n = 0;
        typing = setInterval(() => {
          n += 1;
          en.textContent = full.slice(0, n);
          if (n >= full.length) clearInterval(typing);
        }, 28);
      };
      const next = () => {
        if (ended) return;
        if (i >= 0 && en.textContent.length < full.length) {
          clearInterval(typing);
          en.textContent = full;
          return;
        }
        i += 1;
        if (i >= lines.length) return finish();
        show();
      };
      const finish = () => {
        if (ended) return;
        ended = true;
        clearInterval(typing);
        window.removeEventListener('keydown', onKey, true);
        // Skipping still runs every remaining action once (e.g. the colours come back), so the story state is right.
        if (scene.act) for (let k = i + 1; k < lines.length; k++) if (lines[k].act) scene.act(lines[k].act, true);
        root.classList.add('out');
        document.body.classList.remove('lq-filming');
        cap.classList.remove('show');
        card.classList.remove('show');
        sc.events.off('update', tick);
        actors.destroy();
        cam.zoomTo(zoom0, 600, 'Sine.easeInOut', true);
        cam.startFollow(sc.player, true, 0.1, 0.1);
        if (kb) kb.enabled = true;
        setTimeout(() => {
          root.remove();
          filming = false;
          resolve(true);
        }, 850);
      };
      const onKey = (e) => {
        if (e.key === ' ' || e.key === 'Enter') {
          e.preventDefault();
          next();
        } else if (e.key === 'Escape') {
          e.preventDefault();
          finish();
        }
      };
      window.addEventListener('keydown', onKey, true);
      cap.addEventListener('click', next);
      root.querySelector('.skip').addEventListener('click', finish);
      const first = lines[0].cam || [sc.player.x / 32, sc.player.y / 32];
      cam.centerOn(first[0] * 32 + 16, first[1] * 32 + 16);
      if (scene.act && lines[0].act) scene.act(lines[0].act);
      card.classList.add('show');
      setTimeout(() => {
        if (ended) return;
        card.classList.remove('show');
        cap.classList.add('show');
        next();
      }, 2600);
    });
  }
