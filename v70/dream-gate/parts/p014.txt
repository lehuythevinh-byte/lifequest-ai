      sc.__lqIntro = null;
    });
  }

  // ---- when to start ----------------------------------------------------------------------------------------------
  // A save counts as new when nothing has happened in it yet.
  function isNewSave(g) {
    const quests = Object.values(g.quests || {}).some((v) => Number(v) > 0);
    const ach = g.ach && g.ach.done && Object.keys(g.ach.done).length > 0;
    const msgs = (g.stats && g.stats.messages) || 0;
    return !quests && !ach && msgs === 0 && (g.day || 1) <= 1;
  }
  function readyToPlay(sc) {
    if (!sc || !sc.player || !sc.store || !sc.store.game || sc.switching) return false;
    if (['lq-title', 'lq-device', 'lq-auth', 'lq-cc', 'lq-train'].some(shown)) return false;

  }

    } catch (e) {}
  }

  function tick() {
    const sc = getScene();
    if (sc && sc.add && sc.events && sc.player && !sc.__lqIntro) {
      try {
        attach(sc);
      } catch (e) {

      }
    }
    // After "First Day in Town" is finished, the quest card moves on to The Ten Shadows.

    } catch (e) {}

      // Older saves with progress never see the intro; they can still replay it from the Settings.
      g.tut = isNewSave(g) ? { step: 0 } : { intro: 'old', done: 'old' };
      save();

        base = null;
        coachKey = '';
        save();
        selectFirstQuest();
      });
      return;
    }
    tutorialTick();
  }

    } catch (e) {
      console.warn('intro tick failed', e);
    }

    },
  };

  function startScene(cv) {

    const rnd = (a, b) => a + Math.random() * (b - a);

      H = Math.ceil(innerHeight / SCALE);
      cv.width = W;
      cv.height = H;
      stars = Array.from({ length: Math.floor((W * H) / 260) }, () => ({ x: Math.random() * W, y: Math.random() * H * 0.6, p: Math.random() * 6, s: Math.random() < 0.15 ? 2 : 1 }));
      const ground = H - 12;
      layers = [
        { col: '#101c33', win: '#27406b', min: 26, max: 62, gap: [0, 2], drift: 0 },
