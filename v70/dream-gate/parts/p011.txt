  function showIntro(areaId, meta, holdMs = 2850) {
    if (!INTRO_ENABLED_MAPS.has(areaId) || !meta) return;
    const now = Date.now();
    if (introLast.id === areaId && now - introLast.at < 1200) return;
    introLast = { id: areaId, at: now };
    const root = ensureIntroUi();
    root.querySelector('.icon').textContent = meta.icon || '🗺️';
    root.querySelector('.eyebrow').textContent = meta.eyebrow || 'WORLD';
    root.querySelector('.title').textContent = meta.title || areaId;
    root.querySelector('.subtitle').textContent = (CINEMA_V15[areaId]?.text || meta.subtitle || '');
    root.querySelector('.chapter').textContent = CINEMA_V15[areaId]?.name || '';
    root.querySelector('.shot').textContent = CINEMA_V15[areaId]?.shots?.[0] || '';
    root.querySelector('.progress span').style.width = '0%';
    root.style.setProperty('--v15-accent', CINEMA_V15[areaId]?.accent || '#f6cd86');
    root.classList.add('show');
    clearTimeout(introHideTimer);
    introHideTimer = setTimeout(() => root.classList.remove('show'), holdMs);
  }

  const cinematicSeen = new Map();
  function canPlayCinematic(scene, areaId) {
    if (!scene || scene.area?.id !== areaId || scene.switching || scene.inside || scene.sleeping) return false;
    if (!scene.player?.active || !scene.cameras?.main || !scene.sys?.isActive?.()) return false;
    if (scene.chat?.isOpen || scene.briefing?.isOpen || scene.settings?.isOpen || scene.quests?.isOpen) return false;
    // Story films, boot tutorial and the title screen own the camera whenever present.
    for(const id of ['lq-art','lq-past','lq-intro','lq-shrine']){const el=document.getElementById(id);if(el&&!el.hidden&&getComputedStyle(el).display!=='none'&&getComputedStyle(el).visibility!=='hidden'&&getComputedStyle(el).opacity!=='0')return false;}
    if(window.LQArtDistrict?.isFilming?.() || window.LQPast?.isFilming?.() || window.LQIntro?.isPlaying?.()) return false;
    const lobby=document.getElementById('lq-title');
    if(lobby && !lobby.hidden && getComputedStyle(lobby).display!=='none' && getComputedStyle(lobby).visibility!=='hidden') return false;
    return true;
  }
  function stopAreaCinematic(scene) {
    if(!scene)return;
    const active=scene.__lqAreaCine;
    if(!active)return;
    if(active===__originActive){originRestore(scene,active,false);return;}
    active.closed=true;
    clearTimeout(active.timer);
    if(active.keyHandler)window.removeEventListener('keydown',active.keyHandler,true);
    if(active.skipBtn)active.skipBtn.remove();
    scene.__lqAreaCine=null;
    try{scene.cameras.main.pan(scene.player.x,scene.player.y,150,'Sine.easeOut',true);
      scene.cameras.main.zoomTo(active.zoom0,150,'Sine.easeOut',true);
      scene.cameras.main.startFollow(scene.player,true,.12,.12);}catch(e){}
    if(active.keyboard)active.keyboard.enabled=active.keyboardWasOn;
    if(introRoot)introRoot.classList.remove('show');
  }
  function queueAreaCinematic(scene,p){
    if(!scene||!p)return;
    clearTimeout(scene.__lqAreaCinePending);
    // Seven non-ET maps enter gameplay immediately; no origin film and no V15 intro card.
    if(!INTRO_ENABLED_MAPS.has(scene.area?.id))return;
    const id=scene.area.id;
    let checked=false;
    const attempt=()=>{
      if(scene.area?.id!==id || scene.__lqWorldFx?.id!==id || !scene.sys?.isActive?.())return;
      if(scene.__lqAreaCine || !canPlayCinematic(scene,id)){
        // Don't consume first-discovery state during a story film, title screen, or modal.
        scene.__lqAreaCinePending=setTimeout(attempt,650);
        return;
      }
      if(originSeen(scene,id)){
        // Revisit: a short non-blocking location card; no long camera tour.
        showIntro(id,p.intro,900);
        ensureIntroUi().classList.add('revisit');
        return;
      }
      startOrigin(scene,id,false);
    };
    scene.__lqAreaCinePending=setTimeout(attempt,600);
  }
  function attach(scene) {
    const p = PLAN[scene.area?.id]; if (!p) return;
    const cache = scene.__lqWorldFx;
    if (cache) detach(scene);
    const layer = scene.add.graphics().setDepth(7);
    const nodes = [layer];
    const labels = [];
    p.marks.forEach(([kind, x, y, r, name, description], i) => {
      // Labels are intentionally small and below NPC/player depth; click for a short world-building fact.
      if (i > 5 || !name) return;
      const discoveryId = scene.area.id + ':' + i;
      const discovered = (scene.store?.game?.worldDiscoveries || []).includes(discoveryId);
      const label = scene.add.text((x + .5) * T, (y - .8) * T, (discovered ? '✦ ' : '◇ ') + name, {
        fontFamily: 'monospace', fontSize: '10px', color: discovered ? '#ffda81' : (p.tone === 'creative' ? '#ffe3a1' : '#fff0cf'),
        backgroundColor: '#13212ad9', padding: { x: 5, y: 3 },
      }).setOrigin(.5, .5).setDepth(8).setInteractive({ useHandCursor: true });
      label.on('pointerdown', () => {
        if (scene.inside || scene.switching) return;
        const save = scene.store?.game;
        let found = false;
        if (save) {
          if (!Array.isArray(save.worldDiscoveries)) save.worldDiscoveries = [];
          if (!save.worldDiscoveries.includes(discoveryId)) {
            save.worldDiscoveries.push(discoveryId); found = true;
            scene.store.save();
            label.setText('✦ ' + name).setColor('#ffda81');
          }
        }
        scene.quests?.toast?.([`${found ? '✦ Location discovered' : '🧭 Landmark'} · ${name}`, description]);
      });
      labels.push(label); nodes.push(label);
    });
    const seed = seedFor(scene.area.id);
    const motes = [];
    const lowMotion = (window.innerWidth || 1280) <= 600 || !!window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;
    const count = lowMotion ? (scene.area.theme === 'past' ? 9 : 7) : (scene.area.theme === 'past' ? 18 : 14);
    for (let i = 0; i < count; i++) {
      const x = 20 + hash(i * 9, 31, seed) * (scene.area.w * T - 40);
      const y = 20 + hash(i * 7, 47, seed) * (scene.area.h * T - 40);
      if (!walkable(scene.area, x / T, y / T)) continue;
      motes.push({ x, y, p: hash(i, 23, seed) * TAU, vx: .3 + hash(i, 48, seed), type: p.particle });
    }
    scene.__lqWorldFx = { id: scene.area.id, layer, nodes, labels, motes, time: 0, frame: 0, effects: p.effects, lowMotion, reducedMotion: !!window.matchMedia?.('(prefers-reduced-motion: reduce)').matches };
    queueAreaCinematic(scene, p);
    // Since these items are in areaObjs they are destroyed by the existing area cleanup.
    scene.areaObjs?.push(...nodes);
  }
  function detach(scene) {
    if(!scene)return;
    clearTimeout(scene.__lqAreaCinePending);
    stopAreaCinematic(scene);
    if(scene.__lqWorldFx) scene.__lqWorldFx=null;
  }
