   /* quest panel not ready */
    }

        }
      }
    } catch (e) {
      /* town quests not ready */
    }
    if (placeGoal && !ins) {
      const b = building(placeGoal.building);
      if (b) here.push({ kind: 'place', icon: '🏠', label: b.name, x: b.door * 32 + 16, y: (b.y + b.h) * 32 + 4, k: 1, wall: false, b });
    }
    for (const [id, why] of people) {
      const n = sc.npcs && sc.npcs.find((x) => x.def.id === id);
      if (!n) {
        away.otherTown++; // villagers of the other town are not in this scene
        continue;
      }
      const name = (n.def.name || id).split(' (')[0];
      if (!n.sprite.visible) {
        away.npc.push(name + ' (resting)');
        continue;
      }
      if (!!n.inside !== !!ins) {
        away.npc.push(name);
        continue;
      }
      const k = n.sprite.scaleX || 1;
      here.push({ kind: 'npc', icon: '🗣', label: `${name} · ${why}`, x: n.sprite.x, y: n.sprite.y - 6 * k, k, wall: false, npc: n });

  // ---- drawing ------------------------------------------------------------------------------------------------
  // While the eyes are open the world sinks under a dark layer (depth DIM). Only the player and what the Six Eyes
  // see are lifted above it, each with a glowing outline that follows its shape; nothing is labelled.
  const DIM = 59;
  const DIM_ALPHA = 0.85; // how dark the world gets
  const D = { glow: 59.2, aura: 59.4, player: 59.5, npc: 59.6, npcLabel: 59.65, marks: 59.7, markText: 59.75, top: 59.8 };
  const BASE = { aura: 9.5, player: 10, npc: 9, npcLabel: 11, marks: 8.8, markText: 8.9 };

  // Lift (or put back) the game objects that should shine through the dark layer.
  function lift(sc, s, on, npcs) {
    const set = (o, d) => o && o.depth !== d && o.setDepth(d);
    set(sc.player, on ? D.player : BASE.player);
    set(sc.__lqAura && sc.__lqAura.g, on ? D.aura : BASE.aura);
    set(sc.__lqFingers && sc.__lqFingers.g, on ? D.marks : BASE.marks);
    const ts = sc.__lqTenShadows;
    if (ts) {
      set(ts.g, on ? D.marks : BASE.marks);
      ts.texts.forEach((t) => set(t, on ? D.markText : BASE.markText));
    }
    const want = new Set(on ? npcs : []);
    for (const n of s.lifted) if (!want.has(n)) set(n.sprite, BASE.npc), set(n.label, BASE.npcLabel);
    for (const n of want) set(n.sprite, D.npc), set(n.label, D.npcLabel);
    s.lifted = want;
  }

  // A pixel-crisp outline: four copies of the sprite, filled with one colour, nudged one pixel each way.
  function outline(sc, s, spr, color, alpha) {
    const off = Math.max(1, spr.scaleX || 1);
    for (const [dx, dy] of [[-off, 0], [off, 0], [0, -off], [0, off]]) {
      let img = s.glows[s.gi++];
      if (!img) s.glows.push((img = sc.add.image(0, 0, spr.texture.key).setDepth(D.glow)));
      img
        .setTexture(spr.texture.key, spr.frame.name)
        .setOrigin(spr.originX, spr.originY)
        .setScale(spr.scaleX, spr.scaleY)
        .setFlipX(spr.flipX)
        .setPosition(spr.x + dx, spr.y + dy)
        .setAlpha(alpha)
        .setVisible(true);
      fillTint(img, color);
    }
  }

  // Solid-colour tint: Phaser 4 uses setTint + FILL mode (setTintFill was removed); Phaser 3 has setTintFill.
  function fillTint(img, color) {
    const P = window.Phaser;
    if (img.setTintMode && P && P.TintModes) img.setTint(color).setTintMode(P.TintModes.FILL);
    else img.setTintFill(color);
  }

  // Soft glow with no hard edge: stacked faint discs.
  function halo(g, x, y, r, color, a) {
    for (let i = 0; i < 6; i++) {
      g.fillStyle(color, a);
      g.fillCircle(x, y, r * (1 - i * 0.14));
    }
  }

  function drawWorld(sc, s, seen, time, e) {
    const cam = sc.cameras.main;
    const v = cam.worldView;
    const z = cam.zoom || 1;
    const pulse = 0.5 + 0.5 * Math.sin(time * 0.005);
    s.dim.fillStyle(0x030812, DIM_ALPHA * e);
    s.dim.fillRect(v.x - 4, v.y - 4, v.width + 8, v.height + 8);
    const g = s.g;
    const top = s.top.setAlpha(e);
    s.gi = 0;
    const px = sc.player.x,
      py = sc.player.y;
    const onScreen = (t) => t.x > v.x && t.x < v.right && t.y > v.y && t.y < v.bottom;
    const inView = [],
      outView = [];
    for (const t of seen.here) {
      t.d = Math.hypot(t.x - px, t.y - py);
      (onScreen(t) || (t.b && t.b.x * 32 < v.right && (t.b.x + t.b.w) * 32 > v.x && t.b.y * 32 < v.bottom && (t.b.y + t.b.h) * 32 > v.y) ? inView : outView).push(t);
    }
    // The player keeps a faint blue rim so Gojo still stands out.
    outline(sc, s, sc.player, 0x8fe8ff, 0.55 * e);
    lift(sc, s, true, inView.filter((t) => t.npc).map((t) => t.npc));
    inView.forEach((t, i) => {
      const c = COLORS[t.kind];
      const k = t.k;
      if (t.kind === 'npc') {
        halo(g, t.npc.sprite.x, t.npc.sprite.y - 14 * k, 26 * k, c, 0.035 * e);
        outline(sc, s, t.npc.sprite, c, (0.7 + 0.3 * pulse) * e);
      } else if (t.kind === 'place' && t.b) {
        // The whole building is traced in light.
        const x = t.b.x * 32,
          y = t.b.y * 32,
          w = t.b.w * 32,
          h = t.b.h * 32;
        [[7, 0.12], [4, 0.25], [1.5, 0.9]].forEach(([lw, a]) => {
          g.lineStyle(lw, c, a * (0.75 + 0.25 * pulse) * e);
          g.strokeRect(x, y, w, h);
        });
      } else if (t.kind === 'place') {
        g.lineStyle(2, c, (0.6 + 0.4 * pulse) * e);
        g.strokeEllipse(t.x, t.y + 6, 34, 12);
        halo(g, t.x, t.y, 22, c, 0.04 * e);
      } else {
        halo(g, t.x, t.y, 30 * k, c, (0.05 + 0.03 * pulse) * e);
        if (t.wall) {
          // Seen through the roof: the object itself, drawn over the building.
          if (t.kind === 'finger' && window.LQFingers && window.LQFingers.draw) window.LQFingers.draw(top, t.x, t.y, 1, time, i);
          if (t.kind === 'shadow' && window.LQTenShadows && window.LQTenShadows.drawMark) window.LQTenShadows.drawMark(top, t.x, t.y, 1, time, i);
        }
      }
    });
    for (let i = s.gi; i < s.glows.length; i++) s.glows[i].setVisible(false);
    // Glowing chevrons on the edge of the view for what lies beyond it (nearest first, no text).
    const m = 22 / z;
    outView
      .sort((a, b) => a.d - b.d)
      .slice(0, MAX_ARROWS)
      .forEach((t) => {
        const c = COLORS[t.kind];
        const cx = Math.min(v.right - m, Math.max(v.x + m, t.x));
        const cy = Math.min(v.bottom - m, Math.max(v.y + m, t.y));
        const ang = Math.atan2(t.y - cy, t.x - cx);
        const chevron = (size, a) => {
          const L = size / z,
            W2 = size * 0.7 / z;
          const tipX = cx + Math.cos(ang) * L,
            tipY = cy + Math.sin(ang) * L;
          const bx = cx - Math.cos(ang) * L * 0.5,
            by = cy - Math.sin(ang) * L * 0.5;
          top.fillStyle(c, a);
          top.fillTriangle(tipX, tipY, bx + Math.cos(ang + Math.PI / 2) * W2, by + Math.sin(ang + Math.PI / 2) * W2, bx + Math.cos(ang - Math.PI / 2) * W2, by + Math.sin(ang - Math.PI / 2) * W2);
        };
        chevron(17, 0.15);
        chevron(13, 0.3);
        chevron(9, 0.95);
      });
  }

  // One short line: time left, and what is hidden elsewhere (indoors, outdoors or in the other town).
  function hudText(seen, now) {
    const left = Math.max(0, Math.ceil((st.until - now) / 1000));
    const a = seen.away;
    const bits = [];
    if (a.finger) bits.push(`👆${a.finger}`);
    if (a.shadow) bits.push(`🌑${a.shadow}`);
    if (a.npc.length + a.otherTown) bits.push(`🗣${a.npc.length + a.otherTown}`);
    return `<b>👁 ${left}s</b>${bits.length ? ` · elsewhere ${bits.join(' ')}` : ''}`;
  }

  let lastHud = 0;
  function frame(sc, time, delta) {
    const s = sc.__lqSixEyes;
    if (!s) return;
    s.g.clear();
    s.dim.clear();
    s.top.clear();
    const now = performance.now();

    btn.style.display = hasSixEyes(look) ? 'block' : 'none';
    if (st.active && (now >= st.until || !able)) {
      st.active = false;
      st.readyAt = now + COOLDOWN;
    }
    const target = st.active ? 1 : 0;
    const step = Math.min(1, (delta || 16) / 350);
    st.e = target > st.e ? Math.min(1, st.e + step) : Math.max(0, st.e - step);
    // Button label: ready / open / resting.
    const rest = now < st.readyAt ? Math.ceil((st.readyAt - now) / 1000) : 0;
    const label = st.active ? '👁 Six Eyes ✦' : rest ? `👁 ${rest}s` : '👁 Six Eyes';
    if (btn.textContent !== label) btn.textContent = label;
    btn.classList.toggle('on', st.active);
    btn.classList.toggle('rest', !st.active && !!rest);
    veil.style.opacity = String(st.e);
    // The eyes: blindfold off, glowing blue, while the Six Eyes are open.
    if (st.e > 0 && able) {
      if (s.look !== look) {
        s.look = look;
        s.key = window.LQCharacter.ensure(sc, Object.assign({}, look, { blindfold: false, eyeGlow: true, eye: EYE_BLUE }));
      }
      sc.playerKey = s.key;
      s.worn = true;
    } else if (s.worn) {
      s.worn = false;
      try {
        sc.applyRankLook();
      } catch (e) {
        /* the scene is being rebuilt */
      }
    }
    if (st.e <= 0 || !able) {
      if (s.on) {
        s.on = false;
        lift(sc, s, false, []);
        s.glows.forEach((img) => img.setVisible(false));
      }
      if (now < st.msgUntil) {
        hud.style.display = 'block';
        hud.innerHTML = st.msg;
      } else hud.style.display = 'none';
      return;
    }
    const seen = sense(sc);
    s.on = true;
    drawWorld(sc, s, seen, time, st.e);
    if (now - lastHud > 200) {
      lastHud = now;
      hud.style.display = 'block';
      hud.innerHTML = hudText(seen, now);
    }
  }

  function attach(sc) {
    if (sc.__lqSixEyes) return;
    sc.__lqSixEyes = {
      dim: sc.add.graphics().setDepth(DIM),
      g: sc.add.graphics().setDepth(D.glow),
      top: sc.add.graphics().setDepth(D.top),
      glows: [],
      gi: 0,
      lifted: new Set(),
      on: false,
      key: null,
      look: null,
      worn: false,
    };
    sc.events.on('update', (time, delta) => frame(sc, time, delta));
    sc.events.once('shutdown', () => {
      sc.__lqSixEyes = null;
    });
  }

  setInterval(() => {
    const sc = getScene();
    if (sc && sc.add && sc.events && sc.player && !sc.__lqSixEyes) {
      try {
        attach(sc);
      } catch (e) {
        console.warn('six eyes attach failed', e);
      }
    }
  }, 1000);

  // Ctrl tap (pressed and released alone), the same key as Megumi's Divine Dog; each skill checks its own look.
  let ctrlAlone = false;
  const typing = () => {

  };

    'keydown',

      if (e.key === 'Control') {
        if (!e.repeat) ctrlAlone = !e.altKey && !e.shiftKey && !e.metaKey;
      } else ctrlAlone = false;
    },
    true,

  window.addEventListener(
    'keyup',

    },

  window.LQGojo = {
    greet,
    preview: () => greet(),
    toggleSixEyes: toggle,
    isSixEyesOpen: () => st.active,
    // What the Six Eyes would show right now (for testing in the Console).
    sense: () => {
      const sc = getScene();
      return sc && sc.area ? sense(sc) : null;
    },
  };
})();

/*
 * Device choice: right after Start on the title screen, the player must say how they play: 💻 Computer, 📱 Phone or
 * 📲 iPad / tablet. The likely answer is pre-selected (from the screen size and touch support), but nothing goes on
 * until the player confirms: there is no close button and Esc does nothing. The answer is remembered in localStorage
 * and only asked once; it can be changed later from the Menu (see LQDevice.change()).
 *
 * The choice is written to <html data-device="pc|phone|tablet"> so CSS and other scripts can adapt (touch controls
 * only appear on phone and tablet), and a 'lifequest:device' event is sent whenever it changes.
 *
 * window.LQDevice.ask() returns a promise that resolves once a device is chosen (at once if it was chosen before).
 */
(function () {
  'use strict';

  const KEY = 'lifequest.device';
  const DEVICES = [
    { id: 'pc', icon: '💻', title: 'Computer', desc: 'Keyboard: WASD or arrow keys to move, E to talk.' },
    { id: 'phone', icon: '📱', title: 'Phone', desc: 'On-screen joystick and buttons, sized for small screens.' },
    { id: 'tablet', icon: '📲', title: 'iPad / tablet', desc: 'On-screen joystick and larger buttons.' },
  ];
  const valid = (id) => DEVICES.some((d) => d.id === id);

  function read() {
    try {
      const v = localStorage.getItem(KEY);
      return valid(v) ? v : null;
    } catch (e) {
      return null;
    }
  }

  // Best guess, only used to pre-select an answer. The primary pointer decides (a touchscreen laptop still has a
  // mouse or trackpad as its main pointer); iPads that report themselves as a Mac are caught by 'MacIntel' + touch.
  function guess() {
    const mq = (q) => !!(window.matchMedia && matchMedia(q).matches);
    const ipadAsMac = navigator.platform === 'MacIntel' && (navigator.maxTouchPoints || 0) > 1;
    if (ipadAsMac) return 'tablet';
    if (!mq('(pointer: coarse)')) return 'pc';
    const short = Math.min(screen.width || innerWidth, screen.height || innerHeight);
    return short >= 600 ? 'tablet' : 'phone';
  }

  let current = read();
  function apply(id) {
    current = id;
    document.documentElement.dataset.device = id;
    window.dispatchEvent(new CustomEvent('lifequest:device', { detail: { device: id } }));
  }
  if (current) apply(current);

  const CSS = `
#lq-device{position:fixed;inset:0;z-index:260;display:flex;align-items:center;justify-content:center;padding:16px;box-sizing:border-box;background:#070a16f2;color:#eef2ff;font-family:Inter,system-ui,-apple-system,"Segoe UI",Roboto,sans-serif;animation:lqdev-in .35s ease}

#lq-device .opt{display:flex;flex-direction:column;align-items:center;gap:6px;text-align:center;padding:16px 10px;border-radius:14px;border:2px solid #34416b;background:#121a33;color:inherit;font:inherit;cursor:pointer;transition:border-color .15s,background .15s}
#lq-device .opt:hover{border-color:#6b82c9}
#lq-device .opt.on{border-color:#ffd36b;background:#1e2748}
#lq-device .opt .ic{font-size:34px;line-height:1}
#lq-device .opt>div{display:flex;flex-direction:column;gap:4px}

        <button type="button" class="go">Continue</button>
        <p class="note">You can change this later in Menu → Settings.</p>
      </div>`;
      document.body.appendChild(el);
      let pick = pre;
      const opts = [...el.querySelectorAll('.opt')];
      const select = (id) => {
        pick = id;
        opts.forEach((o) => {
          o.classList.toggle('on', o.dataset.dev === id);
          o.setAttribute('aria-checked', String(o.dataset.dev === id));
        });
      };
      opts.forEach((o) => o.addEventListener('click', () => select(o.dataset.dev)));
      const go = el.querySelector('.go');
      // Esc must not close it, and keys must not reach the game underneath.
      const onKey = (ev) => {
        if (ev.key === 'Escape') ev.preventDefault();
        ev.stopPropagation();
      };
      addEventListener('keydown', onKey, true);
      go.addEventListener('click', () => {
        if (!valid(pick)) return;
        try {
          localStorage.setItem(KEY, pick);
        } catch (e) {
          /* private mode: the choice lasts for this visit only */
        }
        removeEventListener('keydown', onKey, true);
        el.remove();
        st.remove();
        apply(pick);
        resolve(pick);
      });
      setTimeout(() => go.focus(), 50);
    });
  }

  window.LQDevice = {
    list: DEVICES,
    get: () => current,
    guess,
    // First visit: ask (cannot be skipped). Later visits: resolve at once with the saved choice.
    // index.html?device=ask always asks (handy for testing the first-visit screen).
    ask: () => (current && !/[?&]device=ask\b/.test(location.search) ? Promise.resolve(current) : open()),
    // From the Menu: ask again, with the current choice pre-selected.
    change: () => open(),
    isTouch: () => current === 'phone' || current === 'tablet',
  };
})();

  let pointer = null;
  function setDirs(dx, dy) {
    const c = controls();
    // 8 directions: a component counts once it is a good part of the push (about 22° either side of an axis).
    const len = Math.hypot(dx, dy);
    const on = len > 0.25;
    const ux = on ? dx / len : 0,
      uy = on ? dy / len : 0;
    c.right = on && ux > 0.38;
    c.left = on && ux < -0.38;
    c.down = on && uy > 0.38;
    c.up = on && uy < -0.38;
  }
  function release() {
    pointer = null;
    joy.classList.remove('held');
    knob.style.transform = '';
    setDirs(0, 0);
  }
  function bindJoystick() {
    const move = (ev) => {
      const r = joy.getBoundingClientRect();
      const R = r.width / 2;
      let dx = (ev.clientX - (r.left + R)) / R,
        dy = (ev.clientY - (r.top + R)) / R;
      const len = Math.hypot(dx, dy);
      if (len > 1) (dx /= len), (dy /= len);
      knob.style.transform = `translate(${dx * R * 0.62}px,${dy * R * 0.62}px)`;
      setDirs(dx, dy);
    };
    joy.addEventListener('pointerdown', (ev) => {
      ev.preventDefault();
      pointer = ev.pointerId;
      try {
        joy.setPointerCapture(ev.pointerId);
      } catch (e) {
        /* capture is optional */
      }
      joy.classList.add('held');
      move(ev);
    });
    joy.addEventListener('pointermove', (ev) => {
      if (ev.pointerId === pointer) move(ev);
    });
    for (const t of ['pointerup', 'pointercancel', 'lostpointercapture'])
      joy.addEventListener(t, (ev) => {
        if (ev.pointerId === pointer) release();
      });
  }

  function bindButton(el, fn) {

      ev.preventDefault();
      el.classList.add('down');
      fn();
    });

    // Keyboard / screen reader activation (click without a pointer press).
    el.addEventListener('click', (ev) => {

      function train(now) {
        const total = 50 + 4 + 50 + 4 + 57;
        const left = Math.round(W / 2 - total / 2);
        const jig = (k) => (speed > 8 && Math.sin(now / 70 + k * 1.7) > 0.7 ? 1 : 0);
        const y = GY - 2;
        carriage(left, y - jig(0), 0, now, -1);
        carriage(left + 54, y - jig(1), 1, now, 2);
        locomotive(left + 108, y - jig(2));
        // smoke
        // Puffs while moving; once stopped, only a thin wisp that fades quickly.
        const moving = speed > 8;
        smokeT += moving ? 1 : 0.12;
        if (smokeT > 7) {
          smokeT = 0;
          smoke.push({ x: left + 108 + 44, y: GY - 37, r: moving ? 2.5 : 1.5, life: 0, max: moving ? 2.4 : 1.1 });
        }
        for (const p of smoke) {
          p.life += 1 / 60;
          p.x -= (speed / 60) * 0.85 + 0.15;
          p.y -= 0.32;
          p.r += 0.07;
        }
        for (let k = smoke.length - 1; k >= 0; k--) if (smoke[k].life > smoke[k].max) smoke.splice(k, 1);
        for (const p of smoke) disc(p.x, p.y, p.r, `rgba(255,255,255,${Math.max(0, 0.75 * (1 - p.life / p.max)).toFixed(2)})`);

        }
      }

      function show(i) {
        li = i;
        const l = lines[i];
        box.classList.add('show');
        enEl.textContent = '';
        viEl.textContent = l.vi;
        viEl.classList.remove('show');

        if (l.brake && !brake) {
          const D = W + 140;
          brake = { t0: performance.now(), c0: cam, D, T: (2 * D) / CRUISE };

      function finishTyping() {
        if (!typing) return;
        enEl.textContent = typing.text;
        typing = null;
        viEl.classList.add('show');
      }
      function next() {
        if (ended) return;
        if (li < 0) {
          show(0);
          return;
        }
        if (typing) {
          finishTyping();
          return;
        }
        if (li < lines.length - 1) show(li + 1);
        else end();
      }
      function end() {
        if (ended) return;
        ended = true;
        root.classList.remove('on');
        removeEventListener('keydown', onKey, true);
        setTimeout(() => {
          cancelAnimationFrame(raf);
          removeEventListener('resize', size);
          root.remove();
          try {
            if (paused && sc && sc.scene) sc.scene.resume();
          } catch (e) {}
          playing = null;
          resolve();
        }, 650);
      }
      const onKey = (ev) => {
        // The scene owns the keyboard while it plays: the game must not move or open menus behind it.
        ev.preventDefault();
        ev.stopImmediatePropagation();
        if (ev.repeat) return;
        if (ev.key === ' ' || ev.key === 'Enter' || ev.key === 'e' || ev.key === 'E' || ev.key === 'ArrowRight') next();
      };
      addEventListener('keydown', onKey, true);
      root.addEventListener('click', (ev) => {
        if (ev.target.closest('.skip')) {
          end();
          return;
        }
        next();
      });

      function frame(now) {
        if (!document.body.contains(root)) return;
        const dt = Math.min(0.05, (now - last) / 1000);
        last = now;
        if (brake) {
          const p = Math.min(1, (now - brake.t0) / 1000 / brake.T);
          cam = brake.c0 + brake.D * (1 - (1 - p) * (1 - p));
          speed = CRUISE * (1 - p);
        } else cam += speed * dt;
        if (li < 0 && now - t0 > 2500) show(0);
        if (typing) {
          typing.t += dt;
          const n = Math.min(typing.text.length, Math.floor(typing.t * 40));
          if (n !== typing.n) {
            typing.n = n;
            enEl.textContent = typing.text.slice(0, n);
          }
          if (n >= typing.text.length) finishTyping();
        }
        draw(now);
        raf = requestAnimationFrame(frame);
      }
      raf = requestAnimationFrame(frame);
    });
    return playing;
  }

  function renderCoach(i, list, shownStep, note) {
    const el = mountCoach();
    const key = i + '|' + shownStep.id + '|' + (note || '') + '|' + isTouch();
    if (key !== coachKey) {
      const changed = coachKey.split('|')[0] !== String(i);
      coachKey = key;
      el.querySelector('.num').textContent = i + 1 + '/' + list.length;
      el.querySelector('.bar i').style.width = Math.round((i / list.length) * 100) + '%';
      el.querySelector('.en').innerHTML = (note ? note.en + ' ' : '') + shownStep.en;
      el.querySelector('.vi').textContent = (note ? note.vi + ' ' : '') + shownStep.vi;
      const act = el.querySelector('.act');
      act.innerHTML = '';

        const b = document.createElement('button');
        b.type = 'button';
        b.className = 'ex';
        b.dataset.do = 'example';
        b.dataset.text = shownStep.example;

  function advance() {
    const g = game();
    if (!g || !g.tut) return;
    const list = steps();
    g.tut.step = Math.min(list.length, (g.tut.step | 0) + 1);
    base = null;
    if (g.tut.step >= list.length) finish(false);
    else save();
  }
  function jumpTo(id) {
    const g = game();
    const i = stepIndex(id);
    if (!g || !g.tut || i < 0 || (g.tut.step | 0) >= i) return;
    g.tut.step = i;
    base = null;
    save();
  }
  function finish(skipped) {
    const g = game();
    if (!g) return;
    g.tut = Object.assign(g.tut || {}, { done: Date.now(), skipped: !!skipped });
    save();
    hideCoach();
    coachKey = '';
    if (skipped) toast('🎓 Tutorial skipped', 'You can replay it any time', 'Menu → Settings → Intro & tutorial');
    else toast('🎓 Tutorial complete', 'Welcome to English Town!', 'Finish “First Day in Town” on the quest card.');
  }

    const sc = getScene();
    const g = game();
    if (playing || !sc || !g || !sc.player || !g.tut || !g.tut.intro || g.tut.done) return hideCoach();
    const list = steps();
    const i = Math.max(0, Math.min(list.length - 1, g.tut.step | 0));
    const s = list[i];
    const msgs = (g.stats && g.stats.messages) || 0;
    if (!base || base.i !== i) base = { i, x: sc.player.x, y: sc.player.y, msgs };
    const chatId = sc.chat && sc.chat.isOpen ? sc.chat.currentId : null;

    // Talking to Mia early skips the steps before it.
    if (chatId === 'kid' && i < stepIndex('say')) return jumpTo('say');

    const mia = sc.npcs && sc.npcs.find((n) => n.def && n.def.id === 'kid');
    const miaHere = !!(mia && mia.sprite && mia.sprite.visible && !!mia.inside === !!sc.inside);
    const dMia = miaHere ? Math.hypot(sc.player.x - mia.sprite.x, sc.player.y - mia.sprite.y) : Infinity;

    let shownStep = s;
    let note = null;
    if (s.id === 'move' && Math.hypot(sc.player.x - base.x, sc.player.y - base.y) > 80) return advance();
    if (s.id === 'find' && dMia < 90) return advance();
    if (s.id === 'talk' && dMia > 220) shownStep = list[stepIndex('find')];
    if (s.id === 'say') {
      if (chatId !== 'kid') {
        shownStep = list[stepIndex('talk')];
        note = { en: 'Talk to Mia again.', vi: 'Hãy nói chuyện lại với Mia.' };
        if (dMia > 220) shownStep = list[stepIndex('find')];
      } else if (msgs > base.msgs) return advance();
    }
    if (s.id === 'tips' && !chatId) return advance();
    if (s.id === 'menu' && shown('l11-shell')) return advance();

    // Hide behind big panels, and while chatting with someone other than Mia (or chatting at all on the last steps).
    const chatOk = s.id === 'say' || s.id === 'tips';
    if (COVERS.some(shown) || (chatId && (!chatOk || chatId !== 'kid'))) return hideCoach();
    renderCoach(i, list, shownStep, note);
    // On small screens the chat is tall: the card moves up over the minimap so it does not hide the chat's ✕.
    coachEl.classList.toggle('chatting', !!chatId);
  }

  // ---- yellow arrow to Mia (steps "find" and "talk") --------------------------------------------------------------
  function drawGuide(sc, time) {
    const st = sc.__lqIntro;
    if (!st) return;
    const gfx = st.g;
    gfx.clear();
    const g = game();
    if (playing || !g || !g.tut || !g.tut.intro || g.tut.done || !sc.player || sc.switching) return;
    const id = steps()[Math.max(0, g.tut.step | 0)];
    if (!id || (id.id !== 'find' && id.id !== 'talk' && id.id !== 'say')) return;
    if (sc.chat && sc.chat.isOpen) return;
    const mia = sc.npcs && sc.npcs.find((n) => n.def && n.def.id === 'kid');
    if (!mia || !mia.sprite || !mia.sprite.visible || !!mia.inside !== !!sc.inside) return;
    const mx = mia.sprite.x;
    const my = mia.sprite.y;
    const b = Math.sin(time * 0.006) * 4;
    // a bouncing marker over Mia's head
    const hy = my - 52 + b;
    gfx.fillStyle(0x3a2a00, 1);
    gfx.fillTriangle(mx - 10, hy - 2, mx + 10, hy - 2, mx, hy + 12);
    gfx.fillStyle(0xffd84a, 1);
    gfx.fillTriangle(mx - 7, hy, mx + 7, hy, mx, hy + 9);
    // an arrow around the player pointing at her
    const px = sc.player.x;
    const py = sc.player.y - 16;
    const d = Math.hypot(mx - px, my - py);
    if (d < 110) return;
    const a = Math.atan2(my - 16 - py, mx - px);
    const r0 = 30 + Math.sin(time * 0.008) * 3;
    const tip = { x: px + Math.cos(a) * (r0 + 16), y: py + Math.sin(a) * (r0 + 16) };
    const l = { x: px + Math.cos(a) * r0 + Math.cos(a + Math.PI / 2) * 9, y: py + Math.sin(a) * r0 + Math.sin(a + Math.PI / 2) * 9 };
    const rr = { x: px + Math.cos(a) * r0 - Math.cos(a + Math.PI / 2) * 9, y: py + Math.sin(a) * r0 - Math.sin(a + Math.PI / 2) * 9 };
    gfx.lineStyle(3, 0x3a2a00, 1);
    gfx.strokeTriangle(tip.x, tip.y, l.x, l.y, rr.x, rr.y);
    gfx.fillStyle(0xffd84a, 1);
