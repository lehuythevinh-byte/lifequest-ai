          if (!canAct()) return null;
          if (route.kind === 'direction') {
            route.remaining -= delta;
            if (route.remaining <= 0) {
              route = null;
              notify('✅ Movement command completed.');
              return null;
            }
            return { x: route.vector[0], y: route.vector[1] };
          }
          // Check destination NPC's *current* position (NPCs may wander over time).
          const npc = route.npcId && scene.npcs.find((n) => n.def.id === route.npcId);
          if (
            npc &&
            Math.hypot(scene.player.x - npc.sprite.x, scene.player.y - npc.sprite.y) <= 44
          ) {
            const label = route.label;
            const autoTalk = route.autoTalk;
            route = null;
            if (autoTalk && !scene.chat.isOpen) {
              scene.chat.open(npc.def);
              notify(`💬 Reached ${label} and opened a conversation.`);
            } else notify(`✅ Reached ${label}. Press E to chat.`);
            return null;
          }
          while (route?.points.length) {
            const waypoint = route.points[0];
            const gap = Math.hypot(waypoint.x - scene.player.x, waypoint.y - scene.player.y);
            if (gap > 5)
              return {
                x: (waypoint.x - scene.player.x) / gap,
                y: (waypoint.y - scene.player.y) / gap,
              };
            route.points.shift();
          }
          if (route) {
            const label = route.label;
            const autoTalk = route.autoTalk;
            const npcId = route.npcId;
            if (npc && route.retries < 2) {
              const points = routeToPoint(npc.sprite.x, npc.sprite.y, 2);
              if (points) {
                route = {
                  kind: 'path',
                  points,
                  label,
                  npcId,
                  autoTalk,
                  retries: route.retries + 1,
                };
                return null;
              }
            }
            route = null;
            notify(
              npc
                ? `📍 Got close to ${label}; the NPC has moved, look around.`
                : `✅ Reached ${label}.`,
            );
          }
          return null;
        }

        return {
          attachTownScene,
          detachTownScene,
          updateTownAutopilot,
          cancelTownAutopilot,
          executeTownCommand,
          getTownContext,
        };
      })();

    return selected;
  }
  function attachArea(area){
    if(!area?.buildings||!area?.obstacles)return;
    const buildings=area.buildings.flatMap(b=>shapesForBuilding(area.id,b).map(p=>Object.assign(wrap(p),{name:b.name,category:'building'})));
    const world=area.obstacles.map(o=>Object.assign(wrap(rect(o.x*T,o.y*T,o.w*T,o.h*T)),{category:'terrain'}));
    area.__lqPrecise={buildings,terrain:world,stalls:[]};
    buildIndex(area.__lqPrecise,area.id);
    // Tile navigation is rebuilt from the contact geometry. This grid is for
    // NPC route planning only; player movement below tests polygons precisely.
    for(let iy=0;iy<area.h;iy++)for(let ix=0;ix<area.w;ix++){
      const px=(ix+.5)*T,py=(iy+.5)*T;
      area.blocked[iy][ix]=world.some(s=>hitShape(s,px,py,1)) ||
        buildings.some(s=>hitShape(s,px,py,4));
    }
  }
  function installStalls(area,stalls){
    if(!area?.__lqPrecise)return;
    area.__lqPrecise.stalls=stalls.map(s=>{
      const x=(s.x+.5)*T,y=(s.y+.5)*T+(s.row==='north'?11:0);
      return Object.assign(wrap(rect(x-57,y-59,116,93)),{category:'stall',name:s.name});
    });
    buildIndex(area.__lqPrecise,area.id);
    for(const s of area.__lqPrecise.stalls){
      const x0=Math.max(0,Math.floor(s.box.x/T)),x1=Math.min(area.w-1,Math.floor(s.box.right/T));
      const y0=Math.max(0,Math.floor(s.box.y/T)),y1=Math.min(area.h-1,Math.floor(s.box.bottom/T));
      for(let y=y0;y<=y1;y++)for(let x=x0;x<=x1;x++)
        if(hitShape(s,(x+.5)*T,(y+.5)*T,3))area.blocked[y][x]=true;
    }
  }
  function contact(area,x,y,r=R){
    const q=area?.__lqPrecise;
    if(!q)return null;
    let category=null;
    for(const s of candidates(q,x,y,r)){
      if(!hitShape(s,x,y,r))continue;
      if(s.category==='terrain')return 'terrain';
      if(s.category==='building')category='building';
      else if(s.category==='stall'&&category!=='building')category='stall';
    }
    return category||false;
  }
  function playerBlocked(area,x,y){
    if(!area||!Number.isFinite(x)||!Number.isFinite(y)||
      x<R||y<R||x>area.w*T-R||y>area.h*T-R)return true;
    return !!contact(area,x,y,R);
  }
  function debug(area,x,y){
    const q=area?.__lqPrecise;
    if(!q)return null;
    return {map:area.id,worldX:x,worldY:y,category:contact(area,x,y,R)||'walkable',
      footRadius:R,shapes:q.buildings.length,terrain:q.terrain.length,stalls:q.stalls.length};
  }
  return Object.freeze({version:'52.2',radius:R,shapesForBuilding,attachArea,installStalls,
    contact,playerBlocked,debug,hitShape,rect,roof,wrap,indexStats:()=>({...indexMetrics})});
