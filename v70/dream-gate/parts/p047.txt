        function findMention(s, entries, aliases) {
          const hits = [];
          for (const x of entries || []) {
            let names = [
              x.id,
              x.name,
              String(x.name)
                .replace(/\([^)]*\)/g, '')
                .replace(/^(mr|ms|dr|officer|father)\.? /i, ''),
              ...(aliases[x.id] || []),
              ...(aliases[x.name] || []),

            for (const name of names) {
              const n = fold(name);
              if (n.length > 2 && has(s, n)) hits.push({ x, len: n.length });
            }

        function failureText(err) {
          if (err?.code === 'BAD_CORRECTION')
            return 'Gemini trả về bản sửa thiếu thông tin. Phần trên dùng quy tắc ngoại tuyến.';
          if (err?.code === 'NO_KEY')
            return '🔑 No Gemini API key yet, so the AI Coach is in offline mode. Quizzes, word lookups and control commands still work. Enter your key in ⚙ AI (K) for free-form chat.';
          if (err?.code === 429)
            return '⏳ Gemini is rate-limited (429). Control commands and offline quizzes still work; try Gemini again later.';
          if (err?.code === 403 || err?.code === 401)
            return '🔑 The Gemini API key is not authorized or is invalid (401/403). Check the key in ⚙ AI (K).';
          if (err?.code === 404)
            return '🔎 Gemini model not found (404). Check the model name in ⚙ AI (K), for example gemini-2.5-flash.';
          if (err?.code === 400)
            return '⚠️ Gemini rejected the request (400). Check the model name and try a shorter message.';
          if (err?.name === 'AbortError')
            return '⌛ The Gemini connection was interrupted or timed out; please try again.';
          return '🔌 Could not connect to Gemini. Check your internet and the API key in ⚙ AI (K). Control and offline quiz features still work.';
        }
        async function run(message, ctx, history, options = {}) {
          const mode = modes.includes(options.mode) ? options.mode : 'auto',
            level = options.level || 'B1';
          const original = sentenceTutor.request(message);
          if (!original) {
            const q = quizAnswer(message, level);
            if (q) return { ...q, kind: 'text' };
            const lesson = studyLab.handle(message, level);
            if (lesson) return { kind: 'text', reply: lesson, local: true };
            const action = detectAction(message, ctx, mode);
            if (action) return { kind: 'action', command: action, local: true };
          }
          if (mode === 'control')
            return {
              kind: 'text',
              reply:
                '🎮 Control mode: use “đến Lily”, “đến thư viện”, “đi sang phải 2 giây”, “nói chuyện với người gần nhất”, “vị trí” or “dừng”. To ask English questions, switch to Chat/Learn.',
              local: true,
            };
          const s = fold(message);
          // Game facts are always grounded in actual state rather than fabricated by an LLM.
          const knowledge = !original && localKnowledge(message, ctx);
          if (
            knowledge &&
            /\?|ai la|la ai|who is|what is|o dau|where is|nhiem vu|quest|npc|dia diem|toa nha|goi y|may gio|vi tri/.test(
              s,
            )
          )
            return { kind: 'text', reply: knowledge, local: true };
          const quizRequest =
            /^(quiz|trac nghiem|lam bai tap|bai kiem tra|cau hoi tieng anh|luyen tap quiz)$/.test(
              s,
            );
          if (quizRequest) return { kind: 'text', reply: 'Type “quiz” to start.', local: true };
          // With a live key prefer Gemini for open-ended tutoring and accurate corrections/translations.
          const ai = readAI(),
            hasKey = !!String(
              ai.geminiKey || window.LIFEQUEST_AI_CONFIG?.GEMINI_API_KEY || '',
            ).trim(),
            active = hasKey || (await window.LQGemini.proxyAvailable());
          if (!hasKey && !original) {
            if (
              /^(ket thuc luyen noi|thoat luyen noi|stop roleplay|end roleplay)$/.test(s) &&
              practiceState
            ) {
              practiceState = null;
              return { kind: 'text', reply: '✅ Conversation practice finished.', local: true };
            }
            const inRoleplay =
              practiceState &&
              !/^(luyen hoi thoai|roleplay|practice conversation|luyen noi|speaking)/.test(s)
                ? practiceReply(message)
                : null;
            if (inRoleplay) return { kind: 'text', reply: inRoleplay, local: true };
            // Fixed local lessons and words should not require a server at all.
            const definite =
              /^(tu vung|tra tu|nghia cua|what does|meaning of|vocabulary|explain word|luyen hoi thoai|roleplay|practice conversation|luyen noi|speaking)/.test(
                s,
              );
            if (definite) {
              const offline = offlineTutor(message, ctx);
              if (offline) return { kind: 'text', reply: offline, local: true };
            }
          }
          if (!active || /^(hi|hello|chao ban|xin chao|cam on|thank you|thanks)$/.test(s)) {
            const offline = offlineTutor(message, ctx);
            if (offline) return { kind: 'text', reply: offline, local: true };
          }
          if (!active) {
            return {
              kind: 'text',
              reply:
                '🤖 I can answer basic questions in offline mode. For more flexible chat and more accurate corrections, enter a Gemini API key in ⚙ AI (K). Try “quiz”, “sửa câu: I goes to school”, “từ vựng: library”, or “Lily là ai?”.',
              local: true,
            };
          }
          try {
            return {
              kind: 'text',
              reply: await gemini(message, ctx, history, level, options.signal, mode, original),
              local: false,
            };
          } catch (err) {
            if (options.signal?.aborted) throw err;
            const fallback = offlineTutor(message, ctx);
            return {
              kind: 'text',
              reply: (fallback ? fallback + '\n\n' : '') + failureText(err),
              local: true,
            };
          }
        }
        return { run, detectAction, offlineTutor, quizAnswer };
      })();
