  function star(g, cx, cy, r, color, alpha) {
    const pts = [];
    for (let i = 0; i < 10; i++) {
      const a = -Math.PI / 2 + (i * Math.PI) / 5;
      const rr = i % 2 ? r * 0.45 : r;
      pts.push({ x: cx + Math.cos(a) * rr, y: cy + Math.sin(a) * rr });
    }
    g.fillStyle(color, alpha).fillPoints(pts, true);
  }
  function eye(g, cx, cy, r, color, alpha) {
    g.lineStyle(3, color, alpha);
    g.beginPath();
    g.arc(cx, cy + r * 0.9, r * 1.25, Math.PI * 1.22, Math.PI * 1.78, false);
    g.strokePath();
    g.beginPath();
    g.arc(cx, cy - r * 0.9, r * 1.25, Math.PI * 0.22, Math.PI * 0.78, false);
    g.strokePath();
    g.fillStyle(color, alpha).fillCircle(cx, cy, r * 0.32);
  }
  function wave(g, cx, cy, r, color, alpha) {
    g.lineStyle(3, color, alpha);
    g.beginPath();
    for (let i = 0; i <= 20; i++) {
      const x = cx - r + (i / 20) * r * 2;
      const y = cy + Math.sin((i / 20) * Math.PI * 2) * r * 0.35;
      if (i) g.lineTo(x, y);
      else g.moveTo(x, y);
    }
    g.strokePath();
  }
  function glow(g, x, y, r, color, a) {
    for (let k = 3; k >= 1; k--) g.fillStyle(color, a * 0.12 * k).fillCircle(x, y, r * (1 + (3 - k) * 0.6));
  }

      if (!here(s) || s.inside || s.switching || filming) return;
      pending = false;
      play();
    }, 800);

        clearInterval(wait);

      }
    }, 500);
  }

    id: ID,
    play,
    enter,
    isFilming: () => filming,
    // Testing: see the film again on the next arrival.
    reset() {
      const g = game();
      if (g) {
        delete g.artIntroSeen;
        save();
      }
    },
  };
})();

      pts.push({ x: cx + Math.cos(a) * r, y: cy + Math.sin(a) * r });

  function graffiti(g, b, R) {

  function paintFloor(sc, a) {
    const key = 'lq_art_floor67';
    if (sc.textures.exists(key)) return key;
    const R = rnd(4242);
    const g = sc.make.graphics({ x: 0, y: 0, add: false });
    const pick = () => PAL[Math.floor(R() * PAL.length)];
    // 1. paint splatters on streets and squares (not on the buildings)
    for (let i = 0; i < 320; i++) {
      const x = R() * a.w;
      const y = R() * a.h;
      if (onBuilding(a, x, y, 0.4)) continue;
      const c = pick();
      const r = 3 + R() * 9;
      g.fillStyle(c, 0.18 + R() * 0.22).fillCircle(x * T, y * T, r);
      for (let k = 0; k < 5; k++) g.fillCircle(x * T + (R() - 0.5) * r * 4, y * T + (R() - 0.5) * r * 4, 1 + R() * 2.5);
    }
    // 2. paint dripping down from the bottom of every facade
    for (const b of a.buildings) {
      const by = (b.y + b.h) * T;
      for (let x = b.x * T + 6; x < (b.x + b.w) * T - 6; x += 7 + R() * 16) {
        const c = pick();
        const len = 6 + R() * 26;
        g.fillStyle(c, 0.75).fillRect(x, by - 2, 3, len);
        g.fillCircle(x + 1.5, by - 2 + len, 2.6);
      }
    }
    // 3. rainbow crosswalks where the streets cross
    for (const v of a.roadsV)
      for (const h of a.roadsH) {
        if (h.y < (v.y0 ?? 0) || h.y > (v.y1 ?? a.h)) continue;
        for (let i = 0; i < 6; i++) {
          g.fillStyle(PAL[i], 0.55).fillRect(v.x * T + 4, (h.y - 1) * T + 4 + i * 5, v.w * T - 8, 3);
          g.fillRect((v.x - 1) * T + 4 + i * 5, h.y * T + 4, 3, h.h * T - 8);
        }
      }
    // 4. little cyan stars leading from the tunnel up to the plaza
    for (let y = 47; y > 44; y -= 2) sparkle(g, 35.5 * T + (y % 4 ? 10 : -10), y * T, 6, 0x35c8d0, 0.85);
    // 5. the big floor mural of the three marks (scaled up around its centre)
    g.save();
    g.translateCanvas(MURAL.x, MURAL.y);
    g.scaleCanvas(MURAL.s, MURAL.s);
    mural(g, 0, 0);
    g.restore();
    // 6. V67 art sidewalks and planters; don't overpaint dedicated new facades.
    if (window.LQArtV67?.paintStreetDetails) window.LQArtV67.paintStreetDetails(g, a);
    else for (const b of a.buildings) graffiti(g, b, R);
    window.LQArtV69?.paintStreets?.(g,a);
    // 7. Fully accessible Living Canvas and the new walk-through Dream Gate.
    site(g, R);
    g.generateTexture(key, a.w * T, a.h * T);
    g.destroy();
    return key;
  }

  // ---- lights -----------------------------------------------------------------------------------------------------------
  // String lights hang over the streets, from post to post (every 6 tiles), with a little sag.
  function strings(a) {
    const out = [];
    const span = (x1, y1, x2, y2) => out.push([x1 * T, y1 * T, x2 * T, y2 * T]);
    for (const h of a.roadsH) for (let x = (h.x0 ?? 0) + 1; x <= (h.x1 ?? a.w) - 7; x += 6) span(x, h.y + 0.2, x + 6, h.y + 0.2);
    for (const v of a.roadsV) for (let y = (v.y0 ?? 0) + 3; y < (v.y1 ?? a.h) - 6; y += 6) span(v.x - 0.2, y, v.x - 0.2, y + 6);
    return out;
  }
  function makeLights(sc, a) {
    const st = {
      floor: sc.add.image(0, 0, paintFloor(sc, a)).setOrigin(0).setDepth(0.5),
      wires: sc.add.graphics().setDepth(12),
      bulbs: sc.add.graphics().setDepth(12.1).setBlendMode(ADD),
      glow: sc.add.graphics().setDepth(0.6).setBlendMode(ADD),
      motes: [],
      spans: strings(a),
      area: a,
    };
    const R = rnd(9);
    // the wires never change: draw them once
    for (const [x1, y1, x2, y2] of st.spans) {
      st.wires.lineStyle(1, 0x101018, 0.9);
      st.wires.beginPath();
      for (let k = 0; k <= 12; k++) {
        const t = k / 12;

(function () {

  // Older saves wore Gojo before the sixEyes flag existed: the Infinity aura plus the blindfold is the same preset.
  const hasSixEyes = (look) => !!look && (!!look.sixEyes || (look.aura === 'infinity' && !!look.blindfold));

  // ---- 1. greeting ----------------------------------------------------------------------------------------------
  const GREET_LINES = ['Throughout heaven and earth,', 'I alone am the honored one.'];
  const GREET_FONT = 'Cinzel, "Trajan Pro", "Times New Roman", Georgia, serif';
  const G_FADE_IN = 450;
  const G_CHAR = 48; // ms per letter
  const G_GAP = 350; // pause between the two lines
  const G_HOLD = 1500; // the finished line stays up
  const G_FADE_OUT = 800;

  function chime() {
    try {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) return;
      const ac = new AC();
      const t = ac.currentTime;
      [659, 988, 1319].forEach((hz, i) => {
        const o = ac.createOscillator();
        const g = ac.createGain();
        o.type = 'sine';
        o.frequency.setValueAtTime(hz, t + i * 0.09);
        g.gain.setValueAtTime(0.0001, t + i * 0.09);
        g.gain.exponentialRampToValueAtTime(0.12, t + i * 0.09 + 0.04);
        g.gain.exponentialRampToValueAtTime(0.0001, t + i * 0.09 + 1.6);
        o.connect(g).connect(ac.destination);
        o.start(t + i * 0.09);
        o.stop(t + i * 0.09 + 1.7);
      });
      setTimeout(() => ac.close().catch(() => {}), 2400);
    } catch (e) {
      /* sound is optional */
    }
  }

  try {
    if (document.fonts && document.fonts.load) document.fonts.load('700 40px Cinzel').catch(() => {});
  } catch (e) {
    /* the serif fallback is used instead */
  }

  let greeting = null;
  function greet() {
    if (greeting) return greeting;
    greeting = new Promise((resolve) => {
      const cv = document.createElement('canvas');
      cv.setAttribute('aria-hidden', 'true');
      cv.style.cssText = 'position:fixed;inset:0;width:100vw;height:100vh;z-index:9800;pointer-events:none;opacity:0';
      document.body.appendChild(cv);
      const W = (cv.width = Math.max(1, window.innerWidth));
      const H = (cv.height = Math.max(1, window.innerHeight));
      const ctx = cv.getContext('2d');
      const reduce = window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;
      const typeEnd = G_FADE_IN + GREET_LINES[0].length * G_CHAR + G_GAP + GREET_LINES[1].length * G_CHAR;
      const total = typeEnd + G_HOLD + G_FADE_OUT;
      const start = performance.now();
      let sounded = false;
      const frame = (now) => {
        const el = now - start;
        if (el >= total) {
          cv.remove();
          greeting = null;
          resolve();
          return;
        }
        const fadeOut = el > typeEnd + G_HOLD ? 1 - (el - typeEnd - G_HOLD) / G_FADE_OUT : 1;
        cv.style.opacity = String(Math.max(0, Math.min(1, el / G_FADE_IN, fadeOut)));
        ctx.setTransform(1, 0, 0, 1, 0, 0);
        ctx.fillStyle = '#03060d';
        ctx.fillRect(0, 0, W, H);
        const vg = ctx.createRadialGradient(W / 2, H / 2, Math.min(W, H) * 0.08, W / 2, H / 2, Math.max(W, H) * 0.7);
        vg.addColorStop(0, 'rgba(70,170,255,0.20)');
        vg.addColorStop(1, 'rgba(0,0,0,0)');
        ctx.fillStyle = vg;
        ctx.fillRect(0, 0, W, H);
        // How many letters of each line are typed so far.
        const t0 = el - G_FADE_IN;
        const n1 = reduce ? GREET_LINES[0].length : Math.max(0, Math.min(GREET_LINES[0].length, Math.floor(t0 / G_CHAR)));
        const t1 = t0 - GREET_LINES[0].length * G_CHAR - G_GAP;
        const n2 = reduce ? GREET_LINES[1].length : Math.max(0, Math.min(GREET_LINES[1].length, Math.floor(t1 / G_CHAR)));
        if (!sounded && n2 > 0) {
          sounded = true;
          chime();
        }
        const size = Math.max(17, Math.min(W * 0.036, 46));
        ctx.font = `700 ${size}px ${GREET_FONT}`;
        if ('letterSpacing' in ctx) ctx.letterSpacing = `${(size * 0.08).toFixed(1)}px`;
        ctx.textBaseline = 'middle';
        ctx.textAlign = 'left';
        const lh = size * 1.6;
        const shown = [n1, n2];
        GREET_LINES.forEach((ln, i) => {
          const part = ln.slice(0, shown[i]);
          const w = ctx.measureText(ln).width;
          const x = (W - Math.min(w, W * 0.94)) / 2;
          const y = H / 2 + (i - 0.5) * lh;
          // The second line is the boast: a stronger glow.
          ctx.shadowColor = '#4fc3ff';
          ctx.shadowBlur = size * (i ? 0.75 : 0.45);
          ctx.fillStyle = i ? '#ffffff' : '#dcefff';
          ctx.fillText(part, x, y, W * 0.94);
          ctx.shadowBlur = 0;
          const typing = shown[i] > 0 && shown[i] < ln.length;
          const lastIdle = i === 1 && shown[1] === ln.length && Math.floor(el / 380) % 2 === 0;
          if (typing || lastIdle) {
            ctx.fillStyle = '#7fd8ff';
            ctx.fillRect(x + Math.min(ctx.measureText(part).width, W * 0.94) + 3, y - size * 0.55, Math.max(2, size * 0.06), size * 1.1);
          }
        });
        requestAnimationFrame(frame);
      };
      requestAnimationFrame(frame);
    });
    return greeting;
  }

  if (/[?&]honored=1\b/.test(location.search)) window.addEventListener('load', () => setTimeout(() => greet(), 800));

  // ---- 2. Six Eyes ----------------------------------------------------------------------------------------------
  const DURATION = 12000;
  const COOLDOWN = 8000;

  let veil = null,
    hud = null,
    btn = null;

    if (veil) return;
    const s = document.createElement('style');

  function say(text, ms = 1800) {
    st.msg = text;
    st.msgUntil = performance.now() + ms;
  }

  function toggle() {
    const sc = getScene();
    if (!hasSixEyes(lookOf(sc))) return;
    const now = performance.now();
    if (st.active) {
      st.active = false;
      st.readyAt = now + COOLDOWN;
      return;
    }
    if (now < st.readyAt) {
      say(`👁 The Six Eyes are resting… ${Math.ceil((st.readyAt - now) / 1000)}s`);
      return;
    }
    st.active = true;
    st.until = now + DURATION;
    chime();
  }

  // ---- what the Six Eyes can see ---------------------------------------------------------------------------------
  // Returns { here: [{kind, icon, label, x, y, k, wall}], away: {finger, shadow, npc: [names], otherTown} }.
  function sense(sc) {
    const here = [];
    const away = { finger: 0, shadow: 0, npc: [], otherTown: 0 };
    const area = sc.area.id;
    const ins = sc.inside;
    const A = window.LQAchievements;
    const building = (name) => sc.area.buildings.find((b) => b.name === name);
    // Seen from outdoors, things inside a building glow over the middle of its roof.
    const roof = (name, dx) => {
      const b = building(name);
      return b ? { x: (b.x + b.w / 2) * 32 + dx, y: (b.y + b.h / 2) * 32, k: 1 } : null;
    };
    const collectibles = (kind, icon, list, isLive, spot, dx) => {
      for (const it of list) {
        if (!isLive(it)) continue;
        if (it.area !== area) {
          away[kind]++;
          continue;
        }
        let p = null;
        if (ins) p = spot(it.id); // indoors: only what lies in this room
        else p = it.in ? roof(it.in, dx) : { x: it.x * 32 + 16, y: it.y * 32 + 16, k: 1 };
        if (!p) {
          away[kind]++;
          continue;
        }
        here.push({ kind, icon, label: !ins && it.in ? it.in : '', x: p.x, y: p.y, k: p.k || 1, wall: !ins && !!it.in });
      }
    };
    const F = window.LQFingers;
    if (F && A && !A.isUnlocked('sukuna-fingers'))
      collectibles('finger', '👆', F.list, (f) => !F.isFound(f.id), (id) => F.spot(id), -12);
    const TS = window.LQTenShadows;
    if (TS && TS.isStarted())
      collectibles('shadow', '🌑', TS.list, (s) => TS.isTold(s.id) && !TS.isFound(s.id), (id) => TS.spot(id), 12);

    // Villagers to talk to, with the reason (first reason wins).
    const people = new Map();
    const want = (id, why) => id && !people.has(id) && people.set(id, why);
    if (TS && A && !A.isUnlocked('ten-shadows')) {
      if (!TS.isStarted()) want('librarian', 'Ten Shadows');
      else
        for (const s of TS.list)
          if (!TS.isTold(s.id) && !TS.isFound(s.id) && !(s.needs && TS.count() < s.needs)) want(s.npc, 'shadow clue');
    }
    let placeGoal = null;
    try {
      const step = window.LQV8 && window.LQV8.currentStep ? window.LQV8.currentStep() : null;
      if (step && step.kind === 'npc') want(step.npc, 'quest');
      if (step && /^(enter|floor|inspect)$/.test(step.kind) && step.building) placeGoal = { building: step.building };
    } catch (e) {
