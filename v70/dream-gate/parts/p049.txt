        function transactBuy(shop, id, qty) {
          const sc = scene(),
            s = state(),
            def = catalog[id],
            g = game();
          if (!sc || !s || !g) return { ok: false, message: 'The game is not ready yet.' };
          if (busy) return { ok: false, message: 'A transaction is already being processed.' };
          const problem = checkShop(shop);
          if (problem) return { ok: false, message: problem };
          if (!def || !def.shops.includes(shop))
            return { ok: false, message: 'This item is not sold at this shop.' };
          const n = amount(qty);
          if (!Number.isFinite(n) || n > 10)
            return { ok: false, message: 'You can only buy 1–10 items at a time.' };
          if (def.effect === 'room' && n !== 1)
            return { ok: false, message: 'You can only rent one room at a time.' };
          const sig = `${shop}:${id}:${n}`;
          if (lastPurchase.signature === sig && Date.now() - lastPurchase.at < 700)
            return {
              ok: false,
              message: 'Your previous transaction was received. Please wait a moment before buying again.',
            };
          const total = def.price * n;
          if (!validInt(total) || !validInt(g.money) || g.money < total)
            return {
              ok: false,
              message: `You need $${Math.max(0, total - g.money)} more. Complete quests to earn money.`,
            };
          if (def.unique && count(id) > 0)
            return { ok: false, message: 'You already own this item.' };
          if (def.unique && n !== 1)
            return { ok: false, message: 'Upgrade items can only be bought one at a time.' };
          if (def.category !== 'service' && usedSpace() + n > capacity())
            return {
              ok: false,
              message: `Your bag only has ${Math.max(0, capacity() - usedSpace())} free slots. Use some items or buy a backpack.`,
            };
          busy = true;
          try {
            g.money -= total;
            if (def.category === 'service' && def.effect === 'room') {
              s.flags.roomUntilDay = g.day;
              s.flags.roomShop = shop;
              s.flags.roomName = id;
            } else if (def.category === 'service' && def.effect === 'mail') {
              s.flags.pendingMail = (Number(s.flags.pendingMail) || 0) + n;
            } else if (def.category !== 'service') {
              inventory()[id] = count(id) + n;
            }
            if (def.effect === 'lesson') s.unlocked[def.lesson] = true;
            log('purchase', -total, `${def.name} ×${n}`, { shop, item: id, qty: n });
            lastPurchase = { signature: sig, at: Date.now() };
            save();
            return {
              ok: true,
              message: `✅ Bought ${def.name} ×${n} at ${shop}. -$${total}. “Could I have a receipt, please?”`,
            };
          } finally {
            busy = false;
          }
        }
        function bankAction(kind, raw) {
          const c = context(),
            g = game(),
            s = state();
          if (!g || !s) return 'The game is not ready yet.';
          if (!canShop('Bank')) return 'You must go to the Bank to deposit or withdraw money.';
          const n = Number(raw);
          if (!validInt(n, 1, 1000000)) return 'Enter a whole amount from 1 to 1,000,000.';
          if (kind === 'deposit') {
            if (!validInt(g.money) || g.money < n) return 'You do not have enough cash in your wallet.';
            if (s.bank + n > 1000000000) return 'This exceeds the account limit.';
            g.money -= n;
            s.bank += n;
            log('deposit', -n, `Savings deposit $${n}`);
          } else if (kind === 'withdraw') {
            if (s.bank < n || g.money + n > 1000000000)
              return 'Insufficient balance or over the safe limit.';
            s.bank -= n;
            g.money += n;
            log('withdraw', n, `Savings withdrawal $${n}`);
          } else return 'Invalid command.';
          save();
          return `🏦 Success! Wallet $${g.money}; savings $${s.bank}. Your total assets are unchanged because money just moved between two accounts.`;
        }

        function aiShopping(text) {
          const t = norm(text),
            ctx = context(),
            nearShop = ctx.inside && shops[ctx.inside] ? ctx.inside : null;
          const m = t.match(/(?:mua|buy|order|purchase)\s+(?:cho toi|me|a|an|mot|one)?\s*(.*)/);
          if (m) {
            const id = Object.entries(catalog).find(
              ([key, it]) =>
                norm(it.name).includes(m[1]) ||
                norm(m[1]).includes(norm(it.name).replace(/^[^\p{L}]+/u, '')) ||
                key === norm(m[1]).replace(/\s+/g, '_'),
            )?.[0];
            if (id) {
              const it = catalog[id],
                where = nearShop || it.shops[0];
              chosenShop = where;
              show('shop', where);
              return `🛒 ${it.name} costs $${it.price}, available at ${it.shops.join(', ')}. ${canShop(where) ? 'Pick the item and press Buy to confirm the purchase.' : 'Go to the shop first; the AI never deducts money by itself.'}`;
            }
          }
          if (
            /(toi co|i have|ngan sach|budget|afford|mua duoc gi)/.test(t) &&
            /(\$|\b\d+\b|mua|buy|budget)/.test(t)
          ) {
            const n = Number(t.match(/\d+/)?.[0]) || money();
            const results = Object.values(catalog)
              .filter((x) => x.price <= n)
              .slice(0, 8)
              .map((x) => `${x.name} $${x.price}`)
              .join(' · ');
            return `💰 With $${n} you could consider: ${results}. Cash in your wallet: $${money()}. Open the Shop (B) to check prices and confirm a purchase.`;
          }
          if (/(kiem tien|lam viec|earn money|find a job)/.test(t))
            return '💼 Earn money from NPC quests or a once-a-day job at Sunny Cafe, Library, Post Office or Smithy. Go to the matching building and open the Economy window (B), then Inventory → Work.';
          if (/(tiet kiem|ngan hang|bank|deposit|withdraw|saving)/.test(t)) {
            show('bank');
            return `🏦 You have $${money()} in your wallet and $${state()?.bank || 0} in savings. Go to the Bank to deposit/withdraw. Saving does not create extra virtual money.`;
          }
          if (/(gia|bao nhieu|how much|price|cheaper|discount)/.test(t)) {
            const sample = Object.values(catalog)
              .slice(0, 4)
              .map((x) => `${x.name}: $${x.price}`)
              .join(', ');
            return `🛍️ Prices are fixed in shops, for example: ${sample}. Try asking “How much does it cost?” or “Do you have anything cheaper?”`;
          }
          return null;
        }

        function close() {
          const x = shell();
          if (!x || x.hidden) return;
          x.hidden = true;
          el('l11-menu-btn')?.setAttribute('aria-expanded', 'false');
          (lastFocus?.isConnected ? lastFocus : el('l11-menu-btn'))?.focus?.({
            preventScroll: true,
          });
        }
        function existingIsOpen() {
          return [
            'chat',
            'helper',
            'journal',
            'quests',
            'settings',
            'briefing',
            'v10-panel',
            'rpg-panel',
            'v8-panel',
            'ai-panel',
            'lq-guide',
            'lq-levels',
            'lq-voice',
          ].some((id) => {
            const x = el(id);
            if (!x) return false;
            if (x.hidden) return false;
            if (
              id === 'chat' ||
              id === 'helper' ||
              id === 'journal' ||
              id === 'quests' ||
              id === 'settings' ||
              id === 'briefing'
            )
              return getComputedStyle(x).display !== 'none';
            return true;
          });
        }
        function open(name = 'maps') {
          if (!shell()) return;
          if (existingIsOpen()) return;
          lastFocus = document.activeElement;
          shell().hidden = false;
          el('l11-menu-btn')?.setAttribute('aria-expanded', 'true');
          page(name);
          if (name === 'maps') el('l11-nav').scrollLeft = 0;
          el('l11-close')?.focus();
        }
        function toggle() {
          if (!shell().hidden) close();
          else open('maps');
        }
        function adapt(type, tab) {
          close();
          if (type === 'v10') {
            window.LQ10_ECONOMY?.show(tab || 'shop');
            return;
          }
          if (type === 'v8') {
            el('v8-menu-btn')?.click();
            if (tab && tab !== 'quests')
              el('v8-panel')?.querySelector(`[data-v8-tab="${tab}"]`)?.click();
            return;
          }
          if (type === 'rpg') {
            el('rpg-open')?.click();
            if (tab) {
              const node = [...(el('rpg-tabs')?.querySelectorAll('button') || [])].find(
                (b) =>
                  b.dataset.tab === tab ||
                  b.dataset.rpgTab === tab ||
                  b.textContent.toLowerCase().includes(tab),
              );
              node?.click();
            }
            return;
          }
        }
