        function lqNpcExtractJson(value) {
          if (value && typeof value === 'object') return value;
          if (typeof value !== 'string') return null;
          let text = value.trim();
          for (let round = 0; round < 3; round++) {
            text = text
              .replace(/^```(?:json|javascript|js|text)?\s*\n?/i, '')
              .replace(/\n?```\s*$/, '')
              .trim();

              const parsed = JSON.parse(text);
              if (typeof parsed === 'string') {
                text = parsed.trim();
                continue;
              }
              return parsed;
            } catch {}
            const begin = text.indexOf('{'),
              end = text.lastIndexOf('}');
            if (begin >= 0 && end > begin) {

                return parsed;

          return null;
        }
        function lqNpcSafeText(value, fallback = '', preserveText = false) {
          let raw = typeof value === 'string' ? value : String(value ?? '');
          const orig = raw.trim();
          if (!orig || (preserveText && /```|~~~/.test(orig))) return fallback;
          let parsed = lqNpcExtractJson(orig);
          if (parsed && typeof parsed === 'object') {
            if (preserveText) return fallback;
            const content =
              parsed.reply ?? parsed.greeting ?? parsed.message?.content ?? parsed.text;
            if (typeof content === 'string' && content !== orig)
              return lqNpcSafeText(content, fallback);
            return fallback;
          }
          raw = orig
            .replace(/^```(?:json|text|markdown)?\s*/i, '')
            .replace(/\s*```$/, '')
            .trim();
          // Sometimes the provider emits malformed JSON with a valid reply property.
          const embedded = raw.match(/["']reply["']\s*:\s*"((?:\\.|[^"\\])*)"/i);
          if (embedded) {
            if (preserveText) return fallback;
            try {
              return lqNpcSafeText(JSON.parse('"' + embedded[1] + '"'), fallback);
            } catch {}
          }
          // Do not display code blocks, JSON fields, markup documents or stack traces.
          if (
            /```|~~~|<\s*\/?\s*(?:script|style|html|!doctype|body|div|pre|code)\b|\b(?:SyntaxError|ReferenceError|TypeError|Traceback|at\s+\w+\s*\([^)]*:\d+)\b/i.test(
              raw,
            )
          )
            return fallback;
          if (
            /(?:^|[\r\n])\s*(?:(?:const|let|var|function|class|import|export|return|def|print|if|for|while)\s+[\w$]+\s*(?:[=(;:]|\{)|(?:\{\s*["']?(?:reply|goalDone|correction|role|parts|content)["']?\s*:)|(?:["']?(?:goalDone|affinityDelta|correction)["']?\s*:)|<\/?[A-Za-z][^>]*>)/m.test(
              raw,
            )
          )
            return fallback;
          if (/^\s*[\[{]/.test(raw) && /[\]}]\s*$/.test(raw)) return fallback;
          if (
            /\b(?:console\.log|JSON\.parse|document\.getElementById|=>|```|\{\s*"reply"|\\n\s*"goalDone")/.test(
              raw,
            )
          )
            return fallback;
          raw = raw
            .replace(/^\s*(?:reply|greeting|answer)\s*:\s*/i, '')
            .replace(/^['“”"]+|['“”"]+$/g, '');
          raw = raw
            .replace(/\*\*([^*]+)\*\*/g, '$1')
            .replace(/__([^_]+)__/g, '$1')
            .replace(/`([^`]+)`/g, '$1')
            .trim();
          raw = raw
            .replace(/\r\n?/g, '\n')
            .replace(/\n{3,}/g, '\n\n')
            .trim();
          if (
            !raw ||
            /^[\[{]/.test(raw) ||
            /[{}][\s]*["']?(?:goalDone|reply|correction)["']?\s*[:=]/i.test(raw)
          )
            return fallback;
          return preserveText ? orig : raw.slice(0, 380);
        }
