  /* V35: a shared geometric model for rendering, water collision and bridge decks.
     Crossing is possible only on the one tile-wide visible bridge platform. */
  const WV_BRIDGES=[23,45,51];
  const wvRiverX=y=>44+Math.round(3*Math.sin(y*.15)+2*Math.sin(y*.34));
  const WV_WORLD_GEOMETRY=Object.freeze({
    width:88,height:60,tile:T,
    bridgeRows:Object.freeze([23,45,51]),
    moonveil:Object.freeze({x:79,y:55.6,rx:2.55,ry:1.72}),
    river(y){
      const x=wvRiverX(y)*T+16;
      const half=(y<14?8:y<23?6:4)*T+((y>7&&y<16)?(15-y)*20:0);
      return {x,half};
    },
    isPond(x,y){
      const p=this.moonveil;
      return ((x+.5-p.x)/p.rx)**2+((y+.5-p.y)/p.ry)**2<.84;
    },
    creekY(x){return 56.4+.47*Math.sin((x-50)*.18);},
    isCreek(x,y){
      const start=this.river(56).x/T+3;
      return x+.5>=start&&x+.5<=this.moonveil.x&&
        Math.abs(y+.5-this.creekY(x+.5))<.76;
    },
    isRiverTile(x,y){
      const s=this.river(y+.5);
      return Math.abs((x+.5)*T-s.x)<=s.half-T*.09;
    },
    hasDeck(x,y){return this.bridgeRows.includes(y)&&this.isRiverTile(x,y);},
    isWaterBlocked(x,y){
      return (this.isRiverTile(x,y)&&!this.hasDeck(x,y))||this.isPond(x,y)||this.isCreek(x,y);
    },
    bridgeEndpoints(row){
      const s=this.river(row+.5);
      return {left:s.x-s.half-20,right:s.x+s.half+20,y:(row+.5)*T};
    }
  });
