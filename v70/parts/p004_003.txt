            if (!step.words.some((w) => input.includes(normalize(w)))) return false;
            // Protect against a repeated keypress or duplicate NPC callback.
            const id = 'npc:' + step.npc + ':' + progress(q),
              at = Date.now();
            if (onEvent.lastId === id && at - (onEvent.lastAt || 0) < 900) return false;
            onEvent.lastId = id;
            onEvent.lastAt = at;
          } else {
            if (step.building !== payload.building) return false;
            if (step.floor !== undefined && step.floor !== (Number(payload.floor) || 0))
              return false;
            if (step.objects && !step.objects.includes(payload.object)) return false;
            if (step.area) {
              let a = payload.area || identifyArea(scene);
              if (step.area === 'minerals' && !isMinerals(a)) return false;
            }
          }
          state.progress[q.id] = progress(q) + 1;
          store();
          sound('click');
          if (progress(q) >= q.steps.length) reward(q);
          else {
            markActivity('Quest ' + q.title + ': ' + progress(q) + '/' + q.steps.length);
            toast('✅ Step completed ' + progress(q) + '/' + q.steps.length);
            renderQuest();
          }
          return true;
        }

        function destinationGate(scene, label) {
          const entry = scene.area.buildings.find((b) => scene.area.portals?.[b.name]);
          if (entry)
            return {
              x: entry.door * 32 + 16,
              y: (entry.y + entry.h) * 32 + 16,
              name: 'Gate at ' + entry.name,
            };
          const warp = scene.area.warps?.[0];
          if (warp)
            return {
              x: (warp.x + warp.w / 2) * 32,

        function target() {
          const q = currentQuest();
          if (done(q)) return null;
          const step = nextStep();
          if (!step || step.noMark) return null;
          const scene = getScene();
          if (!scene) return null;
          if (step.kind === 'enter' || step.kind === 'floor' || step.kind === 'inspect') {
            const b = scene.area.buildings.find((x) => x.name === step.building);
            if (!b) return destinationGate(scene, step.building);
            return { x: b.door * 32 + 16, y: (b.y + b.h) * 32 + 16, name: b.name };
          }
          if (step.kind === 'npc') {
            const n = scene.npcs.find((x) => x.def.id === step.npc);
            if (!n) return destinationGate(scene, 'NPC ' + step.npc);
            if (n.inside !== !!scene.inside || !n.sprite.visible)
              return { remote: n.def.name + ' (on their daily schedule)', area: 'near' };
            return { x: n.sprite.x, y: n.sprite.y, name: n.def.name };
          }
          return null;
        }

        function paintWeather(now) {
          const c = $('v8-weather-canvas');
          if (!c) return;
          if (!preferences.weather || preferences.performance === 'low') {
            const cx = c.getContext('2d');
            cx?.clearRect(0, 0, c.width, c.height);
            return;
          }
          const sc = getScene();
          if (!sc || sc.inside || weatherType === 'clear') {
            c.getContext('2d')?.clearRect(0, 0, c.width, c.height);
            return;
          }
          const fps = preferences.performance === 'high' ? 30 : 20;
          if (now - fxFrame < 1000 / fps) return;
          fxFrame = now;
          const ctx = c.getContext('2d');
          if (!ctx) return;
          ctx.clearRect(0, 0, c.width, c.height);
          const count =
            weatherType === 'rain' ? (preferences.performance === 'high' ? 65 : 36) : 18;
          while (fxParticles.length < count)
            fxParticles.push({
              x: Math.random() * c.width,
              y: Math.random() * c.height,
              speed: 1 + Math.random() * 2.4,
            });
          if (fxParticles.length > count) fxParticles.length = count;
          ctx.strokeStyle =
            weatherType === 'rain' ? 'rgba(164,209,250,0.45)' : 'rgba(234,244,255,0.16)';
          ctx.lineWidth = weatherType === 'rain' ? 1.2 : 8;
          for (const p of fxParticles) {
            p.x += weatherType === 'rain' ? 2 : 0.3;
            p.y += weatherType === 'rain' ? 6 * p.speed : 0.2 * p.speed;
            if (p.y > c.height) {
              p.y = -15;
              p.x = Math.random() * c.width;
            }
            if (p.x > c.width) p.x = 0;
            ctx.beginPath();
            ctx.moveTo(p.x, p.y);
            ctx.lineTo(
              p.x + (weatherType === 'rain' ? 3 : 17),
              p.y + (weatherType === 'rain' ? 14 : 7),
            );
            ctx.stroke();
          }
        }

            lastFoot = now;

          s.ledger.unshift({
            id: tx,

          });

        function context() {
          const sc = scene();
          if (!sc) return { inside: '', near: [], area: '', day: 1 };
          const p = sc.player;
          return {
            inside: sc.inside?.b?.name || '',
            area: sc.area?.id || '',
            day: sc.store.game.day,
            near: (sc.npcs || [])
              .filter(
                (n) =>
                  n.sprite?.visible &&
                  Boolean(n.inside) === Boolean(sc.inside) &&
                  Math.hypot(n.sprite.x - p.x, n.sprite.y - p.y) < 74,
              )
              .map((n) => n.def.id),
          };
        }
        function canShop(shop) {
          const c = context();
          if (c.inside === shop) return true;
          const sc = scene();
          if (!sc || sc.inside) return false;
          if (
            shop === 'Rail Station' &&
            ['english-town', 'mining-town'].includes(sc.area?.id) &&
            sc.player.x >= 4 * 32 &&
            sc.player.x < 56 * 32 &&
            sc.player.y >= 47 * 32 &&
            sc.player.y < 72 * 32
          )
            return true;
          const b = sc.area?.buildings?.find((x) => x.name === shop);
          if (!b) return false;
          return (
            Math.hypot(sc.player.x - (b.door * 32 + 16), sc.player.y - ((b.y + b.h) * 32 + 12)) < 70
          );
        }
        function checkShop(shop) {
          if (!shops[shop]) return 'Shop not found.';
          if (!canShop(shop))
            return `You need to be at the entrance of or inside ${shop} to trade. Press M for the map or ask the AI for directions.`;
          return '';
        }

        function transactBuy(shop, id, qty) {
          const sc = scene(),
            s = state(),
            def = catalog[id],
            g = game();
          if (!sc || !s || !g) return { ok: false, message: 'The game is not ready yet.' };
          if (busy) return { ok: false, message: 'A transaction is already being processed.' };
          const problem = checkShop(shop);
          if (problem) return { ok: false, message: problem };
          if (!def || !def.shops.includes(shop))
            return { ok: false, message: 'This item is not sold at this shop.' };
          const n = amount(qty);
          if (!Number.isFinite(n) || n > 10)
            return { ok: false, message: 'You can only buy 1–10 items at a time.' };
          if (def.effect === 'room' && n !== 1)
            return { ok: false, message: 'You can only rent one room at a time.' };
          const sig = `${shop}:${id}:${n}`;
          if (lastPurchase.signature === sig && Date.now() - lastPurchase.at < 700)
            return {
              ok: false,
              message: 'Your previous transaction was received. Please wait a moment before buying again.',
            };
          const total = def.price * n;
          if (!validInt(total) || !validInt(g.money) || g.money < total)
            return {
              ok: false,
              message: `You need $${Math.max(0, total - g.money)} more. Complete quests to earn money.`,
            };
          if (def.unique && count(id) > 0)
            return { ok: false, message: 'You already own this item.' };
          if (def.unique && n !== 1)
            return { ok: false, message: 'Upgrade items can only be bought one at a time.' };
          if (def.category !== 'service' && usedSpace() + n > capacity())
            return {
              ok: false,
              message: `Your bag only has ${Math.max(0, capacity() - usedSpace())} free slots. Use some items or buy a backpack.`,
            };
          busy = true;
          try {
            g.money -= total;
            if (def.category === 'service' && def.effect === 'room') {
              s.flags.roomUntilDay = g.day;
              s.flags.roomShop = shop;
              s.flags.roomName = id;
            } else if (def.category === 'service' && def.effect === 'mail') {
              s.flags.pendingMail = (Number(s.flags.pendingMail) || 0) + n;
            } else if (def.category !== 'service') {
              inventory()[id] = count(id) + n;
            }
            if (def.effect === 'lesson') s.unlocked[def.lesson] = true;
            log('purchase', -total, `${def.name} ×${n}`, { shop, item: id, qty: n });
            lastPurchase = { signature: sig, at: Date.now() };
            save();
            return {
              ok: true,
              message: `✅ Bought ${def.name} ×${n} at ${shop}. -$${total}. “Could I have a receipt, please?”`,
            };
          } finally {
            busy = false;
          }
        }
        function bankAction(kind, raw) {
          const c = context(),
            g = game(),
            s = state();
          if (!g || !s) return 'The game is not ready yet.';
          if (!canShop('Bank')) return 'You must go to the Bank to deposit or withdraw money.';
          const n = Number(raw);
          if (!validInt(n, 1, 1000000)) return 'Enter a whole amount from 1 to 1,000,000.';
          if (kind === 'deposit') {
            if (!validInt(g.money) || g.money < n) return 'You do not have enough cash in your wallet.';
            if (s.bank + n > 1000000000) return 'This exceeds the account limit.';
            g.money -= n;
            s.bank += n;
            log('deposit', -n, `Savings deposit $${n}`);
          } else if (kind === 'withdraw') {
            if (s.bank < n || g.money + n > 1000000000)
              return 'Insufficient balance or over the safe limit.';
            s.bank -= n;
            g.money += n;
            log('withdraw', n, `Savings withdrawal $${n}`);
          } else return 'Invalid command.';
          save();
          return `🏦 Success! Wallet $${g.money}; savings $${s.bank}. Your total assets are unchanged because money just moved between two accounts.`;
        }

        }
