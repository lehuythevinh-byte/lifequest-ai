  function actorPath(sc,actor,tx,ty){
    const area=sc.area,sx=Math.floor(actor.x/T),sy=Math.floor(actor.y/T);
    // Existing findPath() returns the blocked destination on failure, hence
    // Tech-Co uses its own 4-neighbour bounded BFS with explicit no-path result.

    const key=(x,y)=>y*area.w+x,from=new Int32Array(area.w*area.h).fill(-2);
    const q=[[sx,sy]];from[key(sx,sy)]=-1;
    let found=false;
    for(let h=0;h<q.length;h++){
      const [x,y]=q[h];if(x===tx&&y===ty){found=true;break;}
      for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){
        const nx=x+dx,ny=y+dy;if(!inBounds(area,nx,ny)||from[key(nx,ny)]!==-2)continue;
