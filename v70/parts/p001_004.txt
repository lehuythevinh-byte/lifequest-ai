    st.active = false;
    st.lastOrigin = undefined;
    st.back.clear();
    st.front.clear();
    try {
      sc.applyRankLook();
    } catch (e) {
      /* the scene is being rebuilt */
    }
  }

  function frame(sc, time, delta) {
    const st = sc.__lqShadow;
    if (!st) return;
    st.back.clear();
    st.front.clear();
    const look = sc.store && sc.store.game && sc.store.game.rpg && sc.store.game.rpg.look;
    const p = sc.player;
    const able = !!(look && look.shadow && p && p.visible);
    if (!able) {
      state.riding = false;
      st.t = 0;
      if (st.active) release(sc, st);
      return;
    }
    const target = state.riding ? 1 : 0;
    const step = Math.min(1, (delta || 16) / 450);
    st.t = target > st.t ? Math.min(target, st.t + step) : Math.max(target, st.t - step);
    if (st.t <= 0) {
      if (st.active) release(sc, st);
      return;
    }
    st.active = true;
    if (st.look !== look) {
      st.look = look;
      st.key = window.LQCharacter.ensure(sc, Object.assign({}, look, { riding: true }));
    }
    // The scene picks the sprite frame from playerKey each frame, so keep it on the riding set while the dog is out.
    sc.playerKey = st.key;
    const e = st.t * st.t * (3 - 2 * st.t);
    const s = p.scaleX || 1;
    // applyRankLook resets the origin; detect that and re-base (same trick as aura.js).
    if (st.lastOrigin === undefined || Math.abs(p.originY - st.lastOrigin) > 1e-6) st.baseOrigin = p.originY;
    const moving = !!sc.pMoving;
    const bob = moving ? Math.abs(Math.sin(time * 0.014)) * 0.9 : Math.sin(time * 0.002) * 0.5;
    // The side-view dog is smaller, so the rider sits lower on its back there.
    const dirNow = (sc.pFace && sc.pFace.dir) || 'down';
    const lift = dirNow === 'up' || dirNow === 'down' ? LIFT : LIFT_SIDE;
    st.lastOrigin = st.baseOrigin + (lift * e + bob) / (p.height || 50);
    p.setOrigin(0.5, st.lastOrigin);
    const face = sc.pFace || {};
    const o = {
      x: p.x,
      y: p.y + 9 * s + (1 - e) * 12 * s,
      s,
      dir: face.dir || 'down',
      flip: !!face.flip,
      moving,
      time,
      alpha: e,
      e: st.t,
    };
    st.back.fillStyle(0x000000, 0.28 * e);
    st.back.fillEllipse(p.x, p.y + 10 * s, 38 * s, 7 * s);
    summonFx(st.back, o);
    drawDog(st.back, Object.assign({ part: 'back' }, o));
    drawDog(st.front, Object.assign({ part: 'front' }, o));
  }

  function attach(sc) {
    if (sc.__lqShadow) return;
    sc.__lqShadow = {
      back: sc.add.graphics().setDepth(9.9),
      front: sc.add.graphics().setDepth(10.1),
      t: 0,
      active: false,
    };
    sc.events.on('update', (time, delta) => frame(sc, time, delta));
    sc.events.once('shutdown', () => {
      sc.__lqShadow = null;
    });
  }

  function toggle() {
    const sc = getScene();
    const look = sc && sc.store && sc.store.game && sc.store.game.rpg && sc.store.game.rpg.look;
    if (!look || !look.shadow) return;
    state.riding = !state.riding;
  }

  // A "tap" is Ctrl pressed and released with nothing else in between (so Ctrl+C, Ctrl+click etc. do not count).
  let ctrlAlone = false;
  const typing = () => {
    const a = document.activeElement;
    return !!a && (/^(INPUT|TEXTAREA|SELECT)$/.test(a.tagName) || a.isContentEditable);
  };
  window.addEventListener(
    'keydown',
    (e) => {
      if (e.key === 'Control') {
        if (!e.repeat) ctrlAlone = !e.altKey && !e.shiftKey && !e.metaKey;
      } else ctrlAlone = false;
    },
    true,
  );
  window.addEventListener('mousedown', () => (ctrlAlone = false), true);
  window.addEventListener('blur', () => (ctrlAlone = false));
  window.addEventListener(
    'keyup',
    (e) => {
      if (e.key !== 'Control' || !ctrlAlone) return;
      ctrlAlone = false;
      if (!typing()) toggle();
    },
    true,
  );

  setInterval(() => {
    const sc = getScene();
    if (sc && sc.add && sc.events && sc.player && !sc.__lqShadow) {
      try {
        attach(sc);
      } catch (err) {
        console.warn('shadow attach failed', err);
      }
    }
  }, 1000);

  window.LQShadow = { toggle, isRiding: () => state.riding, drawDog };
})();

  // While the eyes are open the world sinks under a dark layer (depth DIM). Only the player and what the Six Eyes

  // Lift (or put back) the game objects that should shine through the dark layer.
  function lift(sc, s, on, npcs) {

    for (const n of want) set(n.sprite, D.npc), set(n.label, D.npcLabel);
    s.lifted = want;

  // A pixel-crisp outline: four copies of the sprite, filled with one colour, nudged one pixel each way.
  function outline(sc, s, spr, color, alpha) {
    const off = Math.max(1, spr.scaleX || 1);
    for (const [dx, dy] of [[-off, 0], [off, 0], [0, -off], [0, off]]) {
      let img = s.glows[s.gi++];
      if (!img) s.glows.push((img = sc.add.image(0, 0, spr.texture.key).setDepth(D.glow)));
      img
        .setTexture(spr.texture.key, spr.frame.name)
        .setOrigin(spr.originX, spr.originY)
        .setScale(spr.scaleX, spr.scaleY)
        .setFlipX(spr.flipX)
        .setPosition(spr.x + dx, spr.y + dy)
        .setAlpha(alpha)
        .setVisible(true);
      fillTint(img, color);
    }
  }

  // Solid-colour tint: Phaser 4 uses setTint + FILL mode (setTintFill was removed); Phaser 3 has setTintFill.
  function fillTint(img, color) {
    const P = window.Phaser;
    if (img.setTintMode && P && P.TintModes) img.setTint(color).setTintMode(P.TintModes.FILL);
    else img.setTintFill(color);
  }

  // Soft glow with no hard edge: stacked faint discs.
  function halo(g, x, y, r, color, a) {
    for (let i = 0; i < 6; i++) {
      g.fillStyle(color, a);
      g.fillCircle(x, y, r * (1 - i * 0.14));
    }
  }

  function drawWorld(sc, s, seen, time, e) {
    const cam = sc.cameras.main;
    const v = cam.worldView;
    const z = cam.zoom || 1;
    const pulse = 0.5 + 0.5 * Math.sin(time * 0.005);
    s.dim.fillStyle(0x030812, DIM_ALPHA * e);
    s.dim.fillRect(v.x - 4, v.y - 4, v.width + 8, v.height + 8);
    const g = s.g;
    const top = s.top.setAlpha(e);
    s.gi = 0;
    const px = sc.player.x,
      py = sc.player.y;
    const onScreen = (t) => t.x > v.x && t.x < v.right && t.y > v.y && t.y < v.bottom;
    const inView = [],
      outView = [];
    for (const t of seen.here) {
      t.d = Math.hypot(t.x - px, t.y - py);
      (onScreen(t) || (t.b && t.b.x * 32 < v.right && (t.b.x + t.b.w) * 32 > v.x && t.b.y * 32 < v.bottom && (t.b.y + t.b.h) * 32 > v.y) ? inView : outView).push(t);
    }
    // The player keeps a faint blue rim so Gojo still stands out.
    outline(sc, s, sc.player, 0x8fe8ff, 0.55 * e);
    lift(sc, s, true, inView.filter((t) => t.npc).map((t) => t.npc));
    inView.forEach((t, i) => {
      const c = COLORS[t.kind];
      const k = t.k;
      if (t.kind === 'npc') {
        halo(g, t.npc.sprite.x, t.npc.sprite.y - 14 * k, 26 * k, c, 0.035 * e);
        outline(sc, s, t.npc.sprite, c, (0.7 + 0.3 * pulse) * e);
      } else if (t.kind === 'place' && t.b) {
        // The whole building is traced in light.
        const x = t.b.x * 32,
          y = t.b.y * 32,
          w = t.b.w * 32,
          h = t.b.h * 32;
        [[7, 0.12], [4, 0.25], [1.5, 0.9]].forEach(([lw, a]) => {
          g.lineStyle(lw, c, a * (0.75 + 0.25 * pulse) * e);
          g.strokeRect(x, y, w, h);
        });
      } else if (t.kind === 'place') {
        g.lineStyle(2, c, (0.6 + 0.4 * pulse) * e);
        g.strokeEllipse(t.x, t.y + 6, 34, 12);
        halo(g, t.x, t.y, 22, c, 0.04 * e);
      } else {
        halo(g, t.x, t.y, 30 * k, c, (0.05 + 0.03 * pulse) * e);
        if (t.wall) {
          // Seen through the roof: the object itself, drawn over the building.
          if (t.kind === 'finger' && window.LQFingers && window.LQFingers.draw) window.LQFingers.draw(top, t.x, t.y, 1, time, i);
          if (t.kind === 'shadow' && window.LQTenShadows && window.LQTenShadows.drawMark) window.LQTenShadows.drawMark(top, t.x, t.y, 1, time, i);
        }
      }
    });
    for (let i = s.gi; i < s.glows.length; i++) s.glows[i].setVisible(false);
    // Glowing chevrons on the edge of the view for what lies beyond it (nearest first, no text).
    const m = 22 / z;
    outView
      .sort((a, b) => a.d - b.d)
      .slice(0, MAX_ARROWS)
      .forEach((t) => {
        const c = COLORS[t.kind];
        const cx = Math.min(v.right - m, Math.max(v.x + m, t.x));
        const cy = Math.min(v.bottom - m, Math.max(v.y + m, t.y));
        const ang = Math.atan2(t.y - cy, t.x - cx);
        const chevron = (size, a) => {
          const L = size / z,
            W2 = size * 0.7 / z;
          const tipX = cx + Math.cos(ang) * L,
            tipY = cy + Math.sin(ang) * L;
          const bx = cx - Math.cos(ang) * L * 0.5,
            by = cy - Math.sin(ang) * L * 0.5;
          top.fillStyle(c, a);
          top.fillTriangle(tipX, tipY, bx + Math.cos(ang + Math.PI / 2) * W2, by + Math.sin(ang + Math.PI / 2) * W2, bx + Math.cos(ang - Math.PI / 2) * W2, by + Math.sin(ang - Math.PI / 2) * W2);
        };
        chevron(17, 0.15);
        chevron(13, 0.3);
        chevron(9, 0.95);
      });
  }

  }

      function train(now) {
        const total = 50 + 4 + 50 + 4 + 57;
        const left = Math.round(W / 2 - total / 2);
        const jig = (k) => (speed > 8 && Math.sin(now / 70 + k * 1.7) > 0.7 ? 1 : 0);
        const y = GY - 2;
        carriage(left, y - jig(0), 0, now, -1);
        carriage(left + 54, y - jig(1), 1, now, 2);
        locomotive(left + 108, y - jig(2));
        // smoke
        // Puffs while moving; once stopped, only a thin wisp that fades quickly.
        const moving = speed > 8;
        smokeT += moving ? 1 : 0.12;
        if (smokeT > 7) {
          smokeT = 0;
          smoke.push({ x: left + 108 + 44, y: GY - 37, r: moving ? 2.5 : 1.5, life: 0, max: moving ? 2.4 : 1.1 });
        }
        for (const p of smoke) {
          p.life += 1 / 60;
          p.x -= (speed / 60) * 0.85 + 0.15;
          p.y -= 0.32;
          p.r += 0.07;
        }
        for (let k = smoke.length - 1; k >= 0; k--) if (smoke[k].life > smoke[k].max) smoke.splice(k, 1);
        for (const p of smoke) disc(p.x, p.y, p.r, `rgba(255,255,255,${Math.max(0, 0.75 * (1 - p.life / p.max)).toFixed(2)})`);

        }
      }

      function show(i) {
        li = i;
        const l = lines[i];
        box.classList.add('show');
        enEl.textContent = '';
        viEl.textContent = l.vi;
        viEl.classList.remove('show');

        dots.forEach((d, k) => d.classList.toggle('on', k <= i));
        nextBtn.textContent = i === lines.length - 1 ? "Let's go! ▸" : 'Next ▸';
        if (l.brake && !brake) {
          const D = W + 140;
          brake = { t0: performance.now(), c0: cam, D, T: (2 * D) / CRUISE };
        }
      }
      function finishTyping() {
        if (!typing) return;
        enEl.textContent = typing.text;
        typing = null;
        viEl.classList.add('show');
      }
      function next() {
        if (ended) return;
        if (li < 0) {
          show(0);
          return;
        }
        if (typing) {
          finishTyping();
          return;
        }
        if (li < lines.length - 1) show(li + 1);
        else end();
      }
      function end() {
        if (ended) return;
        ended = true;
        root.classList.remove('on');
        removeEventListener('keydown', onKey, true);
        setTimeout(() => {
          cancelAnimationFrame(raf);
          removeEventListener('resize', size);
          root.remove();
          try {
            if (paused && sc && sc.scene) sc.scene.resume();
          } catch (e) {}
          playing = null;
          resolve();
        }, 650);
      }
      const onKey = (ev) => {
        // The scene owns the keyboard while it plays: the game must not move or open menus behind it.
        ev.preventDefault();
        ev.stopImmediatePropagation();
        if (ev.repeat) return;
        if (ev.key === ' ' || ev.key === 'Enter' || ev.key === 'e' || ev.key === 'E' || ev.key === 'ArrowRight') next();
      };
      addEventListener('keydown', onKey, true);
      root.addEventListener('click', (ev) => {
        if (ev.target.closest('.skip')) {
          end();
          return;
        }
        next();
      });

      function frame(now) {
        if (!document.body.contains(root)) return;
        const dt = Math.min(0.05, (now - last) / 1000);
        last = now;
        if (brake) {
          const p = Math.min(1, (now - brake.t0) / 1000 / brake.T);
          cam = brake.c0 + brake.D * (1 - (1 - p) * (1 - p));
          speed = CRUISE * (1 - p);
        } else cam += speed * dt;
        if (li < 0 && now - t0 > 2500) show(0);
        if (typing) {
          typing.t += dt;
          const n = Math.min(typing.text.length, Math.floor(typing.t * 40));
          if (n !== typing.n) {
            typing.n = n;
            enEl.textContent = typing.text.slice(0, n);
          }
          if (n >= typing.text.length) finishTyping();
        }
        draw(now);
        raf = requestAnimationFrame(frame);
      }
      raf = requestAnimationFrame(frame);
    });
    return playing;
  }

        b.type = 'button';

  function advance() {
    const g = game();
    if (!g || !g.tut) return;
    const list = steps();
    g.tut.step = Math.min(list.length, (g.tut.step | 0) + 1);
    base = null;
    if (g.tut.step >= list.length) finish(false);
    else save();
  }
  function jumpTo(id) {
    const g = game();
    const i = stepIndex(id);
    if (!g || !g.tut || i < 0 || (g.tut.step | 0) >= i) return;
    g.tut.step = i;
    base = null;
    save();
  }
  function finish(skipped) {

    if (!g) return;
    g.tut = Object.assign(g.tut || {}, { done: Date.now(), skipped: !!skipped });

    if (skipped) toast('🎓 Tutorial skipped', 'You can replay it any time', 'Menu → Settings → Intro & tutorial');
    else toast('🎓 Tutorial complete', 'Welcome to English Town!', 'Finish “First Day in Town” on the quest card.');

  function tutorialTick() {
    const sc = getScene();
    const g = game();
    if (playing || !sc || !g || !sc.player || !g.tut || !g.tut.intro || g.tut.done) return hideCoach();
    const list = steps();
    const i = Math.max(0, Math.min(list.length - 1, g.tut.step | 0));
    const s = list[i];
    const msgs = (g.stats && g.stats.messages) || 0;
    if (!base || base.i !== i) base = { i, x: sc.player.x, y: sc.player.y, msgs };
    const chatId = sc.chat && sc.chat.isOpen ? sc.chat.currentId : null;

    // Talking to Mia early skips the steps before it.
    if (chatId === 'kid' && i < stepIndex('say')) return jumpTo('say');

    const mia = sc.npcs && sc.npcs.find((n) => n.def && n.def.id === 'kid');
    const miaHere = !!(mia && mia.sprite && mia.sprite.visible && !!mia.inside === !!sc.inside);
    const dMia = miaHere ? Math.hypot(sc.player.x - mia.sprite.x, sc.player.y - mia.sprite.y) : Infinity;

    let shownStep = s;
    let note = null;
    if (s.id === 'move' && Math.hypot(sc.player.x - base.x, sc.player.y - base.y) > 80) return advance();
    if (s.id === 'find' && dMia < 90) return advance();
    if (s.id === 'talk' && dMia > 220) shownStep = list[stepIndex('find')];
    if (s.id === 'say') {
      if (chatId !== 'kid') {
        shownStep = list[stepIndex('talk')];
        note = { en: 'Talk to Mia again.', vi: 'Hãy nói chuyện lại với Mia.' };
        if (dMia > 220) shownStep = list[stepIndex('find')];
      } else if (msgs > base.msgs) return advance();
    }
    if (s.id === 'tips' && !chatId) return advance();
    if (s.id === 'menu' && shown('l11-shell')) return advance();

    // Hide behind big panels, and while chatting with someone other than Mia (or chatting at all on the last steps).
    const chatOk = s.id === 'say' || s.id === 'tips';
    if (COVERS.some(shown) || (chatId && (!chatOk || chatId !== 'kid'))) return hideCoach();
    renderCoach(i, list, shownStep, note);
    // On small screens the chat is tall: the card moves up over the minimap so it does not hide the chat's ✕.
    coachEl.classList.toggle('chatting', !!chatId);
  }

  // ---- yellow arrow to Mia (steps "find" and "talk") --------------------------------------------------------------
  function drawGuide(sc, time) {
    const st = sc.__lqIntro;
    if (!st) return;
    const gfx = st.g;
    gfx.clear();
    const g = game();
    if (playing || !g || !g.tut || !g.tut.intro || g.tut.done || !sc.player || sc.switching) return;
    const id = steps()[Math.max(0, g.tut.step | 0)];
    if (!id || (id.id !== 'find' && id.id !== 'talk' && id.id !== 'say')) return;
    if (sc.chat && sc.chat.isOpen) return;
    const mia = sc.npcs && sc.npcs.find((n) => n.def && n.def.id === 'kid');
    if (!mia || !mia.sprite || !mia.sprite.visible || !!mia.inside !== !!sc.inside) return;
    const mx = mia.sprite.x;
    const my = mia.sprite.y;
    const b = Math.sin(time * 0.006) * 4;
    // a bouncing marker over Mia's head
    const hy = my - 52 + b;
    gfx.fillStyle(0x3a2a00, 1);
    gfx.fillTriangle(mx - 10, hy - 2, mx + 10, hy - 2, mx, hy + 12);
    gfx.fillStyle(0xffd84a, 1);
    gfx.fillTriangle(mx - 7, hy, mx + 7, hy, mx, hy + 9);
    // an arrow around the player pointing at her
    const px = sc.player.x;
    const py = sc.player.y - 16;
    const d = Math.hypot(mx - px, my - py);
    if (d < 110) return;
    const a = Math.atan2(my - 16 - py, mx - px);
    const r0 = 30 + Math.sin(time * 0.008) * 3;
    const tip = { x: px + Math.cos(a) * (r0 + 16), y: py + Math.sin(a) * (r0 + 16) };
    const l = { x: px + Math.cos(a) * r0 + Math.cos(a + Math.PI / 2) * 9, y: py + Math.sin(a) * r0 + Math.sin(a + Math.PI / 2) * 9 };
    const rr = { x: px + Math.cos(a) * r0 - Math.cos(a + Math.PI / 2) * 9, y: py + Math.sin(a) * r0 - Math.sin(a + Math.PI / 2) * 9 };
    gfx.lineStyle(3, 0x3a2a00, 1);
    gfx.strokeTriangle(tip.x, tip.y, l.x, l.y, rr.x, rr.y);
    gfx.fillStyle(0xffd84a, 1);
    gfx.fillTriangle(tip.x, tip.y, l.x, l.y, rr.x, rr.y);
  }
  function attach(sc) {
    if (sc.__lqIntro) return;
    sc.__lqIntro = { g: sc.add.graphics().setDepth(12) };
    sc.events.on('update', (time) => drawGuide(sc, time));
    sc.events.once('shutdown', () => {
      sc.__lqIntro = null;
    });
  }

  // ---- when to start ----------------------------------------------------------------------------------------------
  // A save counts as new when nothing has happened in it yet.
  function isNewSave(g) {
    const quests = Object.values(g.quests || {}).some((v) => Number(v) > 0);
    const ach = g.ach && g.ach.done && Object.keys(g.ach.done).length > 0;
    const msgs = (g.stats && g.stats.messages) || 0;
    return !quests && !ach && msgs === 0 && (g.day || 1) <= 1;
  }
  function readyToPlay(sc) {
    if (!sc || !sc.player || !sc.store || !sc.store.game || sc.switching) return false;
    if (['lq-title', 'lq-device', 'lq-auth', 'lq-cc', 'lq-train'].some(shown)) return false;
    if (window.LQCreatorUI && window.LQCreatorUI.hasLook && !window.LQCreatorUI.hasLook()) return false;
    return true;
  }
  function selectFirstQuest() {
    const v = window.LQV8;
    try {
      if (v && v.select && !(v.isDone && v.isDone('first-day'))) v.select('first-day');
    } catch (e) {}
  }

  function tick() {
    const sc = getScene();
    if (sc && sc.add && sc.events && sc.player && !sc.__lqIntro) {
      try {
        attach(sc);
      } catch (e) {
        console.warn('intro guide attach failed', e);
      }
    }
    // After "First Day in Town" is finished, the quest card moves on to The Ten Shadows.
    const v = window.LQV8;
    try {
      if (v && v.selected && v.isDone && v.selected() === 'first-day' && v.isDone('first-day')) v.select('ten-shadows');
    } catch (e) {}

    if (playing || !readyToPlay(sc)) return hideCoach();
    const g = sc.store.game;
    if (!g.tut) {
      // Older saves with progress never see the intro; they can still replay it from the Settings.
      g.tut = isNewSave(g) ? { step: 0 } : { intro: 'old', done: 'old' };
      save();
    }
    if (!g.tut.intro) {
      playIntro().then(() => {
        const gg = game();
        if (!gg) return;
        gg.tut = Object.assign(gg.tut || {}, { intro: Date.now(), step: 0 });
        delete gg.tut.done;
        base = null;
        coachKey = '';
        save();
        selectFirstQuest();
      });
      return;
    }
    tutorialTick();
  }
  setInterval(() => {
    try {
      tick();
    } catch (e) {
      console.warn('intro tick failed', e);
    }
  }, 300);

  // ?intro=1 plays the opening scene right away (for checking it), without touching the save.
  try {
    if (new URLSearchParams(location.search).get('intro') === '1') {
      const go = () => playIntro();
      if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', go);
      else go();
    }
  } catch (e) {}

  window.LQIntro = {
    // Opening scene only; does not change the save.
    preview: () => playIntro(),
    // Menu → Settings: play the scene again, then start the tutorial from the first step.
    replay() {
      const g = game();
      if (!g) return;
      g.tut = { step: 0 };
      base = null;
      coachKey = '';
      save();
      tick();
    },
    skip,
    isPlaying: () => !!playing,
    status() {
      const g = game();
      const t = g && g.tut;
      const list = steps();
      return t ? { intro: !!t.intro, done: !!t.done, skipped: !!t.skipped, step: t.step | 0, stepId: (list[t.step | 0] || {}).id || null } : null;
    },
  };
})();

  // ---- animated night town drawn at low resolution (pixel art look) ----
  function startScene(cv) {
    const cx = cv.getContext('2d');
    const reduce = window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;
    const SCALE = 3;
    let W = 0, H = 0, stars = [], layers = [], raf = 0, last = 0, trainX = -200, trainWait = 3;
    const rnd = (a, b) => a + Math.random() * (b - a);

    function build() {
      W = Math.ceil(innerWidth / SCALE);
      H = Math.ceil(innerHeight / SCALE);
      cv.width = W;
      cv.height = H;
      stars = Array.from({ length: Math.floor((W * H) / 260) }, () => ({ x: Math.random() * W, y: Math.random() * H * 0.6, p: Math.random() * 6, s: Math.random() < 0.15 ? 2 : 1 }));
      const ground = H - 12;
      layers = [
        { col: '#101c33', win: '#27406b', min: 26, max: 62, gap: [0, 2], drift: 0 },
        { col: '#0a1326', win: '#f3d27a', min: 14, max: 44, gap: [0, 3], drift: 0 },
      ].map((L) => {
        const items = [];
        let x = -4;
        while (x < W + 10) {
          const w = Math.floor(rnd(8, 18)), h = Math.floor(rnd(L.min, L.max));
          const wins = [];
          for (let wx = x + 2; wx < x + w - 2; wx += 4) for (let wy = ground - h + 3; wy < ground - 3; wy += 5) wins.push({ x: wx, y: wy, on: Math.random() < 0.4, t: rnd(2, 9) });
          items.push({ x, w, h, wins, roof: Math.random() < 0.3 });
          x += w + Math.floor(rnd(L.gap[0], L.gap[1] + 1));
        }
        return Object.assign({}, L, { items });
      });
    }

    function draw(t, dt) {
      const ground = H - 12;
      const sky = cx.createLinearGradient(0, 0, 0, H);
      sky.addColorStop(0, '#060917');
      sky.addColorStop(0.6, '#10163a');
      sky.addColorStop(1, '#262a66');
      cx.fillStyle = sky;
      cx.fillRect(0, 0, W, H);
      for (const s of stars) {
        const a = reduce ? 0.8 : 0.45 + 0.55 * Math.abs(Math.sin(t / 900 + s.p));
        cx.fillStyle = `rgba(255,255,255,${a})`;
        cx.fillRect(Math.floor(s.x), Math.floor(s.y), s.s, s.s);
      }
      // moon
      const mx = Math.floor(W * 0.8), my = Math.floor(H * 0.2);
      cx.fillStyle = '#fff6c8';
      cx.beginPath();
      cx.arc(mx, my, 7, 0, 7);
      cx.fill();
      cx.fillStyle = '#12234a';
      cx.beginPath();
      cx.arc(mx + 3, my - 1, 6, 0, 7);
      cx.fill();
      // far hills
      cx.fillStyle = '#0d1a35';
      for (let x = 0; x < W; x++) cx.fillRect(x, ground - 14 - Math.floor(6 * Math.sin(x / 17) + 4 * Math.sin(x / 7)), 1, 30);
      // skyline (two layers) with twinkling windows
      for (const L of layers) {
        for (const b of L.items) {
          cx.fillStyle = L.col;
          cx.fillRect(b.x, ground - b.h, b.w, b.h);
          if (b.roof) cx.fillRect(b.x + 2, ground - b.h - 3, b.w - 4, 3);
          for (const w of b.wins) {
            if (!reduce) {
              w.t -= dt / 1000;
              if (w.t <= 0) {
                w.on = !w.on;
                w.t = rnd(3, 12);
              }
            }
            if (w.on) {
              cx.fillStyle = L.win;
              cx.fillRect(w.x, w.y, 2, 2);
            }
          }
        }
      }
      // ground and rails
      cx.fillStyle = '#060b16';
      cx.fillRect(0, ground, W, H - ground);
      cx.fillStyle = '#3b4a66';
      cx.fillRect(0, ground + 3, W, 1);
      cx.fillStyle = '#1d2740';
      for (let x = (Math.floor(t / 60) % 6) - 6; x < W; x += 6) cx.fillRect(x, ground + 4, 2, 2);
      // a train crossing now and then
      if (!reduce) {
        if (trainX > W + 20 || trainWait > 0) {
          trainWait -= dt / 1000;
          if (trainWait <= 0 && trainX > W + 20) trainX = -90;
        }
        if (trainX <= W + 20 && trainWait <= 0) {
          trainX += (dt / 1000) * 46;
          if (trainX > W + 20) trainWait = rnd(6, 11);
          for (let i = 0; i < 4; i++) {
            const x = Math.floor(trainX - i * 22);
            cx.fillStyle = i === 0 ? '#16b3a5' : '#e8eef2';
            cx.fillRect(x, ground - 9, 20, 8);
            cx.fillStyle = '#16b3a5';
            cx.fillRect(x, ground - 4, 20, 2);
            cx.fillStyle = '#ffe9a8';
            for (let w = 2; w < 18; w += 5) cx.fillRect(x + w, ground - 7, 3, 2);
          }
        }
      }
    }

    function frame(now) {
      const dt = Math.min(80, now - (last || now));
      last = now;
      draw(now, dt);
      raf = requestAnimationFrame(frame);
    }
