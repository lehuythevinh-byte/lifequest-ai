  function wvCivGridSafe(a,x,y){
    const tx=Math.floor(x),ty=Math.floor(y);
    if(!a||tx<2||ty<2||tx>=a.w-2||ty>=a.h-2)return false;
    if(a.isTileBlocked(tx,ty)||a.warpAt?.(tx,ty))return false;
    // The decorative 1-tile-wide banks cannot be crossed outside the real deck.
    if(WV_WORLD_GEOMETRY.isWaterBlocked(tx,ty))return false;
    return true;
  }
  function wvCivNearestSafe(a,point){
    if(!point||point.length<2)return null;
    if(wvCivGridSafe(a,point[0],point[1]))return [Math.floor(point[0]),Math.floor(point[1])];
    for(let r=1;r<Math.max(a.w,a.h);r++){
      let best=null,bestDist=Infinity;
      for(let dx=-r;dx<=r;dx++)for(let dy=-r;dy<=r;dy++){
        if(Math.max(Math.abs(dx),Math.abs(dy))!==r)continue;
        const x=Math.floor(point[0])+dx,y=Math.floor(point[1])+dy;
        if(!wvCivGridSafe(a,x,y))continue;
        const d=Math.hypot(dx,dy);
        if(d<bestDist){best=[x,y];bestDist=d;}
      }
      if(best)return best;
    }
    return null; // Never substitute an unverified spawn or teleport through water.
  }

  function wvCivRoute(a,start,target){
    if(!a||!target)return [];
    const sx=Math.floor(start.x/T),sy=Math.floor(start.y/T);
    const dst=wvCivNearestSafe(a,target);
    if(!dst||!wvCivGridSafe(a,sx,sy))return [];
    const [tx,ty]=dst,w=a.w,h=a.h;
    const n=w*h,from=sy*w+sx,to=ty*w+tx;
    const movementCosts=wvNavCostGrid(a);
    if(from===to)return [];
    const cost=new Float64Array(n).fill(Infinity),previous=new Int32Array(n).fill(-1),closed=new Uint8Array(n);
    const heap=[];
    function push(node,score){
      const item={node,score};heap.push(item);
      for(let i=heap.length-1;i>0;){const p=(i-1)>>1;if(heap[p].score<=item.score)break;heap[i]=heap[p];i=p;heap[i]=item;}
    }
    function pop(){
      const out=heap[0],last=heap.pop();if(heap.length){let i=0;
        while(i*2+1<heap.length){let child=i*2+1;if(child+1<heap.length&&heap[child+1].score<heap[child].score)child++;
          if(heap[child].score>=last.score)break;heap[i]=heap[child];i=child;}
        heap[i]=last;
      }return out;
    }
    const heuristic=(x,y)=>(Math.abs(x-tx)+Math.abs(y-ty));
    cost[from]=0;push(from,heuristic(sx,sy));
    let steps=0;
    while(heap.length&&steps++<n*7){
      const {node}=pop();if(closed[node])continue;closed[node]=1;
      if(node===to)break;
      const x=node%w,y=Math.floor(node/w);
      for(const [dx,dy] of [[1,0],[0,1],[-1,0],[0,-1]]){
        const nx=x+dx,ny=y+dy;if(!wvCivGridSafe(a,nx,ny))continue;
        const next=ny*w+nx;if(closed[next])continue;
        const price=cost[node]+movementCosts[next];
        if(price+1e-6<cost[next]){cost[next]=price;previous[next]=node;push(next,price+heuristic(nx,ny));}
      }
    }
    if(previous[to]===-1)return [];
    const raw=[];for(let at=to;at!==from;at=previous[at]){
      if(at<0||raw.length>n)return [];
      raw.push({x:(at%w+.5)*T,y:(Math.floor(at/w)+.5)*T});
    }
    raw.reverse();
    // Compress straight segments only; never smooth across diagonal wall corners.
    const route=[];

      const before=i===0?{x:(sx+.5)*T,y:(sy+.5)*T}:raw[i-1];
      const after=raw[i+1];
      if(!after||Math.sign(raw[i].x-before.x)!==Math.sign(after.x-raw[i].x)||
        Math.sign(raw[i].y-before.y)!==Math.sign(after.y-raw[i].y))route.push(raw[i]);
    }
    return route;
  }

  /* Species-specific schedules: decisions only every 100ms; no positional motion while sleeping / grazing. */
  function wvAnimalStep(sc,animal,time,dt,night){
    const k=animal.index,t=animal.type,sec=time*.001,ph=sec*.7+animal.phase;
    const ox=animal.x,oy=animal.y;
    let nx=ox,ny=oy,state='IDLE';
    const steer=(gx,gy,speed)=>{
      const dx=gx-ox,dy=gy-oy,d=Math.hypot(dx,dy);
      if(d>.8){const reach=Math.min(d,speed*dt);nx=ox+dx/d*reach;ny=oy+dy/d*reach;}
      return d;
    };
    if(t==='fish'){
      const dart=Math.sin(ph*2.1+k)>.88;
      const targetY=animal.baseY+Math.sin(ph*.77)*25;
      const targetX=animal.baseX>72*T?animal.baseX+Math.cos(ph*1.2)*17:wvRiverSurface(targetY/T).x+Math.sin(ph*1.8+k)*18;
      steer(targetX,targetY,dart?65:23);state=dart?'DART':'SWIM';
    }else if(t==='otter'){
      const floating=Math.sin(ph*.91+k)>.65;
      const targetY=animal.baseY+Math.sin(ph*.51+k)*22;
      const targetX=animal.baseX>72*T?animal.baseX+Math.cos(ph*1.05)*16:wvRiverSurface(targetY/T).x+Math.cos(ph*1.35+k)*28;
      if(!floating)steer(targetX,targetY,17);
      state=floating?'FLOAT':'SWIM';
    }else if(t==='waterbird'){
      const grooming=Math.sin(ph*.6+k)>.63;
      const targetY=animal.baseY+Math.sin(ph*.45+k)*20;
      const targetX=(animal.baseX>72*T?animal.baseX:wvRiverSurface(targetY/T).x)+Math.sin(ph*.8+k)*28;
      if(!grooming)steer(targetX,targetY,10);
      state=grooming?'PREEN':'PADDLE';
    }else if(t==='aru'){
      const fly=Math.sin(ph*.85+k)>.16;
      const gx=animal.baseX+Math.sin(ph*.69+k)*29,gy=animal.baseY+Math.cos(ph*.57+k)*31;
      if(fly)steer(gx,gy,29);
      state=fly?'FLY':'PERCH';
    }else if(t==='butterfly'){
      const rest=Math.sin(ph*.64+k)<-.80;
      if(!rest)steer(animal.baseX+Math.sin(ph*1.2+k)*34+Math.sin(ph*2.7)*10,
                     animal.baseY+Math.cos(ph*1.9+k)*15,24);
      state=rest?'HOVER':'FLUTTER';
    }else if(t==='firefly'){
      if(night)steer(animal.baseX+Math.sin(ph*.57+k)*23,animal.baseY+Math.cos(ph*.61+k)*15,9);
      state=night?'GLOW':'REST';
    }else if(t==='varek'){
      const caretaker=sc.__wvLiving?.civilians.find(n=>n.id==='varel');
      if(caretaker){
        const goalX=caretaker.x+(k%3-1)*32+Math.sin(ph*.6+k)*10;
        const goalY=caretaker.y+21+(k%2)*12;
        const d=Math.hypot(goalX-ox,goalY-oy);
        if(d>22){steer(goalX,goalY,d>75?48:31);state=d>75?'TROT':'FOLLOW';}
        else state=Math.sin(ph*.85+k)>.15?'SIT':'IDLE';
      }
    }else if(t==='deer'){
      const grazing=Math.sin(ph*.48+k)>.08;
      const herd=sc.__wvLiving?.creatures.filter(o=>o!==animal&&o.type==='deer'&&Math.hypot(o.baseX-animal.baseX,o.baseY-animal.baseY)<90)||[];
      if(!grazing){
        const hx=herd.length?herd.reduce((sum,o)=>sum+o.baseX,0)/herd.length:animal.baseX;
        const hy=herd.length?herd.reduce((sum,o)=>sum+o.baseY,0)/herd.length:animal.baseY;
        const aim=sec*.17+animal.phase;
        steer(animal.baseX+Math.sin(aim)*24+(hx-animal.baseX)*.1,
              animal.baseY+Math.cos(aim)*15+(hy-animal.baseY)*.1,11);
      }
      state=grazing?'GRAZE':(Math.sin(ph*1.3+k)>.76?'LOOK':'WALK');
    }else if(t==='lunari'){
      const sleep=Math.sin(ph*.31+k)>.25;
      const stretch=!sleep&&Math.sin(ph*1.16+k)>.62;
      if(!sleep&&!stretch)steer(animal.baseX+Math.sin(sec*.23+animal.phase)*15,
                                animal.baseY+Math.cos(sec*.19+animal.phase)*10,8);
      state=sleep?'SLEEP':stretch?'STRETCH':'PROWL';
    }else if(t==='squirrel'){
      const hopping=Math.sin(ph*1.4+k)>.27;
      if(hopping)steer(animal.baseX+Math.sin(sec*.7+animal.phase)*18,
                       animal.baseY+Math.cos(sec*.64+animal.phase)*14,24);
      state=hopping?'HOP':'FORAGE';
    }
    if(WV_HABITAT_SYSTEM.aquatic.includes(t)){
      if(WV_HABITAT_SYSTEM.inWater(nx,ny)){animal.x=nx;animal.y=ny;}
    }else if(WV_HABITAT_SYSTEM.land.includes(t)){
      if(wvCivGridSafe(sc.area,nx/T,ny/T)){animal.x=nx;animal.y=ny;}
      else state='REST';
    }else{
      animal.x=Math.min((sc.area.w-2)*T,Math.max(2*T,nx));
      animal.y=Math.min((sc.area.h-2)*T,Math.max(2*T,ny));
    }
    const dx=animal.x-ox,dy=animal.y-oy;
    if(Math.abs(dx)>.08)animal.dir=dx>0?1:-1;
    animal.state=state;
  }
