    function makePuff(R) {
      const S = R * 2.7;
      const [c, x] = offscreen(S, S);
      const m = S / 2;

        const z = rnd(-1, 1),
          p = rnd(0, TAU),
          q = Math.sqrt(1 - z * z);
        pts.push({ x: q * Math.cos(p), y: q * Math.sin(p), z });
      }
      pts.sort((a, b) => a.z - b.z);
      x.lineCap = 'round';
      for (const p of pts) {
        const depth = (p.z + 1) / 2;
        const a = 0.22 + 0.62 * depth;

          ty = m + p.y * R;
        x.strokeStyle = `rgba(255,255,255,${(a * 0.75).toFixed(3)})`;
        x.lineWidth = 0.7 + depth * 0.5;

        x.lineTo(tx, ty);
        x.stroke();

        x.arc(tx, ty, 1.1, 0, TAU);
        x.fill();

      x.arc(m, m, R * 0.055, 0, TAU);

      }

      v.addColorStop(0, 'rgba(5,8,20,0)');
      v.addColorStop(1, 'rgba(5,8,20,.45)');
      ctx.fillStyle = v;
      ctx.fillRect(0, 0, W, H);
    }

    function frame(now) {
      if (stopped) return;
      const dt = Math.min(0.05, (now - (last || now)) / 1000);
      last = now;
      draw(now, dt);
      raf = requestAnimationFrame(frame);
    }

    layout();
    window.addEventListener('resize', layout);
    raf = requestAnimationFrame(frame);
    return () => {
      stopped = true;
      cancelAnimationFrame(raf);
      window.removeEventListener('resize', layout);

  async function loadSdk() {

  async function fetchRow() {
    const r = await client.from('saves').select('data,updated_at').eq('user_id', user.id).maybeSingle();
    if (r.error) throw r.error;
    return r.data || null;
  }

  function sameData(a, b) {
    const ka = Object.keys(a).sort(),
      kb = Object.keys(b).sort();
    return ka.length === kb.length && ka.every((k, i) => k === kb[i] && a[k] === b[k]);
  }

  async function push(opts = {}) {
    if (!user || !client || busy) return false;
    busy = true;
    setStatus('sync');
    try {
      const data = snapshot();
      if (!data[SAVE_KEY]) {
        setStatus('ok');
        return true;
      }
      const body = { v: 1, data, savedAt: new Date().toISOString() };
      if (JSON.stringify(body).length > MAX_BYTES) throw new Error('Save data is too large');
      const meta = readMeta();
      if (!opts.force) {
        const row = await fetchRow();
        if (row && row.updated_at !== meta.lastCloudAt && !sameData(row.data.data || {}, data)) {
          busy = false;
          const pick = await chooseSave(data, row, 'Server progress was changed from another device.');
          if (pick === 'cloud') {
            applySnapshot(row.data.data || {});
            writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false });
            hardReload();
            return false;
          }
          return await push({ force: true });
        }
      }
      const r = await client.from('saves').upsert({ user_id: user.id, data: body }).select('updated_at').single();
      if (r.error) throw r.error;
      writeMeta({ userId: user.id, lastCloudAt: r.data.updated_at, dirty: false, guest: false });
      setStatus('ok');
      return true;
    } catch (e) {
      console.warn('cloud save failed', e);
      setStatus('bad', friendly(e));
      return false;
    } finally {
      busy = false;
    }
  }

  // Best-effort save while the tab is closing (fetch with keepalive survives page unload).
  function pushOnLeave() {
    if (!user || !token || !readMeta().dirty) return;
    const data = snapshot();
    if (!data[SAVE_KEY]) return;
    const body = JSON.stringify({ user_id: user.id, data: { v: 1, data, savedAt: new Date().toISOString() } });
    if (body.length > 60000) return; // keepalive requests are limited to 64 KB
    try {
      fetch(URL_ + '/rest/v1/saves?on_conflict=user_id', {
        method: 'POST',
        keepalive: true,
        headers: { apikey: ANON, Authorization: 'Bearer ' + token, 'Content-Type': 'application/json', Prefer: 'resolution=merge-duplicates,return=minimal' },
        body,
      });
      writeMeta({ dirty: false, lastCloudAt: '' });
    } catch (e) {
      /* ignore */
    }
  }

  function markDirty() {
    const m = readMeta();
    if (!m.dirty) writeMeta({ dirty: true });
    if (user) {
      setStatus('sync');
      clearTimeout(pushTimer);
      pushTimer = setTimeout(() => push(), PUSH_DELAY);
    }
  }

  /** Decide which copy to keep after signing in. Returns 'applied' when this device's data was replaced. */
  async function reconcile() {
    const meta = readMeta();
    const row = await fetchRow();
    const local = snapshot();
    const hasLocal = !!local[SAVE_KEY];
    if (!row) {
      if (hasLocal) await push({ force: true });
      else writeMeta({ userId: user.id, lastCloudAt: '', dirty: false, guest: false });
      return 'kept';
    }
    const cloudData = row.data.data || {};
    if (hasLocal && sameData(cloudData, local)) {
      writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false, guest: false });
      return 'kept';
    }
    if (!hasLocal) {
      applySnapshot(cloudData);
      writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false, guest: false });
      return 'applied';
    }
    if (meta.userId === user.id && row.updated_at === meta.lastCloudAt) {
      if (meta.dirty) await push({ force: true });
      return 'kept';
    }
    if (meta.userId === user.id && !meta.dirty) {
      applySnapshot(cloudData);
      writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false });
      return 'applied';
    }
    const why = meta.userId === user.id ? 'Both places have unsynced changes.' : 'This device has game progress (guest or another account), and your account also has progress on the server.';
    const pick = await chooseSave(local, row, why);
    if (pick === 'cloud') {
      applySnapshot(cloudData);
      writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false, guest: false });
      return 'applied';
    }
    await push({ force: true });
    return 'kept';
  }

  async function checkRemote() {
    if (!user || busy || Date.now() - lastCheck < 20000) return;
    lastCheck = Date.now();
    try {
      const row = await fetchRow();
      const meta = readMeta();
      if (!row || row.updated_at === meta.lastCloudAt) return;
      if (sameData(row.data.data || {}, snapshot())) {
        writeMeta({ lastCloudAt: row.updated_at, dirty: false });
        return;
      }
      if (meta.dirty) await push();
      else showBanner('Newer progress is available from another device.', 'Reload', () => reloadWithCloud(row));
    } catch (e) {
      /* offline: ignore */
    }
  }
  function reloadWithCloud(row) {
    applySnapshot(row.data.data || {});
    writeMeta({ userId: user.id, lastCloudAt: row.updated_at, dirty: false });
    hardReload();
  }

  async function signOut() {
    if (!confirm('Sign out? Your progress will be saved to the server, then removed from this device.')) return;
    try {
      clearTimeout(pushTimer);
      if (readMeta().dirty) await push({ force: false });
      if (readMeta().dirty) {
        if (!confirm('Could not save to the server. Sign out anyway and delete data on this device?')) return;
      }
      await client.auth.signOut();
    } catch (e) {
      console.warn(e);
    }
    applySnapshot({});
    nativeRemove.call(store, META_KEY);
    hardReload();
  }

  function installHooks() {
    Storage.prototype.setItem = function (k, v) {
      if (this === store && blockWrites && isSyncKey(k)) return;
      nativeSet.call(this, k, v);
      if (this === store && !suppress && isSyncKey(k)) markDirty();
    };
    Storage.prototype.removeItem = function (k) {
      if (this === store && blockWrites && isSyncKey(k)) return;
      nativeRemove.call(this, k);
      if (this === store && !suppress && isSyncKey(k)) markDirty();
    };
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'hidden') pushOnLeave();
      else checkRemote();
    });
    window.addEventListener('pagehide', pushOnLeave);
    window.addEventListener('lifequest:lookchanged', render);
    setInterval(() => {
      if (user && readMeta().dirty && !busy && !pushTimer) pushTimer = setTimeout(() => push(), PUSH_DELAY);
    }, 60000);
  }

  // ---------- entry point (called by js/loader.js before the game starts) ----------
  async function start() {
    const boot = () => mountChipWhenReady();
    if (!ENABLED) {
      await showLogin({ allowGuest: true, disabled: true });
      boot();
      return;
    }
    try {
      await loadSdk();
      client = window.supabase.createClient(URL_, ANON, { auth: { persistSession: true, autoRefreshToken: true, detectSessionInUrl: true } });
    } catch (e) {
      console.warn('Đăng nhập không khả dụng:', e);
      boot();
      return;
    }
    installHooks();
    client.auth.onAuthStateChange((evt, session) => {
      token = session ? session.access_token : '';
      if (session) user = session.user;
      if (evt === 'SIGNED_OUT') user = null;
    });
    try {
      const s = await client.auth.getSession();
      if (s.data && s.data.session) {
        user = s.data.session.user;
        token = s.data.session.access_token;
      }
      if (!user) {
        const r = await showLogin({ allowGuest: true });
        if (r === 'guest') writeMeta({ guest: true });
      }
      if (user) await reconcile();
    } catch (e) {
      console.warn('Không đồng bộ được lúc khởi động:', e);
      setStatus('bad', friendly(e));
    }
    boot();
  }

  function draw() {

    if (!root || !sc || !look) return;
    const cv = root.querySelector('canvas'),
      cx = cv.getContext('2d');
    cx.clearRect(0, 0, cv.width, cv.height);
    cx.imageSmoothingEnabled = false;
    try {
      const img = window.LQCharacter.imageFor(sc, look, dir === 'left' ? 'side' : dir, 0);
      cx.save();
      if (dir === 'left') {
        cx.translate(cv.width, 0);
        cx.scale(-1, 1);
      }
      cx.drawImage(img, 0, 0, cv.width, cv.height);
      cx.restore();
    } catch (e) {
      console.warn('creator preview failed', e);
    }
  }

  function onClick(ev) {
    const b = ev.target.closest('button');
    if (!b || !root.contains(b)) return;
    if (b.dataset.tab) {
      tab = b.dataset.tab;
      render();
      return;
    }
    if (b.dataset.lock) {
      if (window.LQAchievements) window.LQAchievements.hint(b.dataset.lock);
      return;
    }
    if (b.dataset.p) {
      const preset = PRESETS[Number(b.dataset.p)];
      if (isLocked(preset)) return;
      const typed = (root.querySelector('#lq-cc-name') || {}).value;
      look = Object.assign(presetLook(preset), { name: (typed || look.name || '').trim() || preset.name });
      tab = 'custom';
      render();
      return;
    }
    if (b.dataset.d) {
      dir = b.dataset.d;
      render();
      return;
    }
    if (b.dataset.g) {
      const keep = look.name;
      look = Object.assign(defaults(b.dataset.g), { name: keep });
      render();
      return;
    }
    if (b.dataset.hat) {
      look.hat = b.dataset.hat === 'none' ? null : { kind: b.dataset.hat, color: look.hat ? look.hat.color : HAT_COLORS[0] };
      render();
      return;
    }
    if (b.dataset.t) {
      const t = b.dataset.t;
      if (t === 'backpack') look.backpack = look.backpack ? 0 : BAGS[0];
      else if (t === 'collar') {
        look.collar = !look.collar;
        look.tie = look.collar ? look.tie || 0xb8433f : null;
      } else look[t] = !look[t];
      render();
      return;
    }
    if (b.dataset.f) {
      const f = b.dataset.f,
        v = b.dataset.v;
      if (f === 'hatcolor') look.hat = { kind: look.hat.kind, color: Number(v) };
      else if (f === 'style') {
        look.style = v;
        if (STYLE_COLORS[v] !== undefined) look.hair = STYLE_COLORS[v];
      }
      else look[f] = Number(v);
      render();
      return;
    }
    if (b.id === 'lq-cc-random') {
      const g = pick(['male', 'female']);
      const name = look.name;
      look = Object.assign(defaults(g), {
        name,
        skin: pick(SKINS),
        hair: pick(HAIRS),
        style: pick(STYLES)[0],
        eye: pick(EYES),
        outfit: pick(OUTFITS),
        pants: pick(PANTS),
        hat: Math.random() < 0.4 ? { kind: pick(HATS.slice(1))[0], color: pick(HAT_COLORS) } : null,
        glasses: Math.random() < 0.25,
        dress: g === 'female' && Math.random() < 0.6,
        backpack: Math.random() < 0.3 ? pick(BAGS) : 0,
      });
      render();
      return;

    if (b.id === 'lq-cc-save') save();
    if (b.id === 'lq-cc-cancel') close(false);
  }

  function save() {
    const name = (root.querySelector('#lq-cc-name').value || '').trim().replace(/\s+/g, ' ');
    const err = root.querySelector('.err');
    if (!name) {
      err.textContent = 'Please enter a name for your character.';
      root.querySelector('#lq-cc-name').focus();
      return;
    }
    look.name = name.slice(0, 20);
    const sc = getScene();
    if (!sc || !sc.store || !sc.store.game) {
      err.textContent = 'The game has not finished loading; try again in a moment.';
      return;
    }
    const g = sc.store.game;
    if (!g.rpg || typeof g.rpg !== 'object') g.rpg = {};
    g.rpg.look = JSON.parse(JSON.stringify(look));
    sc.store.save();
    try {
      sc.applyRankLook();
    } catch (e) {
      console.warn('applyRankLook failed', e);
    }
    window.dispatchEvent(new CustomEvent('lifequest:lookchanged', { detail: g.rpg.look }));
    close(true);
  }

  function applyPreset(i) {

    if (!sc || !sc.store || !sc.store.game) return;

      console.warn('applyRankLook failed', e);

    window.dispatchEvent(new CustomEvent('lifequest:lookchanged', { detail: g.rpg.look }));

/*
 * Shrine cutscene: plays when the 20th finger is picked up (see fingers.js).
 *
 * 1. Four huge red parallelogram eyes fill a white, speckled screen.
 * 2. The camera pulls back: the eyes shrink up into a blood-red cloudy sky while the white clears, revealing the
 *    domain - a shrine rising out of cyan, rippling water with its reflection.
 * 3. Sukuna is a tiny figure on the shrine roof, dead centre between the eyes, so the domain feels vast around him.
 * 4. The picture holds for a moment, then fades out.
 *
 * It is drawn at 480 x 270 "pixels" and scaled up without smoothing for the pixel-art look.
 * window.LQShrine.play({ freeze }) returns a promise that resolves when the cutscene is over. `freeze` (ms) pins the
 * picture at that moment and never fades; it is only used to take screenshots while tuning the artwork.
 */
(function () {
  'use strict';

  const VW = 480;
  const VH = 270;
  const T_EYES = 600; // eyes fill the screen
  const T_PULL = 900; // camera pulls back, white clears
  const T_HOLD = 1100; // the domain, held
  const T_FADE = 1500; // fade out
  const RED = '#ed1c24';
  // Lead-in before the flash: the screen goes dark and a line is typed out letter by letter.
  const LEAD_TEXT = 'King of Curses has arrived';
  const LEAD_FONT = 'Cinzel, "Trajan Pro", "Times New Roman", Georgia, serif';
  const LEAD_FADE = 400; // fade to black
  const LEAD_CHAR = 55; // ms per letter
  const LEAD_HOLD = 800; // pause on the finished line before the cut
  const T_LEAD = LEAD_FADE + LEAD_TEXT.length * LEAD_CHAR + LEAD_HOLD;
  const HORIZON = 150;
  const CX = VW / 2;

  // The four eyes at full-screen size (upper pair large, lower pair smaller); right side is the mirror image.
  const EYES_BIG = [
    [[34, 52], [212, 96], [212, 150], [34, 106]],
    [[62, 176], [200, 176], [186, 236], [48, 236]],
  ];
  const EYES = [...EYES_BIG, ...EYES_BIG.map((e) => e.map(([x, y]) => [VW - x, y]))];

  function seeded(seed) {
    let s = seed | 0;
    return () => {
      s = (s + 0x6d2b79f5) | 0;
      let t = Math.imul(s ^ (s >>> 15), 1 | s);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  function canvas(w, h) {

  function paintSky() {

    const rnd = seeded(7);

    const bh = HORIZON / bands.length;

        g.fillRect(Math.round(CX + Math.cos(ang) * r * 1.6), Math.round(cy + Math.sin(ang) * r * 0.34), 1, 1);
      }
    }
  }

  function sparkles(g, t) {
    const rnd = seeded(Math.floor(t / 140) + 3);
    for (let i = 0; i < 90; i++) {
      g.fillStyle = rnd() < 0.5 ? 'rgba(120,230,235,0.8)' : 'rgba(255,255,255,0.6)';
      g.fillRect(Math.floor(rnd() * VW), HORIZON + 4 + Math.floor(rnd() * (VH - HORIZON - 4)), 1, 1);
    }
  }

  function ash(g, t) {
    const rnd = seeded(99);
    for (let i = 0; i < 70; i++) {
      const x0 = rnd() * VW;
      const y0 = rnd() * HORIZON;
      const sp = 0.012 + rnd() * 0.02;
      const y = (((y0 - t * sp) % HORIZON) + HORIZON) % HORIZON;
      g.fillStyle = rnd() < 0.5 ? 'rgba(255,90,60,0.75)' : 'rgba(255,176,138,0.6)';
      g.fillRect(Math.floor(x0 + Math.sin(t / 700 + i) * 3), Math.floor(y), 1, 1);
    }
  }

  function scene(g, t, sky, water) {
    const off = (t * 0.006) % VW;
    g.drawImage(sky, -off, 0);
    g.drawImage(sky, VW - off, 0);
    g.fillStyle = 'rgba(160,30,44,0.7)';
    g.fillRect(0, HORIZON - 1, VW, 1);
    g.drawImage(water, 0, HORIZON);

    g.translate(0, BASE * 2);
    g.scale(1, -1);
    shrine(g);
    g.restore();
    ripples(g, t);
    sparkles(g, t);
    shrine(g);
    sukuna(g);
    ash(g, t);
  }

  function eyes(g, e, t) {
    const lw = 22 + (7 - 22) * e;

        const sy = SKY_CENTER[1] + (y - BIG_CENTER[1]) * SKY_SCALE;
        return [x + (sx - x) * e, y + (sy - y) * e];
      });
      g.beginPath();
      g.moveTo(pts[0][0], pts[0][1]);

      g.strokeStyle = RED;
      g.lineWidth = lw;
      g.fill();
      g.stroke();
      g.strokeStyle = '#fff';
      g.lineWidth = 1;
      g.stroke();
    }
    g.restore();

  function speckle(g, alpha) {
    if (alpha <= 0.02) return;
    g.globalAlpha = alpha;
    const centers = [[110, 100], [370, 100], [120, 205], [360, 205]];
    for (let i = 0; i < 9000; i++) {
      let x = Math.random() * VW;
      let y = Math.random() * VH;
      if (Math.random() < 0.7) {
        const c = centers[Math.floor(Math.random() * 4)];
        x = c[0] + gauss() * 70;
        y = c[1] + gauss() * 55;
      }
      g.fillStyle = Math.random() < 0.45 ? RED : '#000';
      g.fillRect(Math.floor(x), Math.floor(y), 1, 1);
    }
    g.globalAlpha = 1;

  function playSound() {
    try {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) return;
      const ac = new AC();
      const t = ac.currentTime;
      const o = ac.createOscillator();
      const g = ac.createGain();
      o.type = 'sine';
      o.frequency.setValueAtTime(110, t);
      o.frequency.exponentialRampToValueAtTime(38, t + 0.9);
      g.gain.setValueAtTime(0.0001, t);
      g.gain.exponentialRampToValueAtTime(0.5, t + 0.03);
      g.gain.exponentialRampToValueAtTime(0.0001, t + 1.4);
      o.connect(g).connect(ac.destination);
      o.start(t);
      o.stop(t + 1.5);
      setTimeout(() => ac.close().catch(() => {}), 2000);
    } catch (e) {
      /* sound is optional */
    }

      // wide screens are filled edge to edge; tall ones keep the whole picture and show dark bars

      const reduce = window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;
      // The dark lead-in: the line is typed in the middle of the screen (split over two lines on narrow screens).
      const drawLead = (el) => {
        cv.style.opacity = String(Math.min(1, el / LEAD_FADE));
        ctx.setTransform(1, 0, 0, 1, 0, 0);
        ctx.fillStyle = '#050207';
        ctx.fillRect(0, 0, W, H);
        const vg = ctx.createRadialGradient(W / 2, H / 2, Math.min(W, H) * 0.1, W / 2, H / 2, Math.max(W, H) * 0.7);
        vg.addColorStop(0, 'rgba(120,12,24,0.28)');
        vg.addColorStop(1, 'rgba(0,0,0,0)');

        ctx.fillRect(0, 0, W, H);
        const shown = Math.max(0, Math.min(LEAD_TEXT.length, Math.floor((el - LEAD_FADE) / LEAD_CHAR)));
        const size = Math.max(18, Math.min(W * 0.04, 52));
        ctx.font = `800 ${size}px ${LEAD_FONT}`;
        if ('letterSpacing' in ctx) ctx.letterSpacing = `${(size * 0.1).toFixed(1)}px`; // wide, monumental spacing
        ctx.textBaseline = 'middle';
        ctx.textAlign = 'left';
        let lines = [LEAD_TEXT];
        if (ctx.measureText(LEAD_TEXT).width > W * 0.88) {
          const words = LEAD_TEXT.split(' ');
          const mid = Math.ceil(words.length / 2);
          lines = [words.slice(0, mid).join(' '), words.slice(mid).join(' ')];
        }
        const lh = size * 1.5;
        let left = shown;
        lines.forEach((ln, i) => {
          const part = ln.slice(0, Math.max(0, Math.min(ln.length, left)));
          left -= ln.length + 1;
          const w = ctx.measureText(ln).width;
          const x = (W - w) / 2;
          const y = H / 2 + (i - (lines.length - 1) / 2) * lh;
          ctx.shadowColor = '#ff1c2c';
          ctx.shadowBlur = size * 0.5;
          ctx.fillStyle = '#f4e9e9';

          // blinking caret at the end of what has been typed so far

            if (part.length > 0 || i === 0) {

              ctx.fillRect(x + ctx.measureText(part).width + 3, y - size * 0.55, Math.max(2, size * 0.06), size * 1.1);
            }
          }
        });
      };
      let soundDone = false;
      const start = performance.now();
      const frame = (now) => {
        const el = now - start;
        if (leadAt !== null || (freeze === null && el < T_LEAD)) {
          drawLead(leadAt !== null ? leadAt : el);
          requestAnimationFrame(frame);
          return;
        }
        if (!soundDone && freeze === null) {
          soundDone = true;
          playSound();
        }
        const t = freeze !== null ? freeze : el - T_LEAD;
        const total = T_EYES + T_PULL + T_HOLD + T_FADE;
        if (freeze === null && t >= total) {
          cv.remove();
          playing = false;
          resolve();
          return;
        }
        const fadeStart = T_EYES + T_PULL + T_HOLD;
        cv.style.opacity = freeze !== null || t < fadeStart ? '1' : String(Math.max(0, 1 - (t - fadeStart) / T_FADE));
        const e = reduce ? 1 : ease((t - T_EYES) / T_PULL);
        // the domain, with the camera easing back from a slight zoom
        g.setTransform(1, 0, 0, 1, 0, 0);
        g.fillStyle = '#1a0610';
        g.fillRect(0, 0, VW, VH);
        g.save();
        const zs = 1.22 - 0.22 * e;
        g.translate(CX, HORIZON);
        g.scale(zs, zs);
        g.translate(-CX, -HORIZON);
        scene(g, t, sky, water);
        g.restore();
        // white screen clears away
        g.fillStyle = `rgba(255,255,255,${(1 - e).toFixed(3)})`;
        g.fillRect(0, 0, VW, VH);
        speckle(g, reduce ? 0 : 1 - e);
        const shake = reduce || freeze !== null ? 0 : Math.max(0, 1 - t / 350) * 3;
        g.save();
        g.translate((Math.random() - 0.5) * shake, (Math.random() - 0.5) * shake);
        eyes(g, e, t);
        g.restore();
        ctx.imageSmoothingEnabled = false;
        ctx.fillStyle = '#1a0610';
        ctx.fillRect(0, 0, W, H);
        ctx.drawImage(lo, Math.round((W - VW * scale) / 2), Math.round((H - VH * scale) / 2), VW * scale, VH * scale);
        requestAnimationFrame(frame);
      };
      requestAnimationFrame(frame);
    });
  }

  window.LQShrine = { play };

  // Preview without DevTools: open the game with ?shrine=1 in the address bar and the cutscene plays once after loading.
  if (/[?&]shrine=1\b/.test(location.search)) window.addEventListener('load', () => setTimeout(() => play(), 800));
})();
