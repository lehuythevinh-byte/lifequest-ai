  function setupScene(sc, actors) {

      H = sc.area.h * 32;
    const R = rnd(77);
    const splashes = [];
    for (let i = 0; i < 46; i++)
      splashes.push({
        x: (12 + R() * 48) * 32,
        y: (14 + R() * 32) * 32,
        r: 5 + R() * 12,
        c: PALETTE[i % PALETTE.length],
        d: R() * 1.6,
      });
    const notes = [];
    const marks = [
      { k: 'star', x: 31.7 * 32, y: 48.4 * 32, c: 0xff4f87 },
      { k: 'eye', x: 38.6 * 32, y: 46.6 * 32, c: 0x35c8d0 },
      { k: 'wave', x: 31.9 * 32, y: 45.2 * 32, c: 0x9470e8 },
    ];
    const st = { dark: 0, darkTo: 0, darkFrom: 0, darkT0: 0, marks: 0, marksT0: 0, paint: 0, paintT0: 0, music: 0, focus: null, focusT0: 0 };
    const fade = (to, ms) => {

    const npcAt = (id) => {

    };
    return {
      tick(time) {
        const now = performance.now();
        if (st.darkT0) {
          const p = Math.min(1, (now - st.darkT0) / st.darkMs);
          st.dark = st.darkFrom + (st.darkTo - st.darkFrom) * p;
        }
        // Paint splashes "land" one after another once the light comes in.
        if (st.paint) {

            const t = (now - n.t0) / 1000;
            if (t > 2.4) {
              notes.splice(i, 1);
              continue;

          }

          }
        }

      act(a, skipping) {

          st.darkTo = 0.8;
          st.darkT0 = 0;

          st.paint = 1;
          st.paintT0 = performance.now();
        }

          st.focus = { id: 'ivo', c: 0x42dce7 };
        }

          st.focus = { id: 'kai', c: 0xff8b54 };

          st.music = 0;
          st.marks = 0;
          st.dark = 0;
          st.darkT0 = 0;
        }
      },
    };

        en.textContent = '';
        clearInterval(typing);
        let n = 0;
        typing = setInterval(() => {
          n += 1;
          en.textContent = full.slice(0, n);
          if (n >= full.length) clearInterval(typing);
        }, 28);
      };

        if (ended) return;

          clearInterval(typing);
          en.textContent = full;
          return;

        i += 1;
        if (i >= FILM.length) return finish();
        show();

        ended = true;
        clearInterval(typing);
        window.removeEventListener('keydown', onKey, true);

        }, 850);
      };
      const onKey = (e) => {

          next();
        } else if (e.key === 'Escape') {
          e.preventDefault();
          finish();
        }
      };
      window.addEventListener('keydown', onKey, true);

      root.querySelector('.skip').addEventListener('click', finish);

      setTimeout(() => {
        if (ended) return;
        card.classList.remove('show');
        cap.classList.add('show');
        next();
      }, 2800);
    });
  }
