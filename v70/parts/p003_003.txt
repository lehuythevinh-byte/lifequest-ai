  const items=sc.inside.spec.items ||[],unit=32*sc.inside.k;

  const allowed=new Set(['desk','table','shelf','counter','forge','stove','board','clock','paint','glasscase','rack','bed','altar','crystal','chair','schooldesk','glass','bookshelf']);
  let best=null,dist=Infinity;

   const x=Yr+(item.x+item.w/2)*unit,y=Xr+(item.y+item.h/2)*unit;

  }
  return best;
 }
