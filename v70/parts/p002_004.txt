        function lqNpcExtractJson(value) {
          if (value && typeof value === 'object') return value;
          if (typeof value !== 'string') return null;
          let text = value.trim();
          for (let round = 0; round < 3; round++) {
            text = text
              .replace(/^```(?:json|javascript|js|text)?\s*\n?/i, '')
              .replace(/\n?```\s*$/, '')
              .trim();

              const parsed = JSON.parse(text);
              if (typeof parsed === 'string') {
                text = parsed.trim();
                continue;
              }
              return parsed;
            } catch {}

              end = text.lastIndexOf('}');
            if (begin >= 0 && end > begin) {
              try {
                const parsed = JSON.parse(text.slice(begin, end + 1));
                return parsed;
              } catch {}
            }
            break;
          }
          return null;
        }
        function lqNpcSafeText(value, fallback = '') {

          const orig = raw.trim();

          if (parsed && typeof parsed === 'object') {
            const content =

            if (typeof content === 'string' && content !== orig)
              return lqNpcSafeText(content, fallback);
            return fallback;
          }
          raw = orig
            .replace(/^```(?:json|text|markdown)?\s*/i, '')
            .replace(/\s*```$/, '')
            .trim();
          // Sometimes the provider emits malformed JSON with a valid reply property.
          const embedded = raw.match(/["']reply["']\s*:\s*"((?:\\.|[^"\\])*)"/i);
          if (embedded) {

              raw,

          )
            return fallback;
          if (/^\s*[\[{]/.test(raw) && /[\]}]\s*$/.test(raw)) return fallback;
          if (
            /\b(?:console\.log|JSON\.parse|document\.getElementById|=>|```|\{\s*"reply"|\\n\s*"goalDone")/.test(
              raw,

            .replace(/^\s*(?:reply|greeting|answer)\s*:\s*/i, '')
            .replace(/^['“”"]+|['“”"]+$/g, '');
