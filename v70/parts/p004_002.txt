        function findMention(s, entries, aliases) {
          const hits = [];
          for (const x of entries || []) {
            let names = [
              x.id,
              x.name,
              String(x.name)
                .replace(/\([^)]*\)/g, '')
                .replace(/^(mr|ms|dr|officer|father)\.? /i, ''),
              ...(aliases[x.id] || []),
              ...(aliases[x.name] || []),

              const n = fold(name);
              if (n.length > 2 && has(s, n)) hits.push({ x, len: n.length });
