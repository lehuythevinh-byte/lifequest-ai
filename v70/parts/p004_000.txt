// Architectural strokes are composed in the 0..100 wall band; floor motifs are flat and passable.
function wallMark(d,x,y,k,p,scale=1){const {a,b,d:dark,glow}=p;
 if(k==='arched'){
  d.rect(x-17*scale,y-2*scale,34*scale,32*scale,dark,.90);
  d.oval(x,y+4*scale,28*scale,29*scale,b,.72);d.rect(x-13*scale,y+4*scale,26*scale,24*scale,b,.72);
  d.rect(x-9*scale,y+7*scale,18*scale,21*scale,dark,.85);d.oval(x,y+8*scale,18*scale,15*scale,dark,.85);
  d.line(x,y+5*scale,x,y+28*scale,glow,.65,2*scale);
 }else if(k==='leaf'){
  d.line(x,y+20*scale,x,y-6*scale,dark,.90,5*scale);
  d.oval(x-8*scale,y+1*scale,17*scale,13*scale,a,.86);d.oval(x+7*scale,y-4*scale,20*scale,16*scale,a,.88);
  d.oval(x-1*scale,y-9*scale,17*scale,10*scale,glow,.33);
 }else if(k==='dial'){
  d.circle(x,y,16*scale,dark,.86);d.circle(x,y,13*scale,b,.95);d.circle(x,y,9*scale,0x604832,.95);
  d.line(x,y,x+5*scale,y-4*scale,glow,1,2*scale);d.line(x,y,x-4*scale,y+1*scale,b,1,2*scale);
  for(let i=0;i<8;i++){const ang=i*Math.PI/4;d.rect(x+Math.cos(ang)*10*scale-1,y+Math.sin(ang)*10*scale-1,2,2,glow,.89)}
 }else if(k==='conduit'){
  d.rect(x-18*scale,y-14*scale,36*scale,29*scale,dark,.83);d.rect(x-14*scale,y-10*scale,28*scale,20*scale,a,.79);
  for(let z=0;z<3;z++)d.rect(x-10*scale,y-5*scale+z*6*scale,20*scale,2*scale,glow,z===1?.74:.42);
  d.circle(x+10*scale,y+12*scale,3*scale,b,1);
 }else if(k==='frame'){
  d.rect(x-16*scale,y-13*scale,32*scale,27*scale,dark,.98);d.rect(x-13*scale,y-10*scale,26*scale,21*scale,b,.88);
  d.rect(x-10*scale,y-7*scale,20*scale,15*scale,a,.79);d.line(x-10*scale,y+3*scale,x+10*scale,y-5*scale,glow,.6,2*scale);
 }else if(k==='loom'){
  d.rect(x-15*scale,y-11*scale,4*scale,26*scale,dark,.9);d.rect(x+11*scale,y-11*scale,4*scale,26*scale,dark,.9);
  d.line(x-12*scale,y-11*scale,x+12*scale,y-11*scale,b,.9,3*scale);d.line(x-11*scale,y+11*scale,x+11*scale,y+11*scale,b,.9,3*scale);
  for(let i=0;i<6;i++)d.line(x-10*scale+i*4*scale,y-8*scale,x-10*scale+i*4*scale,y+10*scale,i%2?glow:a,.8,1*scale);
 }
}
function emblem(p){return p.dominant==='branch'?'leaf':p.dominant==='gantry'?'conduit':p.dominant==='clock'||p.dominant==='arcade'?'dial':p.dominant==='glass'?'conduit':p.dominant==='truss'?'loom':'frame'}
function heroWalls(d,p){const W=p.W,cx=W/2,c=p.region;
 switch(p.dominant){
 case 'branch':{
  // Organic ceilings use curved silhouettes made from layered triangular root supports.
  for(let side of [-1,1]){let xx=side<0?30:W-30;
   d.line(xx,93,xx-side*28,34,0x3e5836,.99,15);d.line(xx,79,xx-side*56,45,0x69805b,.91,9);
   d.oval(xx-side*39,44,92,46,0x426c49,.92);d.oval(xx-side*47,34,58,37,0x7fa36b,.89);
   d.line(xx,90,cx+side*28,91,c.d,.65,4);
  }
  for(let j=0;j<5;j++)wallMark(d,45+j*(W-90)/4,64,'leaf',c,.78);
  break;}
 case 'gantry':{
  d.rect(16,16,W-32,20,c.d,.94);d.rect(22,20,W-44,5,c.b,.72);
  for(let j=0;j<6;j++){let x=23+j*(W-46)/5;
   d.rect(x,36,6,61,c.d,.84);d.rect(x+2,36,2,59,c.b,.74);
   if(j<5)d.line(x+6,39,x+(W-46)/5,73,c.a,.5,3);
  }
  for(let j=0;j<4;j++)wallMark(d,72+j*(W-144)/3,66,'conduit',c,.78);
  break;}
 case 'arch':{
  d.rect(17,13,W-34,15,c.d,.96);d.rect(20,17,W-40,5,c.b,.79);
  for(let j=0;j<5;j++){const x=35+j*(W-70)/4;
   d.rect(x-10,42,20,55,c.d,.9);d.rect(x-7,45,14,48,c.a,.84);d.rect(x-13,36,26,10,c.b,.85);
  }
  for(let j=0;j<4;j++)wallMark(d,77+j*(W-154)/3,64,'arched',c,.78);
  break;}
 case 'clock':case 'arcade':{
  for(let j=0;j<5;j++){let x=27+j*(W-54)/4;
   d.rect(x-10,18,20,75,c.d,.91);d.rect(x-7,21,14,70,c.a,.56);

 case 'glass':{
  for(let j=0;j<5;j++){let x=24+j*(W-48)/4;d.rect(x,13,5,82,c.d,.87);d.rect(x+5,18,(W-48)/4-9,49,c.a,.35);
   d.line(x+6,69,x+(W-48)/4-3,69,c.glow,.67,2);}
  for(let j=0;j<4;j++)wallMark(d,80+j*(W-160)/3,80,'conduit',c,.64);break;}
 case 'sawtooth':{
  for(let j=0;j<6;j++){let x=23+j*(W-46)/5;d.tri(x,20,x+35,20,x+20,62,[0xa6869f,0x9d88b1,0xa98f85][j%3],.77);}
  for(let j=0;j<4;j++)wallMark(d,82+j*(W-164)/3,76,'frame',c,.85);break;}
 case 'truss':{
  for(let j=0;j<5;j++){let x=22+j*(W-44)/4;d.line(x,92,x+24,23,c.d,.75,8);d.line(x+2,85,x+25,28,c.b,.6,2);}
  for(let j=0;j<4;j++)wallMark(d,80+j*(W-160)/3,79,'loom',c,.78);break;}
 default:{
  for(let j=0;j<4;j++){let x=35+j*(W-70)/3;d.rect(x-13,25,26,67,c.d,.86);d.rect(x-9,30,18,54,c.a,.68);}
  wallMark(d,cx,66,'frame',c,1);break;}
 }
}
function floorStage(d,p){const W=p.W,H=p.H,c=p.region,cy=lim(H*.55,160,H-118),cx=p.floorCenterX;
 // Floor embellishment is passable and visually subordinate to the actual furniture and door.

  const rx=Math.min(132,W*.29),ry=Math.min(80,(H-130)*.30);
  d.oval(cx,cy,rx*2+15,ry*2+15,c.d,.24);d.oval(cx,cy,rx*2,ry*2,0xa88755,.28);

function occupationIdentity(d,p){
 const W=p.W,H=p.H,c=p.region,v=p.layoutVariant;
 const flip=(v%2)?1:-1;
 const anchor=W*(flip>0?.75:.25),second=W-anchor;
 const upper=130+v%3*8;
 if(p.role==='family'){
   // Family rooms: conversational rugs, asymmetrical alcove seams, low wall textile.
   d.oval(anchor,upper+48,Math.min(93,W*.24),50,c.b,.16);
   d.oval(anchor,upper+48,Math.min(70,W*.18),35,c.d,.17);
   for(let k=0;k<5;k++)d.line(anchor-28+k*14,upper+26,anchor-20+k*13,upper+62,c.a,.26,2);
   wallMark(d,second,66,'loom',c,.57);
 }else if(p.role==='work'){
   // Shops and workshops: a visual flow lane points from production to display.
   const x1=flip>0?35:W-35,x2=flip>0?W*.37:W*.63;
   for(let j=0;j<3;j++){
    const xx=x1+(x2-x1)*(j+1)/4;
    d.rect(xx-13,upper+j*26,26,4,c.d,.33);
    d.rect(xx-8,upper+j*26+5,16,2,c.b,.35);
   }
   wallMark(d,anchor,70,p.region.kind==='merchant'?'dial':'conduit',c,.62);
 }else if(p.role==='learning'){
   // Study/archive: a vertical reference strip and a reading zone on the wall.
   for(let j=0;j<5;j++){
    const xx=anchor-30+j*13;
    d.rect(xx,39+(j%2)*5,8,47,c.d,.62);
    d.rect(xx+2,42+(j%2)*5,4,41,j%2?c.b:c.a,.45);
   }
   d.rect(second-32,upper,65,4,c.b,.25);
   wallMark(d,second,63,'frame',c,.62);
 }else if(p.role==='tech'){
   // Routing diagram is an inset on the floor, not an extra blocking console.
   for(let j=0;j<4;j++){
    const y=upper+j*19;
    d.line(anchor-28,y,anchor+18,y,c.glow,.26,2);
    d.circle(anchor+20,y,2,c.b,.42);
   }
   wallMark(d,second,66,'conduit',c,.7);
 }else{
   // Communal gathering places: radial floor inlay, ceremonial wall display.
   d.ring(anchor,upper+48,31,c.b,.24,3);
   for(let j=0;j<7;j++){
    const a=j*Math.PI*2/7;
    d.rect(anchor+Math.cos(a)*35-2,upper+48+Math.sin(a)*35-2,4,4,c.a,.39);
   }
   wallMark(d,second,66,p.dominant==='arch'?'arched':'frame',c,.64);
 }
 // Individual construction seams, derived from a stable building ID instead of clutter.
 for(let j=0;j<3;j++){
   const x=26+((v*41+j*103)%(Math.max(50,W-52)));
   d.rect(x,21,3,23+(v+j)%4*7,c.d,.25);
   d.rect(x+3,24,2,13+(v+j)%4*7,c.b,.31);
 }
}
